      SELECT id, "tenantId"

      WHERE "tenantId" = ${tenantId}
        AND "isDeleted" = false
        AND emails IS NOT NULL
        AND EXISTS (

          WHERE lower(elem->>'value') = lower(${senderEmail})

        AND emails IS NOT NULL
        AND EXISTS (

**PILLAR_ITEMS definition** (same as Sidebar.tsx — duplicate is acceptable for ~5 items, avoids a shared file for a trivial constant):
```typescript
interface PillarItem {
  href: string;
  icon: React.ElementType;
  labelKey: string;
}

const PILLAR_ITEMS: PillarItem[] = [
  { href: '/dashboard', icon: Home, labelKey: 'home' },
  { href: '/dashboard/email', icon: Mail, labelKey: 'mail' },
  { href: '/dashboard/calendar', icon: Calendar, labelKey: 'calendar' },
  { href: '/dashboard/contacts', icon: Users, labelKey: 'contacts' },
  { href: '/dashboard/tasks', icon: CheckSquare, labelKey: 'tasks' },
];
```

**Component:**
```typescript
export function MobileTabBar() {
  const theme = useTheme();
  const router = useRouter();
  const t = useTranslations();
  const defaultOrder = PILLAR_ITEMS.map(p => p.href);
  const { order } = useSidebarOrder(defaultOrder);
  // order reflects desktop DnD reorder — mobile shows same order but no DnD

  // Resolve current active path for BottomNavigation value
  const currentPath = PILLAR_ITEMS.find(p => router.pathname === p.href)?.href
    ?? (router.pathname === '/dashboard/settings' ? '/dashboard/settings' : false);

  const orderedPillars = order
    .map(href => PILLAR_ITEMS.find(p => p.href === href))
    .filter(Boolean) as PillarItem[];

  return (
    <BottomNavigation
      value={currentPath}
      onChange={(_, newValue) => router.push(newValue)}
      sx={{
        position: 'fixed',
        bottom: 0,
        left: 0,
        right: 0,
        zIndex: theme.zIndex.appBar,
        borderTop: '1px solid',
        borderColor: alpha(theme.palette.divider, 0.1),
        height: 56,
        bgcolor: 'background.paper',
        pb: 'env(safe-area-inset-bottom, 0px)',
      }}
    >
      {orderedPillars.map((item) => {
        const IconComp = item.icon;
        return (
          <BottomNavigationAction
            key={item.href}
            value={item.href}
            icon={<IconComp size={24} strokeWidth={1.7} />}
            label={t.nav[item.labelKey as keyof typeof t.nav]}
            showLabel={false}
            aria-label={t.nav[item.labelKey as keyof typeof t.nav]}
          />
        );
      })}
      <BottomNavigationAction
        value="/dashboard/settings"
        icon={<Settings size={24} strokeWidth={1.7} />}
        label={t.nav.settings}
        showLabel={false}
        aria-label={t.nav.settings}
      />
    </BottomNavigation>
  );
}
```

**Key specs from UI-SPEC:**
- Height: 56px
- Position: fixed, bottom: 0
- z-index: `theme.zIndex.appBar` (1100)
- Background: `background.paper`
- Border top: `1px solid alpha(divider, 0.1)`
- Icon size: 24px, strokeWidth 1.7
- showLabel: false (icon-only for density)
- Active icon color: `primary.main` (MUI BottomNavigation handles this by default)
- Inactive icon color: `text.secondary` (MUI default)
- Safe area: `pb: 'env(safe-area-inset-bottom, 0px)'`
- No DnD on mobile (static order, but reads from useSidebarOrder for consistent ordering with desktop)

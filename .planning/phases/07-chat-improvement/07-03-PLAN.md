Rewrite TypingIndicator to use the Shimmer component instead of CSS bouncing dots.

**Current implementation uses:**
- CSS `@keyframes typing-bounce` with bouncing dots
- Basic MUI Box with animation

**New implementation:**
```tsx
'use client';

import React from 'react';
import { Box, alpha, useTheme } from '@mui/material';
import { Bot } from 'lucide-react';
import { Shimmer } from '@/src/components/ai-elements/shimmer';

interface TypingIndicatorProps {
  /** Whether the indicator is active/visible */
  isActive: boolean;
  /** Dynamic action text from agent activity (e.g., "Searching emails...") */
  action?: string;
}

export function TypingIndicator({ isActive, action }: TypingIndicatorProps) {
  const theme = useTheme();

  if (!isActive) return null;

  // Use action text if available, otherwise default to "Thinking..."
  const displayText = action || 'Thinking...';

  return (
    <Box
      sx={{
        display: 'flex',
        gap: 1.5,
        alignItems: 'flex-start',
        py: 1,
      }}
    >
      {/* Avatar - matches assistant message style */}
      <Box
        sx={{
          width: 36,
          height: 36,
          borderRadius: 2,
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          flexShrink: 0,
          background: `linear-gradient(135deg, ${theme.palette.primary.main} 0%, ${theme.palette.secondary.main} 100%)`,
          color: 'white',
        }}
      >
        <Bot size={18} />
      </Box>

      {/* Shimmer text */}
      <Box
        sx={{
          display: 'flex',
          alignItems: 'center',
          minHeight: 36,
          px: 1,
        }}
      >
        <Shimmer duration={1.5} className="text-sm text-muted-foreground">
          {displayText}
        </Shimmer>
      </Box>
    </Box>
  );
}

export default TypingIndicator;
```

**Key changes:**
1. Remove all CSS keyframes and bouncing dot animation
2. Import Shimmer from ai-elements
3. Use action prop for dynamic text ("Searching emails...", "Reading thread...")
4. Fallback to "Thinking..." when no specific action
5. Keep avatar styling consistent with assistant messages
6. Use Shimmer with 1.5s duration for smooth animation

```tsx
'use client';

import React from 'react';
import { Copy, Pencil, Download, Check } from 'lucide-react';
import {
  Artifact,
  ArtifactHeader,
  ArtifactTitle,
  ArtifactActions,
  ArtifactAction,
  ArtifactContent,
} from '@/src/components/ai-elements/artifact';

interface ArtifactPanelProps {
  title: string;
  type: 'email-draft' | 'code' | 'document';
  content: React.ReactNode;
  rawContent?: string;  // For copy functionality
  onEdit?: () => void;
  onCopy?: () => void;
}

export function ArtifactPanel({
  title,
  type,
  content,
  rawContent,
  onEdit,
  onCopy,
}: ArtifactPanelProps) {
  const [copied, setCopied] = React.useState(false);

  const handleCopy = React.useCallback(async () => {
    if (!rawContent) return;
    try {
      await navigator.clipboard.writeText(rawContent);
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
      onCopy?.();
    } catch (err) {
      console.error('[ArtifactPanel] Copy failed:', err);
    }
  }, [rawContent, onCopy]);

  const getTypeLabel = () => {
    switch (type) {
      case 'email-draft': return 'Draft Email';
      case 'code': return 'Code';
      case 'document': return 'Document';
      default: return 'Content';
    }
  };

  return (
    <Artifact className="my-3">
      <ArtifactHeader>
        <div className="flex items-center gap-2">
          <span className="text-xs text-muted-foreground uppercase tracking-wide">
            {getTypeLabel()}
          </span>
          <ArtifactTitle>{title}</ArtifactTitle>
        </div>
        <ArtifactActions>
          {rawContent && (
            <ArtifactAction
              tooltip={copied ? 'Copied!' : 'Copy'}
              icon={copied ? Check : Copy}
              onClick={handleCopy}
            />
          )}
          {onEdit && (
            <ArtifactAction
              tooltip="Edit"
              icon={Pencil}
              onClick={onEdit}
            />
          )}
        </ArtifactActions>
      </ArtifactHeader>
      <ArtifactContent>
        {content}
      </ArtifactContent>
    </Artifact>
  );
}

export default ArtifactPanel;
```

```typescript
/**
 * Determine if a tool output should render as an Artifact panel.
 * Based on research recommendations:
 * - draft_email tool outputs -> Artifact
 * - Code blocks >10 lines -> Artifact
 * - Code with class/function definitions -> Artifact
 */
function shouldRenderAsArtifact(part: ToolPart): boolean {
  // Draft email tool outputs
  if ((part as { toolName?: string }).toolName === 'draft_email') {
    return true;
  }

  // Check output for code indicators
  const output = part.output;
  if (typeof output === 'string') {
    const lines = output.split('\n').length;
    const hasDefinitions = /^(class|function|export|import|interface|type|const|let|var)\s/m.test(output);

    if (lines > 10 || hasDefinitions) {
      return true;
    }
  }

  return false;
}

/**
 * Get artifact title from tool part.
 */
function getArtifactTitle(part: ToolPart): string {
  const toolName = (part as { toolName?: string }).toolName;
  if (toolName === 'draft_email') {
    return 'Draft Reply';
  }
  return 'Generated Content';
}
```

```typescript
/**
 * Group consecutive tool parts for ChainOfThought display.
 * Returns groups where each group with >1 tool should use ChainOfThought.
 */
function groupToolParts(parts: UIMessage['parts']): Array<{
  isChain: boolean;
  parts: UIMessage['parts'];
  startIndex: number;
}> {
  const groups: Array<{ isChain: boolean; parts: UIMessage['parts']; startIndex: number }> = [];
  let currentToolGroup: UIMessage['parts'] = [];
  let currentStartIndex = 0;

  parts.forEach((part, index) => {
    const isTool = part.type.startsWith('tool-') || part.type === 'dynamic-tool';

    if (isTool) {
      if (currentToolGroup.length === 0) {
        currentStartIndex = index;
      }
      currentToolGroup.push(part);
    } else {
      // Flush tool group if exists
      if (currentToolGroup.length > 0) {
        groups.push({
          isChain: currentToolGroup.length > 1,
          parts: currentToolGroup,
          startIndex: currentStartIndex,
        });
        currentToolGroup = [];
      }
      // Add non-tool part as single item
      groups.push({
        isChain: false,
        parts: [part],
        startIndex: index,
      });
    }
  });

  // Flush remaining tool group
  if (currentToolGroup.length > 0) {
    groups.push({
      isChain: currentToolGroup.length > 1,
      parts: currentToolGroup,
      startIndex: currentStartIndex,
    });
  }

  return groups;
}

/**
 * Get icon for a tool based on its name.
 */
function getToolIcon(toolName?: string): React.ElementType {
  if (!toolName) return Circle;
  if (toolName.includes('search') || toolName.includes('list')) return Search;
  if (toolName.includes('email') || toolName.includes('mail')) return Mail;
  if (toolName.includes('calendar')) return Calendar;
  if (toolName.includes('contact')) return Users;
  if (toolName.includes('read') || toolName.includes('get')) return FileText;
  return Circle;
}

/**
 * Get tool status for ChainOfThought step.
 */
function getToolStatus(state?: string): 'complete' | 'active' | 'pending' {
  if (state === 'complete' || state === 'result') return 'complete';
  if (state === 'streaming' || state === 'running') return 'active';
  return 'pending';
}
```

```typescript
/**
 * Extract source references from tool outputs.
 * RAG tools (search_emails, list_contacts, etc.) return items that can be cited.
 */
function extractSources(parts: UIMessage['parts']): SourceReference[] {
  const sources: SourceReference[] = [];

  for (const part of parts) {
    if (part.type.startsWith('tool-') || part.type === 'dynamic-tool') {
      const toolPart = part as ToolPart;
      const toolName = (toolPart as { toolName?: string }).toolName;
      const output = toolPart.output;

      // Email search results
      if (toolName?.includes('email') && Array.isArray(output)) {
        for (const item of output.slice(0, 3)) { // Max 3 sources
          if (item.id && item.subject) {
            sources.push({
              id: item.id,
              type: 'email',
              title: item.subject,
              snippet: item.snippet || item.from,
            });
          }
        }
      }

      // Contact search results
      if (toolName?.includes('contact') && Array.isArray(output)) {
        for (const item of output.slice(0, 3)) {
          if (item.id && (item.name || item.displayName)) {
            sources.push({
              id: item.id,
              type: 'contact',
              title: item.name || item.displayName,
              snippet: item.email || item.company,
            });
          }
        }
      }
    }
  }

  return sources;
}

/**
 * Get icon for source type.
 */
function getSourceIcon(type: string): React.ElementType {
  switch (type) {
    case 'email': return MailIcon;
    case 'contact': return User;
    default: return FileText;
  }
}
```

**beforeAll — guard against already-completed onboarding + obtain storageState:**

This spec is once-per-seed. If `test@pmsync.local` has already completed onboarding (e.g. the spec was run previously without re-seeding), navigating to `/` will redirect to `/dashboard` instead of `/onboarding`. Detect this and skip the entire suite with a diagnostic message rather than failing cryptically.

```typescript
import { test, expect } from '@playwright/test';
import { authenticator } from 'otplib';

const ONBOARDING_USER_EMAIL = 'test@pmsync.local';
const ONBOARDING_USER_PASSWORD = 'UserPassword123!';
const ONBOARDING_AUTH_FILE = '/tmp/onboarding-user.json';

test.describe('Onboarding wizard', () => {
  test.beforeAll(async ({ browser }) => {
    const ctx = await browser.newContext();
    const page = await ctx.newPage();
    await page.addInitScript(() => {
      localStorage.setItem('pmsync:cookie-consent', JSON.stringify({ necessary: true, analytics: false, marketing: false }));
    });
    await page.goto('/auth/login');
    await page.fill('input[name="email"], input[type="email"]', ONBOARDING_USER_EMAIL);
    await page.fill('input[name="password"], input[type="password"]', ONBOARDING_USER_PASSWORD);
    await page.click('button[type="submit"]');
    await page.waitForURL(/\/(onboarding|dashboard)/, { timeout: 15_000 });

    // Guard: if middleware sent us to /dashboard, onboardingComplete is already true.
    // No reset API exists — only `cd backend && npx prisma db seed` can reset it.
    if (page.url().includes('/dashboard')) {
      await ctx.close();
      test.skip(
        true,
        'test@pmsync.local already completed onboarding — run `cd backend && npx prisma db seed` to reset'
      );
      return;
    }

    await ctx.storageState({ path: ONBOARDING_AUTH_FILE });
    await ctx.close();
  });

  test('completes full onboarding wizard', async ({ browser }) => {
    const ctx = await browser.newContext({ storageState: ONBOARDING_AUTH_FILE });
    const page = await ctx.newPage();
    // ... wizard steps below ...
    await ctx.close();
  });
});
```

    ```typescript
    import { execSync } from 'child_process';
    import * as path from 'path';

    describe('Rate Tier Audit', () => {
      it('no @Throttle decorator uses the "default" key anywhere in controllers', () => {
        const result = execSync(
          'grep -rn "Throttle.*default" --include="*.controller.ts" src/ || true',
          { cwd: path.resolve(__dirname, '..'), encoding: 'utf-8' },
        );
        // Filter out comments and imports
        const matches = result.split('\n').filter(line =>
          line.includes('@Throttle') && line.includes('default')
        );
        expect(matches).toHaveLength(0);
      });

      it('no orphaned @Throttle imports remain in controllers that only use @RateTierDecorator', () => {
        // Verify that if a controller imports Throttle, it actually uses @Throttle()
        // (If @SkipThrottle is used, Throttle import may still be needed)
        const result = execSync(
          'grep -rl "import.*Throttle.*throttler" --include="*.controller.ts" src/ || true',
          { cwd: path.resolve(__dirname, '..'), encoding: 'utf-8' },
        );
        const files = result.trim().split('\n').filter(Boolean);
        for (const file of files) {
          const content = execSync(`cat "${file}"`, { encoding: 'utf-8' });
          const hasThrottleUsage = content.includes('@Throttle(') || content.includes('@SkipThrottle');
          // If file imports from @nestjs/throttler, it should actually use it
          if (content.includes("from '@nestjs/throttler'")) {
            expect(hasThrottleUsage).toBe(true);
          }
        }
      });
    });
    ```

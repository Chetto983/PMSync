**Frontend — Add `useRespondToCalendarEvent` mutation in use-calendar.ts:**

Add after the existing `useDeleteCalendar` hook:
```typescript
/**
 * Hook for responding to a calendar event invite (RSVP)
 * Two-step: looks up source email, then calls email RSVP endpoint
 */
export function useRespondToCalendarEvent() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async ({ eventId, response }: { eventId: string; response: 'accepted' | 'declined' | 'tentative' }) => {
      // Step 1: Get the source email for this calendar event
      const { data } = await calendarApi.getEventSourceEmail(eventId);
      if (!data.emailId) {
        throw new Error('No email invite found for this calendar event');
      }
      // Step 2: Map response format (CalendarEventDetail uses 'accepted'/'declined'/'tentative',
      // but the email API expects 'accept'/'decline'/'tentative')
      const responseMap: Record<string, string> = {
        accepted: 'accept',
        declined: 'decline',
        tentative: 'tentative',
      };
      const { emailsApi } = await import('@/lib/api/emails');
      await emailsApi.respondToCalendarInvite(data.emailId, { response: responseMap[response] as 'accepted' | 'declined' | 'tentative' });
      return { eventId, emailId: data.emailId };
    },
    onSuccess: ({ eventId, emailId }) => {
      queryClient.invalidateQueries({ queryKey: calendarKeys.eventDetail(eventId) });
      queryClient.invalidateQueries({ queryKey: calendarKeys.events() });
      // Also invalidate the email detail if cached
      queryClient.invalidateQueries({ queryKey: ['emails', 'detail', emailId] });
    },
    onError: (error) => {
      logger.error('Failed to respond to calendar event', {
        error: error instanceof Error ? error.message : String(error),
      });
    },
  });
}
```

Import `emailsApi` dynamically inside the mutationFn (to avoid circular dependency between calendar and email modules).

Note on response format: The `RSVPResponse` type from `RSVPButtons.tsx` is `'accepted' | 'declined' | 'tentative'`, but the email API's `RespondToInviteDto` expects `response: 'accepted' | 'declined' | 'tentative'` (check line 283 of emails.ts — it's the same format). Actually, re-reading emails.ts line 283: `response: 'accepted' | 'declined' | 'tentative'`. But the backend controller at email-calendar-invite.controller.ts line 54 expects `'accept' | 'decline' | 'tentative'` (without -ed suffix). The frontend emailsApi sends whatever is in the DTO. Check both sides carefully and map accordingly.

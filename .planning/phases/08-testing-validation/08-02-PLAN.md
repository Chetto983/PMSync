```typescript
async function scenario2_RedisLockContention(): Promise<{ passed: boolean; details: string }> {
  const WORKER_COUNT = 20;
  const LOCK_KEY = 'stress-test:contention';
  const LOCK_TTL_MS = 5000;
  const MAX_ATTEMPTS = 10;
  const BACKOFF_BASE_MS = 50;

  // Spawn 20 workers competing for same lock
  const workers = Array.from({ length: WORKER_COUNT }, (_, i) =>
    competeLock(i, LOCK_KEY, LOCK_TTL_MS, MAX_ATTEMPTS, BACKOFF_BASE_MS)
  );

  const results = await Promise.all(workers);

  // Calculate success rate
  const successes = results.filter(r => r.acquired).length;
  const successRate = (successes / WORKER_COUNT) * 100;

  // Pass if 80%+ success rate (16/20 workers)
  const passed = successRate >= 80;
  return {
    passed,
    details: `${successes}/${WORKER_COUNT} workers acquired lock (${successRate.toFixed(1)}% success rate)`
  };
}

async function competeLock(workerId: number, key: string, ttl: number, maxAttempts: number, backoffBase: number) {
  let attempts = 0;
  while (attempts < maxAttempts) {
    const acquired = await redis.set(key, `worker-${workerId}`, 'PX', ttl, 'NX');
    if (acquired === 'OK') {
      // Simulate work (100ms)
      await new Promise(r => setTimeout(r, 100));
      await redis.del(key);
      return { workerId, acquired: true, attempts };
    }
    attempts++;
    // Exponential backoff with jitter
    const delay = backoffBase * Math.pow(2, attempts) + Math.random() * 50;
    await new Promise(r => setTimeout(r, delay));
  }
  return { workerId, acquired: false, attempts };
}
```

```typescript
async function scenario3_LargeBatchOperation(): Promise<{ passed: boolean; details: string }> {
  const BATCH_SIZE = 1000;
  const TIMEOUT_MS = 60000; // 60 seconds baseline

  // Get test provider
  const provider = await getTestProvider();

  const startTime = Date.now();

  // Create 1000 test emails in batches of 100 (to avoid query timeout)
  for (let batch = 0; batch < 10; batch++) {
    const emails = Array.from({ length: 100 }, (_, i) => ({
      externalId: `batch-test-${Date.now()}-${batch}-${i}`,
      providerId: provider.id,
      tenantId: provider.tenantId,
      folder: 'INBOX',
      subject: `Batch test email ${batch * 100 + i}`,
      from: 'test@example.com',
      receivedAt: new Date(),
    }));

    await prisma.email.createMany({ data: emails, skipDuplicates: true });
  }

  const duration = Date.now() - startTime;

  // Verify count
  const count = await prisma.email.count({
    where: { externalId: { startsWith: 'batch-test-' } }
  });

  // Cleanup
  await prisma.email.deleteMany({
    where: { externalId: { startsWith: 'batch-test-' } }
  });

  const passed = duration < TIMEOUT_MS && count >= BATCH_SIZE;
  return {
    passed,
    details: `Created ${count} emails in ${duration}ms (baseline: ${TIMEOUT_MS}ms)`
  };
}
```

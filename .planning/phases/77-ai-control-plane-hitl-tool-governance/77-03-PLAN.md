```typescript
export type ToolDomain = 'email' | 'calendar' | 'contacts' | 'knowledge' | 'productivity' | 'system';
export type RiskLevel = 'LOW' | 'MEDIUM' | 'HIGH';

export interface IAiTool {
  name: string;
  description: string;
  parameters: Record<string, unknown>;
  domain: ToolDomain;
  riskLevel: RiskLevel;
  requiresApproval: boolean;
  inputSchema: unknown;
  execute(input: Record<string, unknown>, context: IToolContext): Promise<unknown>;
}

export interface IToolRegistryService {
  registerTool(tool: IAiTool): void;
  getTool(name: string): IAiTool | undefined;
  getAllTools(): IAiTool[];
  getToolNames(): string[];
}
```

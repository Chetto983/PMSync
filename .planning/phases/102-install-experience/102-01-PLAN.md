function subscribe(callback: () => void): () => void {

  window.addEventListener('offline', callback);
  return () => {

    window.removeEventListener('offline', callback);
  };
}

function getSnapshot(): boolean {
  return navigator.onLine;
}

function getServerSnapshot(): boolean {
  return true;
}

export function useOnlineStatus(): boolean {
  return useSyncExternalStore(subscribe, getSnapshot, getServerSnapshot);
}

Create `frontend/hooks/use-install-prompt.ts` following the exact `useSyncExternalStore` pattern from `use-online-status.ts`. Key implementation details:

1. **Types** at top of file:
```typescript
interface BeforeInstallPromptEvent extends Event {
  prompt(): Promise<{ outcome: 'accepted' | 'dismissed' }>;
  readonly userChoice: Promise<{ outcome: 'accepted' | 'dismissed' }>;
}

declare global {
  interface WindowEventMap {
    beforeinstallprompt: BeforeInstallPromptEvent;
    appinstalled: Event;
  }
}

export interface UseInstallPromptReturn {
  shouldShow: boolean;
  isIOS: boolean;
  triggerInstall: () => Promise<'accepted' | 'dismissed' | null>;
  dismiss: () => void;
}
```

2. **Module-level state** (outside React, same as use-online-status pattern):
- `let deferredPrompt: BeforeInstallPromptEvent | null = null;`
- `let installed = false;`
- `const listeners = new Set<() => void>();`
- `notify()` function that calls all listeners
- Register `beforeinstallprompt` listener at module scope (guarded by `typeof window !== 'undefined'`). Call `e.preventDefault()`, store event, call `notify()`.
- Register `appinstalled` listener at module scope. Set `deferredPrompt = null`, `installed = true`, call `notify()`.

3. **Helper functions** (module-level):
- `getIsStandalone()`: Check `navigator.standalone` (Safari) first with `'standalone' in navigator` guard, then `window.matchMedia('(display-mode: standalone)').matches`. Return false if `typeof window === 'undefined'`.
- `getIsIOS()`: Check `/iPhone|iPad|iPod/.test(navigator.userAgent)` OR `(navigator.platform === 'MacIntel' && navigator.maxTouchPoints > 1)`. Return false if `typeof window === 'undefined'`.
- `isDismissedByUser()`: Read `localStorage.getItem('pmsync:install-prompt-dismissed')`, parse as int, return `Date.now() - timestamp < 30 * 24 * 60 * 60 * 1000`. Wrap in try/catch, return false on error.
- `dismissPrompt()`: `localStorage.setItem('pmsync:install-prompt-dismissed', Date.now().toString())`. Wrap in try/catch.

4. **subscribe function**: Add callback to `listeners` set. Return cleanup that removes from set.

5. **getSnapshot function**: Compute and return `{ shouldShow, isIOS }` where:
- `isIOS = getIsIOS()`
- `isStandalone = getIsStandalone()`
- `isDismissed = isDismissedByUser()`
- `shouldShow = !installed && !isStandalone && !isDismissed && (deferredPrompt !== null || isIOS)`
- Return a stable reference: use a module-level cached snapshot, only create new object if values changed.

6. **getServerSnapshot function**: Return `{ shouldShow: false, isIOS: false }` (constant).

7. **useInstallPrompt hook**: Use `useSyncExternalStore(subscribe, getSnapshot, getServerSnapshot)`. Add `triggerInstall` callback (useCallback): if `deferredPrompt`, call `deferredPrompt.prompt()`, get result, set `deferredPrompt = null`, call `notify()`, return `result.outcome`. Otherwise return null. Add `dismiss` callback: call `dismissPrompt()`, call `notify()`. Return `{ shouldShow, isIOS, triggerInstall, dismiss }`.

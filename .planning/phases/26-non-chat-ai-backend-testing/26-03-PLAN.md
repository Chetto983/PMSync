**Extend existing mistral-chat.handler.ts** (DO NOT create a separate mistral-generate.handler.ts):

The existing `mistral-chat.handler.ts` already handles both streaming and non-streaming requests on `https://api.mistral.ai/v1/chat/completions`. Per research pitfall #5, a single handler must own this URL. The non-streaming branch (lines 18-36) currently returns a hardcoded "Hello from MSW mock" response.

Add an exported `createGenerateHandler` factory function to the SAME file. This factory returns a handler that non-chat AI tests use via `server.use()` to override the default non-streaming behavior with fixture-specific responses:

```typescript
// Add to the BOTTOM of mistral-chat.handler.ts (after the existing mistralChatHandler export)

/**
 * Factory for fixture-driven non-streaming handler overrides.
 * Usage in tests: server.use(createGenerateHandler({ 'executive assistant': briefingFixture }))
 * Matches system prompt keywords to fixture data. Returns as non-streaming chat completion.
 */
export function createGenerateHandler(fixtureMap: Record<string, unknown>) {
  return http.post(
    'https://api.mistral.ai/v1/chat/completions',
    async ({ request }) => {
      const body = (await request.json()) as {
        stream?: boolean;
        messages?: Array<{ role: string; content: string }>;
      };

      // Only handle non-streaming requests; streaming requests passthrough (MSW returns undefined -> next handler)
      if (body.stream) return;

      const systemMsg = body.messages?.find(m => m.role === 'system')?.content || '';
      let responseContent = JSON.stringify({ error: 'no matching fixture' });

      for (const [key, fixture] of Object.entries(fixtureMap)) {
        if (systemMsg.toLowerCase().includes(key.toLowerCase())) {
          responseContent = JSON.stringify(fixture);
          break;
        }
      }

      return HttpResponse.json({
        id: 'chatcmpl-msw-generate',
        object: 'chat.completion',
        model: 'mistral-small-latest',
        choices: [{ index: 0, message: { role: 'assistant', content: responseContent }, finish_reason: 'stop' }],
        usage: { prompt_tokens: 100, completion_tokens: 50, total_tokens: 150 },
      });
    },
  );
}
```

DO NOT modify `index.ts` -- the default handlers array stays unchanged. The `createGenerateHandler` is used ONLY via `server.use()` in individual test files to temporarily override the default non-streaming response for that test's MSW server instance. This avoids handler ordering issues entirely.

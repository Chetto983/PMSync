Structure:
```typescript
import { test, expect } from './fixtures';
import AxeBuilder from '@axe-core/playwright';

test.describe('Accessibility', () => {
  test('email list page has no critical WCAG violations', async ({ authedPage }) => {
    await authedPage.goto('/dashboard/email');
    await authedPage.waitForSelector('[data-testid="thread-list-item"]');

    const results = await new AxeBuilder({ page: authedPage })
      .withTags(['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa'])
      .analyze();

    // Filter to only critical/serious violations
    const critical = results.violations.filter(v =>
      v.impact === 'critical' || v.impact === 'serious'
    );

    expect(critical).toEqual([]);
  });

  test('compose page has no critical WCAG violations', async ({ authedPage }) => {
    // Mock send endpoint
    await authedPage.route('**/api/v1/email/send', route =>
      route.fulfill({ status: 200, body: '{}' })
    );

    await authedPage.goto('/dashboard/email/compose');
    await authedPage.waitForSelector('[data-testid="compose-body"]');

    const results = await new AxeBuilder({ page: authedPage })
      .withTags(['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa'])
      .analyze();

    const critical = results.violations.filter(v =>
      v.impact === 'critical' || v.impact === 'serious'
    );

    expect(critical).toEqual([]);
  });

  test('thread view has no critical WCAG violations', async ({ authedPage }) => {
    await authedPage.goto('/dashboard/email');
    await authedPage.waitForSelector('[data-testid="thread-list-item"]');

    // Open first email
    await authedPage.locator('[data-testid="thread-list-item"]').first().click();
    await authedPage.waitForSelector('[data-testid="reading-pane"]');

    const results = await new AxeBuilder({ page: authedPage })
      .withTags(['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa'])
      .analyze();

    const critical = results.violations.filter(v =>
      v.impact === 'critical' || v.impact === 'serious'
    );

    expect(critical).toEqual([]);
  });
});
```

  <action>
If Task 1 reveals critical WCAG violations, fix them:

**Common fixes:**

1. **Missing form labels**: Add aria-label or htmlFor
```tsx
<input aria-label="Search emails" ... />
// or
<label htmlFor="search-input">Search</label>
<input id="search-input" ... />
```

2. **Color contrast**: Use theme tokens instead of hardcoded colors
```tsx
// Bad
<Typography sx={{ color: '#999' }}>
// Good
<Typography color="text.secondary">
```

3. **Missing alt text**: Add alt to images
```tsx
<img alt="User avatar" ... />
// or for decorative
<img alt="" role="presentation" ... />
```

4. **Missing button names**: Add aria-label to icon buttons
```tsx
<IconButton aria-label="Close dialog">
  <CloseIcon />
</IconButton>
```

5. **Focus management**: Ensure modals trap focus
```tsx
<Dialog aria-modal="true" ...>
```

Fix ONLY critical/serious violations. Document any intentional exclusions with code comments.

If no violations found, skip this task.

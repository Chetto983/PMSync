**Complete rewrite of search.spec.ts** to use real API calls instead of mocks.

Key changes:
1. **Remove custom fixture** - Use standard Playwright test with `storageState` from setup project
2. **Remove all route mocks** - Let requests hit real backend
3. **Use URL navigation** - Navigate to `/dashboard/email?search=from:X` to test filters
4. **Wait for API response** - Use `page.waitForResponse()` before assertions
5. **Sequential execution** - Use `test.describe.configure({ mode: 'serial' })` for tenant data safety
6. **Remove all test.skip()** - Tests should pass or fail, never skip on auth

**New test structure:**

```typescript
import { test, expect } from '@playwright/test';

test.describe('Search Filters - Real API', () => {
  // Sequential for shared tenant data
  test.describe.configure({ mode: 'serial' });

  test('from: filter shows filtered results', async ({ page }) => {
    // Navigate with search param
    await page.goto('/dashboard/email?search=from:alice');

    // Wait for API response
    await page.waitForResponse(
      (res) => res.url().includes('/emails') && res.status() === 200
    );

    // Verify chip renders
    await expect(
      page.getByRole('button', { name: /from.*alice/i })
    ).toBeVisible();

    // Verify email list shows results (at least 1)
    const emailItems = page.locator('[data-testid="email-list-item"]');
    await expect(emailItems.first()).toBeVisible({ timeout: 10000 });
  });

  test('has:attachment filter shows only emails with attachments', async ({ page }) => {
    await page.goto('/dashboard/email?search=has:attachment');

    await page.waitForResponse(
      (res) => res.url().includes('/emails') && res.status() === 200
    );

    // Verify chip
    await expect(
      page.getByRole('button', { name: /attachment/i })
    ).toBeVisible();

    // Verify attachment icon visible on results
    const emailItems = page.locator('[data-testid="email-list-item"]');
    const count = await emailItems.count();
    if (count > 0) {
      // At least one result should have attachment indicator
      await expect(
        page.locator('[data-testid="attachment-indicator"]').first()
      ).toBeVisible();
    }
  });

  test('subject: filter shows filtered results', async ({ page }) => {
    await page.goto('/dashboard/email?search=subject:meeting');

    await page.waitForResponse(
      (res) => res.url().includes('/emails') && res.status() === 200
    );

    await expect(
      page.getByRole('button', { name: /subject.*meeting/i })
    ).toBeVisible();
  });

  test('combined filters (from: AND subject:) work together', async ({ page }) => {
    await page.goto('/dashboard/email?search=from:alice subject:test');

    await page.waitForResponse(
      (res) => res.url().includes('/emails') && res.status() === 200
    );

    // Both chips should render
    await expect(page.getByRole('button', { name: /from.*alice/i })).toBeVisible();
    await expect(page.getByRole('button', { name: /subject.*test/i })).toBeVisible();
  });

  test('chip X button removes filter', async ({ page }) => {
    await page.goto('/dashboard/email?search=from:alice');

    await page.waitForResponse((res) => res.url().includes('/emails'));

    // Click X on chip
    const chip = page.getByRole('button', { name: /from.*alice/i });
    await chip.locator('svg').click();

    // Chip should disappear
    await expect(chip).not.toBeVisible();

    // URL should update
    await expect(page).not.toHaveURL(/search=/);
  });

  test('clear button removes all filters', async ({ page }) => {
    await page.goto('/dashboard/email?search=from:alice has:attachment');

    await page.waitForResponse((res) => res.url().includes('/emails'));

    // Find and click clear button in search field
    const clearButton = page.getByTestId('email-search').locator('button[aria-label*="clear"], button[aria-label*="Clear"]');
    if (await clearButton.isVisible()) {
      await clearButton.click();

      // All chips should disappear
      await expect(page.getByRole('button', { name: /from/i })).not.toBeVisible();
      await expect(page.getByRole('button', { name: /attachment/i })).not.toBeVisible();
    }
  });

  test('invalid filter shows input without chip', async ({ page }) => {
    // Incomplete filter (no value after colon)
    await page.goto('/dashboard/email?search=from:');

    await page.waitForResponse((res) => res.url().includes('/emails'));

    // Input should contain raw text
    const searchInput = page.getByTestId('email-search').locator('input');
    await expect(searchInput).toHaveValue('from:');

    // No chip should render for incomplete filter
    await expect(page.getByRole('button', { name: /from:/i })).not.toBeVisible({ timeout: 2000 });
  });
});
```

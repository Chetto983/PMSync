    **`streamChat(params)` method:**
    ```typescript
    interface StreamChatParams {
      tenantId: string;
      userId: string;
      message: string;
      conversationId?: string;
      model?: string;
      abortSignal?: AbortSignal;
    }

    async streamChat(params: StreamChatParams): Promise<ReadableStream> {
      // 1. Get or create conversation
      const conversation = await this.conversationService.getOrCreateConversation(
        params.conversationId, params.tenantId, params.userId
      );

      // 2. Save user message to conversation
      await this.conversationService.saveUserMessage(conversation.id, params.message);

      // 3. Load conversation history (last 10 messages + summary)
      const messages = await this.conversationService.getMessagesForContext(conversation.id);

      // 4. Append current user message
      messages.push({ role: 'user', content: params.message });

      // 5. Create UI message stream
      return createUIMessageStream({
        execute: async ({ writer }) => {
          const result = streamText({
            model: this.mistralProvider.getChatModel(params.model),
            system: SYSTEM_PROMPT,
            messages,
            // tools: {} -- empty for now, Plan 02 adds tools with PII resolution
            stopWhen: stepCountIs(5),
            abortSignal: params.abortSignal,
            onStepFinish: async ({ text, toolCalls, toolResults, usage }) => {
              // Persist assistant message after each step
              if (text) {
                await this.conversationService.saveAssistantMessage(
                  conversation.id,
                  { type: 'text', text },
                  usage?.totalTokens,
                );
              }
              // Tool messages persisted here when tools are added (Plan 02)
            },
            onFinish: async ({ text, usage, steps }) => {
              // Generate title for new conversations (first message)
              if (!params.conversationId && text) {
                const title = text.slice(0, 100).replace(/\n/g, ' ').trim();
                await this.conversationService.updateConversationTitle(
                  conversation.id,
                  title || 'New conversation',
                );
              }
              // Check if summary generation is needed (> 20 messages)
              // Summary generation is async -- don't block the response
              this.maybeTriggerSummaryGeneration(conversation.id).catch(err =>
                this.logger.warn(`Summary generation failed: ${err.message}`)
              );
            },
          });

          // Merge streamText output into UI message stream
          writer.merge(result.toUIMessageStream({
            sendReasoning: false,
            onError: (error) => error instanceof Error ? error.message : String(error),
          }));
        },
      });
    }
    ```

    **`maybeTriggerSummaryGeneration(conversationId)` private method:**
    - Check message count via `conversationService.getMessageCount()`
    - If > 20 messages and no existing summary (or summary is stale -- more than 10 new messages since last summary):
    - Load messages 1 through (total - 10) -- the older messages to summarize
    - Call `generateText()` with a summarization prompt and those messages
    - Store result in `AiConversation.summary`
    - This runs async (fire-and-forget from onFinish callback). Failures are logged but don't break chat.

    ```typescript
    @Post('chat')
    @HttpCode(HttpStatus.OK)
    async chat(
      @Req() req: Request & { user: AuthenticatedRequest['user'] },
      @Res() res: Response,
      @Body() dto: ChatRequestDto,
    ): Promise<void> {
      const abortController = new AbortController();
      req.on('close', () => abortController.abort());

      const stream = await this.aiChatService.streamChat({
        tenantId: req.user.tenantId,
        userId: req.user.userId,
        message: dto.message,
        conversationId: dto.conversationId,
        model: dto.model,
        abortSignal: abortController.signal,
      });

      pipeUIMessageStreamToResponse({ stream, response: res });
    }
    ```

    CRITICAL: The handler returns `void` and uses `@Res()`. Do NOT return a value -- pipeUIMessageStreamToResponse handles the response.

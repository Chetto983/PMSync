```typescript
/**
 * EMA-blend two SenderBehaviorScores with Phase 149 D-02 time-aware decay.
 *
 * effectivePrevWeight = basePrevWeight * exp(-ln2/halfLifeDays * max(0, dtDays))
 *
 * Rationale (Boyd/Luxenberg 2024, EWMM arXiv:2404.08136): a static EMA
 * prev-weight overweights stale history when updates are irregular.
 * Scaling prevWeight by exponential time-decay makes recent signals
 * dominate naturally — score halves every halfLifeDays of idle time.
 */
function blendScores(
  prev: SenderBehaviorScore,
  curr: SenderBehaviorScore,
  basePrevWeight: number,
  dtDays: number,
  halfLifeDays: number,
): SenderBehaviorScore {
  const decayFactor = Math.exp(
    (-Math.LN2 / halfLifeDays) * Math.max(0, dtDays),
  );
  const prevWeight = basePrevWeight * decayFactor;
  const currWeight = 1 - prevWeight;
  return {
    replyRate: prev.replyRate * prevWeight + curr.replyRate * currWeight,
    avgResponseTimeMs:
      prev.avgResponseTimeMs * prevWeight +
      curr.avgResponseTimeMs * currWeight,
    archiveWithoutReplyRate:
      prev.archiveWithoutReplyRate * prevWeight +
      curr.archiveWithoutReplyRate * currWeight,
    flagRate: prev.flagRate * prevWeight + curr.flagRate * currWeight,
    totalSignals: prev.totalSignals + curr.totalSignals,
    lastUpdated: curr.lastUpdated,
  };
}
```

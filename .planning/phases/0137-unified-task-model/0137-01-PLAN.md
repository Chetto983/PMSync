```prisma
model ManualTask {
  id            String    @id @default(cuid())
  tenantId      String
  userId        String
  title         String
  description   String?
  status        String    @default("pending")  // pending, in_progress, completed, dismissed
  priority      String    @default("medium")   // low, medium, high, urgent
  dueDate       DateTime?
  contactId     String?
  contactTenant String?
  dealId        String?
  emailId       String?
  emailTenant   String?
  completedAt   DateTime?
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt

  tenant        Tenant    @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  user          User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  contact       Contact?  @relation(fields: [contactId, contactTenant], references: [id, tenantId], onDelete: SetNull)
  deal          CrmDeal?  @relation(fields: [dealId], references: [id], onDelete: SetNull)
  email         Email?    @relation(fields: [emailId, emailTenant], references: [id, tenantId], onDelete: SetNull)

  @@index([tenantId, userId])
  @@index([tenantId, userId, status])
  @@index([tenantId, userId, dueDate])
  @@map("manual_tasks")
}
```

```typescript
const FOLLOW_UP_STATUS_MAP: Record<FollowUpStatus, UnifiedTaskStatus> = {
  PENDING: 'pending',
  REMINDED: 'pending',
  RESOLVED: 'completed',
  EXPIRED: 'expired',
};

const REMINDER_STATUS_MAP: Record<ReminderStatus, UnifiedTaskStatus> = {
  attivo: 'pending',
  completato: 'completed',
  annullato: 'dismissed',
  sospeso: 'pending',
};

const REMINDER_PRIORITY_MAP: Record<ReminderPriorita, UnifiedTaskPriority> = {
  alta: 'high',
  media: 'medium',
  bassa: 'low',
};

const NBA_STATUS_MAP: Record<string, UnifiedTaskStatus> = {
  SUGGESTED: 'pending',
  ACCEPTED: 'in_progress',
  EXECUTED: 'completed',
  DISMISSED: 'dismissed',
};

const TICKET_STATUS_MAP: Record<string, UnifiedTaskStatus> = {
  open: 'pending',
  in_progress: 'in_progress',
  waiting: 'pending',
  resolved: 'completed',
  closed: 'completed',
};
```

Create 5 exported normalizer functions. Each takes the Prisma type (with includes) and returns `UnifiedTaskDto`:

1. `normalizeFollowUp(record: EmailFollowUp & { email: { subject: string; from: string | null } }): UnifiedTaskDto`
   - `sourceType: 'follow_up'`
   - `title`: `"[TODO] email.subject"` or `"[Waiting] email.subject"` based on `followUpType`
   - `description`: `record.reason`
   - `priority`: `'medium'` (follow-ups have no priority field)
   - `contactName`: `record.email.from`
   - `emailId`: `record.emailId`

2. `normalizeReminder(record: Reminder): UnifiedTaskDto`
   - `sourceType: 'reminder'`
   - `title`: `record.titolo`
   - `description`: `record.descrizione`
   - `priority`: `REMINDER_PRIORITY_MAP[record.priorita]`
   - `contactName`: `record.contatto`

3. `normalizeNBAAction(record: NBAAction & { contact?: { firstName: string | null; lastName: string | null } | null }): UnifiedTaskDto`
   - `sourceType: 'deal_action'`
   - `title`: `record.title`
   - `description`: `record.reasoning`
   - `priority`: `record.expectedImpact >= 75 ? 'high' : record.expectedImpact >= 50 ? 'medium' : 'low'`
   - `contactId`: `record.contactId`
   - `contactName`: contact first+last name joined
   - `dealId`: `record.dealId`

4. `normalizeTicket(record: Ticket & { contact?: { firstName: string | null; lastName: string | null } | null }): UnifiedTaskDto`

   - `ticketId`: `record.id`
   - `contactId`: `record.contactId`

5. `normalizeManualTask(record: ManualTask & { contact?: { firstName: string | null; lastName: string | null } | null }): UnifiedTaskDto`

    ```typescript
     * Phase 151 D-B4: Light consolidation forces a flush per touched sender
     * even when buffer hasn't reached FLUSH_THRESHOLD. Wraps private
     * `drainAndFlush` for an explicit per-sender compute+blend+persist pass.
     *
     * Uses the existing drain-lua + EMA blend pipeline — no new math.
     * Safe no-op when buffer is empty (drainAndFlush early-returns per its line ~379).
     *
     * Buffer-key composition mirrors the canonical shape at line 277
     * (`${REDIS_BUFFER_PREFIX}:${tenantId}:${senderEmail}`) byte-for-byte.
     *
     * DO NOT change visibility of flushSenderSignals / drainAndFlush — this wrapper
     * is the entire public surface needed for Phase 151 Light handler.
     */
    async forceFlushSenderSignalsForTenant(
      tenantId: string,
      senderEmails: string[],
    ): Promise<void> {
      for (const senderEmail of senderEmails) {
        const normalized = senderEmail.trim().toLowerCase();
        const bufferKey = `${REDIS_BUFFER_PREFIX}:${tenantId}:${normalized}`;
        await this.drainAndFlush(bufferKey, tenantId, normalized).catch((err) => {
          this.logger.warn(
            `Force-flush failed for ${tenantId}:${normalized}: ${err instanceof Error ? err.message : String(err)}`,
          );
        });
      }
    }
    ```

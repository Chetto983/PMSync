    ```typescript
    export async function processRemConsolidationForTenant(
      tenantId: string,
      deps: RemConsolidationDeps,
      jobTimestamp: Date,

        result.kind1ContradictionsFound = kind1.hits;

      } catch (err) {

        result.edgesDemoted += kind2.edgesDemoted;
      } catch (err) {
        log.error(`REM Kind-2 detection failed for ${tenantId}: ${err instanceof Error ? err.message : String(err)}`);
    ```

    ```typescript
    // Find correction clusters per domain grouping:
    const feedbackRows = await prisma.emailTriageFeedback.findMany({
      where: {
        tenantId,
        senderEmail: { not: null },
        newValue: { not: null },
        type: 'label_correction',  // or whatever the canonical type is — verify
        createdAt: { gte: new Date(jobTimestamp.getTime() - /* 90 days */ 90 * 24 * 60 * 60 * 1000) },
      },
      select: { senderEmail: true, newValue: true, createdAt: true },
    });

    // Group by (domain, newValue):
    type DomainLabelBucket = { domain: string; label: string; senders: Set<string>; weightedSum: number };
    const buckets = new Map<string, DomainLabelBucket>();
    for (const row of feedbackRows) {
      const domain = extractDomain(row.senderEmail!);
      if (!domain) continue;
      const localPart = row.senderEmail!.split('@')[0]?.toLowerCase() ?? '';
      if (ROLE_ACCOUNT_LOCAL_PARTS.has(localPart)) continue;  // D-C4 role-account exclusion

      const key = `${domain}|||${row.newValue}`;
      const bucket = buckets.get(key) ?? { domain, label: row.newValue!, senders: new Set(), weightedSum: 0 };
      bucket.senders.add(row.senderEmail!);
      const ageDays = (jobTimestamp.getTime() - row.createdAt.getTime()) / 86_400_000;
      bucket.weightedSum += weightCorrection(ageDays, { halfLifeDays: deps.halfLifeDays });
      buckets.set(key, bucket);
    }

    // Filter per D-C4: distinct non-role senders >= 3 AND weightedSum >= 0.8.
    let promoted = 0;
    for (const bucket of buckets.values()) {
      if (bucket.senders.size < MIN_DISTINCT_SENDERS_FOR_DOMAIN) continue;
      if (bucket.weightedSum < 0.8) continue;

      // Skip if individual senders already passed Deep's 3/0.8 gate — Deep owns those.
      // Check: do 0 of these senders have >=3 individual corrections for this label?
      // If any sender has its own promoted SENDER_LABEL_AFFINITY edge to this label,
      // Deep already handled it. This is the complementary signal Deep cannot see.

      // Find-or-create email_domain node (Landmine §1 Option b):
      const domainNode = await findOrCreateGraphNode(prisma, {
        tenantId, nodeType: 'email_domain', externalId: bucket.domain, label: bucket.domain,
      });
      // Find-or-create label node:
      const labelNode = await findOrCreateGraphNode(prisma, {
        tenantId, nodeType: 'label', externalId: bucket.label, label: bucket.label,
      });

      // Upsert DOMAIN_LABEL_AFFINITY edge with Phase 150 D-01a metadata shape:
      const computedConfidence = Math.min(bucket.weightedSum / bucket.senders.size, 1.0);
      await prisma.graphEdge.upsert({
        where: {
          tenantId_sourceId_targetId_edgeType: {
            tenantId, sourceId: domainNode.id, targetId: labelNode.id, edgeType: 'DOMAIN_LABEL_AFFINITY',
          },
        },
        create: {
          tenantId, sourceId: domainNode.id, targetId: labelNode.id,
          edgeType: 'DOMAIN_LABEL_AFFINITY',
          confidence: computedConfidence,
          validFrom: jobTimestamp,
          validUntil: null,
          metadata: {
            derivedFromOverrideIds: [],
            promotionAudit: {
              jobId: 'rem-sub-threshold-rollup',
              jobTimestamp: jobTimestamp.toISOString(),
              computedConfidence,
              feedbackRowCount: Array.from(bucket.senders).length,
              halfLifeDays: deps.halfLifeDays,
              promotionSource: 'rem-sub-threshold',
            },
          },
        },
        update: { confidence: computedConfidence, validFrom: jobTimestamp },
      });
      deps.promotionCounter?.inc({ tenant_id: tenantId, edge_type: 'DOMAIN_LABEL_AFFINITY', action: 'rem_sub_threshold' });
      promoted++;
    }

    return promoted;
    ```

    ```typescript
    const hits = await prisma.$queryRaw<Array<{
      sourceId: string;
      targetIds: string[];
      edgeIds: string[];
      confidences: number[];
    }>>`
      SELECT e."sourceId",
             array_agg(e."targetId" ORDER BY e."confidence" DESC) AS "targetIds",
             array_agg(e."id" ORDER BY e."confidence" DESC) AS "edgeIds",
             array_agg(e."confidence" ORDER BY e."confidence" DESC) AS "confidences"
      FROM "graph_edges" e
      WHERE e."tenantId" = ${tenantId}
        AND e."edgeType" IN ('SENDER_LABEL_AFFINITY', 'DOMAIN_LABEL_AFFINITY')
        AND e."confidence" >= ${REM_CONTRADICTION_THRESHOLD}
        AND (e."validUntil" IS NULL OR e."validUntil" > now())
      GROUP BY e."sourceId"
      HAVING COUNT(DISTINCT e."targetId") > 1
    `;

    for (const hit of hits) {
      await prisma.triageConsolidationAudit.create({
        data: {
          tenantId,
          kind: 'competing_source',
          sourceNodeId: hit.sourceId,
          targetNodeIds: hit.targetIds,
          edgeIds: hit.edgeIds,
          confidencesBefore: hit.confidences,
          confidencesAfter: hit.edgeIds.map(() => REM_DEMOTE_CONFIDENCE),
          notes: {},
    ```

    **`detectKind2Contradictions(tenantId, deps)` (D-D1 Kind 2)** — SQL JOIN via `split_part(externalId, '@', 2)` to link sender → email_domain node. Use the exact query from PATTERNS §3 lines 345-371:

    ```typescript
    const hits = await prisma.$queryRaw<Array<{
      senderEdgeId: string;
      domainEdgeId: string;
      senderLabelId: string;
      domainLabelId: string;
      senderConfidence: number;
      domainConfidence: number;
      senderNodeId: string;
    }>>`
      SELECT se."id" AS "senderEdgeId", de."id" AS "domainEdgeId",
             stgt."externalId" AS "senderLabelId", dtgt."externalId" AS "domainLabelId",
             se."confidence" AS "senderConfidence", de."confidence" AS "domainConfidence",
             ssrc."id" AS "senderNodeId"
      FROM "graph_edges" se
      JOIN "graph_nodes" ssrc ON ssrc."id" = se."sourceId" AND ssrc."nodeType" = 'sender'
      JOIN "graph_nodes" stgt ON stgt."id" = se."targetId"
      JOIN "graph_nodes" dsrc ON dsrc."nodeType" = 'email_domain'
                             AND dsrc."tenantId" = se."tenantId"
                             AND dsrc."externalId" = split_part(ssrc."externalId", '@', 2)
      JOIN "graph_edges" de ON de."sourceId" = dsrc."id"
                           AND de."tenantId" = se."tenantId"
                           AND de."edgeType" = 'DOMAIN_LABEL_AFFINITY'
                           AND (de."validUntil" IS NULL OR de."validUntil" > now())
                           AND de."confidence" >= ${REM_CONTRADICTION_THRESHOLD}
      JOIN "graph_nodes" dtgt ON dtgt."id" = de."targetId"
      WHERE se."tenantId" = ${tenantId}
        AND se."edgeType" = 'SENDER_LABEL_AFFINITY'
        AND (se."validUntil" IS NULL OR se."validUntil" > now())
        AND se."confidence" >= ${REM_CONTRADICTION_THRESHOLD}
        AND stgt."externalId" != dtgt."externalId"
    `;

    let edgesDemoted = 0;
    for (const hit of hits) {
      await prisma.triageConsolidationAudit.create({
        data: {
          tenantId,
          kind: 'sender_vs_domain',
          sourceNodeId: hit.senderNodeId,
          targetNodeIds: [hit.senderLabelId, hit.domainLabelId],
          edgeIds: [hit.senderEdgeId, hit.domainEdgeId],
          confidencesBefore: [hit.senderConfidence, hit.domainConfidence],
          confidencesAfter: [REM_DEMOTE_CONFIDENCE, REM_DEMOTE_CONFIDENCE],
          notes: {},
        },
      });
      deps.contradictionsCounter?.inc({ kind: 'sender_vs_domain', tenant_id: tenantId });

      const demoteResult = await prisma.graphEdge.updateMany({
        where: { tenantId, id: { in: [hit.senderEdgeId, hit.domainEdgeId] } },
        data: { confidence: REM_DEMOTE_CONFIDENCE },
      });
      edgesDemoted += demoteResult.count;
    }
    return { hits: hits.length, edgesDemoted };
    ```

import { Injectable, Logger, ConflictException } from '@nestjs/common';
import { PrismaService } from '../../../prisma/prisma.service';
import { Folder } from '@prisma/client';
import { RedisLockService } from '../../../common/services/redis-lock.service';
import { AlertingService } from '../../../common/services/alerting.service';
import { FolderSyncQueryService, FolderInfo } from './folder-sync-query.service';
import { ImapFolderSyncService } from './imap-folder-sync.service';
import { GmailFolderService } from './gmail/gmail-folder.service';
import { MicrosoftFolderService } from './microsoft/microsoft-folder.service';

@Injectable()
export class FolderSyncService {
  private readonly logger = new Logger(FolderSyncService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly redisLock: RedisLockService,
    private readonly alertingService: AlertingService,
    private readonly queryService: FolderSyncQueryService,
    private readonly imapFolderSync: ImapFolderSyncService,
    private readonly gmailFolderService: GmailFolderService,
    private readonly microsoftFolderService: MicrosoftFolderService,
  ) {}

  // --- Sync Operations (delegated to provider services) ---

  async syncProviderFolders(tenantId: string, providerId: string): Promise<void> {
    const lockKey = `folder-sync:${providerId}`;
    const lockTtl = 300000;

    await this.redisLock.withLock(lockKey, lockTtl, async () => {
      // Lock monitoring logic (keep from original)
      const syncStartTime = Date.now();
      let alertFired = false;
      const alertThresholdMs = 240000;

      const monitorInterval = setInterval(() => {
        const durationMs = Date.now() - syncStartTime;
        if (!alertFired) {
          const alert = this.alertingService.checkLockDuration(lockKey, durationMs, alertThresholdMs);
          if (alert) {
            alertFired = true;
            this.logger.warn(`[RC-A1] ${alert.message} for provider ${providerId}`);
          }
        }
      }, 30000);

      try {
        const provider = await this.prisma.providerConfig.findUnique({
          where: { id: providerId },
        });

        if (!provider) throw new Error(`Provider ${providerId} not found`);

        // Delegate to provider-specific service
        if (provider.providerType === 'imap') {
          await this.imapFolderSync.syncImapFolders(tenantId, providerId);
        } else if (provider.providerType === 'google') {
          await this.gmailFolderService.syncFolders(tenantId, providerId);
        } else if (provider.providerType === 'microsoft') {
          await this.microsoftFolderService.syncMicrosoftFolders(tenantId, providerId);
        } else {
          throw new Error(`Unsupported provider type: ${provider.providerType}`);
        }
      } finally {
        clearInterval(monitorInterval);
      }
    });
  }

  // --- Query Operations (delegated to query service) ---

  getFolders(tenantId: string, providerId: string) {
    return this.queryService.getFolders(tenantId, providerId);
  }

  getFolderByPath(tenantId: string, providerId: string, path: string) {
    return this.queryService.getFolderByPath(tenantId, providerId, path);
  }

  updateFolderCounts(folderId: string) {
    return this.queryService.updateFolderCounts(folderId);
  }

  updateAllFolderCounts(tenantId: string, providerId: string) {
    return this.queryService.updateAllFolderCounts(tenantId, providerId);
  }

  auditFolderSync(tenantId: string, providerId: string) {
    return this.queryService.auditFolderSync(tenantId, providerId);
  }

  cleanupOrphanedFolders(tenantId: string, providerId: string, serverFolderPaths: string[]) {
    return this.queryService.cleanupOrphanedFolders(tenantId, providerId, serverFolderPaths);
  }

  // --- CRUD Operations (routed by provider type) ---

  async createFolder(tenantId: string, providerId: string, name: string, parent?: string) {
    const provider = await this.prisma.providerConfig.findUnique({ where: { id: providerId } });
    if (!provider) throw new Error(`Provider ${providerId} not found`);

    if (provider.providerType === 'imap') {
      return this.imapFolderSync.createImapFolder(tenantId, providerId, name, parent);
    } else if (provider.providerType === 'google') {
      return this.gmailFolderService.createGmailFolder(tenantId, providerId, name, parent);
    } else if (provider.providerType === 'microsoft') {
      return this.microsoftFolderService.createMicrosoftFolder(tenantId, providerId, name, parent);
    }
    throw new Error(`Unsupported provider type: ${provider.providerType}`);
  }

  async renameFolder(tenantId: string, providerId: string, folderId: string, newName: string) {
    // Similar pattern - delegate to provider
  }

  async deleteFolder(tenantId: string, providerId: string, folderId: string) {
    // Similar pattern - delegate to provider
  }
}

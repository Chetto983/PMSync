```typescript
/**
 * Sanitize text by redacting PII for GDPR-compliant embeddings.
 * Gracefully degrades if PiiDetectionService is unavailable.
 */
private sanitize(text: string): string {
  if (!this.piiDetectionService || !text) return text;

  try {
    const result = this.piiDetectionService.detect(text);
    if (result.hasPii) {
      this.logger.debug(
        `Sanitized ${result.piiCount} PII items for embedding: ${result.detectedTypes.join(', ')}`,
      );
    }
    return result.redactedText;
  } catch (error) {
    this.logger.warn(`PII detection failed, using original text: ${error}`);
    return text; // Graceful degradation
  }
}
```

```typescript
describe('PII sanitization', () => {
  // Create a mock PiiDetectionService
  const mockPiiDetection = {
    detect: jest.fn(),
  };

  let serviceWithPii: EmbeddingComposerService;

  beforeEach(() => {
    // Create service instance with mocked PiiDetectionService
    serviceWithPii = new EmbeddingComposerService(
      mockPrisma as any,
      mockConfig as any,
      mockPiiDetection as any,
    );
    mockPiiDetection.detect.mockReset();
  });

  it('should sanitize email body content', () => {
    mockPiiDetection.detect.mockReturnValue({
      hasPii: true,
      redactedText: 'Contact me at [EMAIL_REDACTED:abc123]',
      piiCount: 1,
      detectedTypes: ['EMAIL'],
    });

    const result = serviceWithPii.composeEmailEmbeddingContent({
      subject: 'Test Subject',
      bodyText: 'Contact me at john@example.com',
    });

    expect(mockPiiDetection.detect).toHaveBeenCalledWith('Contact me at john@example.com');
    expect(result).toContain('[EMAIL_REDACTED:');
    expect(result).toContain('Test Subject'); // Subject NOT sanitized
    expect(result).not.toContain('john@example.com');
  });

  it('should sanitize contact emails and phones', () => {
    mockPiiDetection.detect
      .mockReturnValueOnce({
        hasPii: true,
        redactedText: '[EMAIL_REDACTED:def456]',
        piiCount: 1,
        detectedTypes: ['EMAIL'],
      })
      .mockReturnValueOnce({
        hasPii: true,
        redactedText: '[PHONE_REDACTED:ghi789]',
        piiCount: 1,
        detectedTypes: ['PHONE'],
      });

    const result = serviceWithPii.composeContactEmbeddingContent({
      displayName: 'John Smith',
      emails: [{ value: 'john@example.com' }],
      phoneNumbers: [{ value: '+1-555-123-4567' }],
    });

    expect(result).toContain('John Smith'); // Name NOT sanitized
    expect(result).toContain('[EMAIL_REDACTED:');
    expect(result).toContain('[PHONE_REDACTED:');
    expect(result).not.toContain('john@example.com');
    expect(result).not.toContain('+1-555-123-4567');
  });

  it('should gracefully handle PiiDetectionService errors', () => {
    mockPiiDetection.detect.mockImplementation(() => {
      throw new Error('Detection failed');
    });

    const result = serviceWithPii.composeEmailEmbeddingContent({
      subject: 'Test',
      bodyText: 'Contact john@example.com',
    });

    // Should return original text on error
    expect(result).toContain('john@example.com');
  });

  it('should work without PiiDetectionService (graceful degradation)', () => {
    const serviceWithoutPii = new EmbeddingComposerService(
      mockPrisma as any,
      mockConfig as any,
      null, // No PII service
    );

    const result = serviceWithoutPii.composeEmailEmbeddingContent({
      subject: 'Test',
      bodyText: 'Contact john@example.com',
    });

    // Should return original text when service unavailable
    expect(result).toContain('john@example.com');
  });
});
```

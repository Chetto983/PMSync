/**
 * PII Tokenization Integration Tests
 *
 * Tests the complete PII round-trip:
 * 1. Detection: User input -> redacted text with [TYPE_REDACTED:hash] tokens
 * 2. Storage: Tokens stored in Redis (mocked) for session
 * 3. Tool Resolution: Tool payloads have tokens resolved before execution
 * 4. Display Restoration: AI responses have tokens resolved for user display
 *
 * Verifies: PII-01, PII-02, PII-03 requirements
 * @see .planning/phases/13-pii-tokenization-foundation/13-RESEARCH.md
 */
import { PiiDetectionService } from '../../../guards/pii-detection-sentinel.service';
import { PiiResolutionService, type PiiMapping } from '../../../guards/pii-resolution.service';
import { applyPiiMappings } from '../chat-response-processor.service';

describe('PII Tokenization Integration', () => {
  let piiDetection: PiiDetectionService;
  let mockCacheService: {
    get: jest.Mock;
    set: jest.Mock;
    del: jest.Mock;
  };
  let piiResolution: PiiResolutionService;

  const TENANT_ID = 'test-tenant';
  const SESSION_ID = 'test-session';

  beforeEach(() => {
    // Create fresh instances for each test
    piiDetection = new PiiDetectionService();

    // Mock CacheService for Redis simulation
    mockCacheService = {
      get: jest.fn(),
      set: jest.fn().mockResolvedValue(undefined),
      del: jest.fn().mockResolvedValue(undefined),
    };

    // Create PiiResolutionService with mocked cache
    piiResolution = new PiiResolutionService(mockCacheService as any);
  });

  describe('PII-01: Detection and Tokenization', () => {
    it('should detect and redact email addresses', () => {
      const input = 'Please contact john.doe@example.com for details';
      const result = piiDetection.detect(input);

      expect(result.hasPii).toBe(true);
      expect(result.detectedTypes).toContain('EMAIL');
      expect(result.redactedText).toMatch(/\[EMAIL_REDACTED:[a-f0-9]+\]/);
      expect(result.redactedText).not.toContain('john.doe@example.com');
    });

    it('should detect and redact phone numbers', () => {
      const input = 'Call me at +1-555-123-4567';
      const result = piiDetection.detect(input);

      expect(result.hasPii).toBe(true);
      expect(result.detectedTypes).toContain('PHONE');
      expect(result.redactedText).toMatch(/\[PHONE_REDACTED:[a-f0-9]+\]/);
      expect(result.redactedText).not.toContain('+1-555-123-4567');
    });

    it('should generate consistent tokens for same value', () => {
      const input1 = 'Contact john@example.com today';
      const input2 = 'Email john@example.com tomorrow';

      const result1 = piiDetection.detect(input1);
      const result2 = piiDetection.detect(input2);

      // Same email should produce same token (deterministic hash)
      const token1 = result1.redactedText.match(/\[EMAIL_REDACTED:[a-f0-9]+\]/)?.[0];
      const token2 = result2.redactedText.match(/\[EMAIL_REDACTED:[a-f0-9]+\]/)?.[0];

      expect(token1).toBe(token2);
    });

    it('should detect multiple PII types in same message', () => {
      const input = 'Contact john@example.com or call +1-555-123-4567';
      const result = piiDetection.detect(input);

      expect(result.hasPii).toBe(true);
      expect(result.piiCount).toBe(2);
      expect(result.detectedTypes).toContain('EMAIL');
      expect(result.detectedTypes).toContain('PHONE');
      expect(result.redactedText).toMatch(/\[EMAIL_REDACTED:[a-f0-9]+\]/);
      expect(result.redactedText).toMatch(/\[PHONE_REDACTED:[a-f0-9]+\]/);
    });
  });

  describe('PII-02: Display Restoration', () => {
    it('should restore PII tokens in AI response text', () => {
      // Simulate detection result
      const originalEmail = 'john@example.com';
      const detectionResult = piiDetection.detect(`Contact ${originalEmail}`);
      const emailToken = detectionResult.details[0].redactedValue;

      // Create mappings as stored in Redis
      const mappings: PiiMapping[] = [
        {
          placeholder: emailToken,
          originalValue: originalEmail,
          type: 'EMAIL',
          detectedAt: new Date().toISOString(),
        },
      ];

      // Simulate AI response containing token
      const aiResponse = `I found the email ${emailToken} in your contacts.`;

      // Apply restoration (used by ChatResponseProcessorService)
      const restored = applyPiiMappings(aiResponse, mappings);

      expect(restored).toBe(`I found the email ${originalEmail} in your contacts.`);
      expect(restored).not.toMatch(/\[EMAIL_REDACTED:/);
    });

    it('should restore multiple PII tokens in same response', () => {
      const originalEmail = 'jane@corp.com';
      const originalPhone = '+1-800-555-1234';

      const detectEmail = piiDetection.detect(`Email: ${originalEmail}`);
      const detectPhone = piiDetection.detect(`Phone: ${originalPhone}`);

      const emailToken = detectEmail.details[0].redactedValue;
      const phoneToken = detectPhone.details[0].redactedValue;

      const mappings: PiiMapping[] = [
        {
          placeholder: emailToken,
          originalValue: originalEmail,
          type: 'EMAIL',
          detectedAt: new Date().toISOString(),
        },
        {
          placeholder: phoneToken,
          originalValue: originalPhone,
          type: 'PHONE',
          detectedAt: new Date().toISOString(),
        },
      ];

      const aiResponse = `Contact ${emailToken} or call ${phoneToken}`;
      const restored = applyPiiMappings(aiResponse, mappings);

      expect(restored).toBe(`Contact ${originalEmail} or call ${originalPhone}`);
    });

    it('should handle response with no tokens gracefully', () => {
      const mappings: PiiMapping[] = [];
      const aiResponse = 'Here is the information you requested.';

      const restored = applyPiiMappings(aiResponse, mappings);

      expect(restored).toBe(aiResponse);
    });
  });

  describe('PII-03: Tool Payload Resolution', () => {
    it('should resolve PII tokens in flat payload', async () => {
      const originalEmail = 'recipient@company.com';
      const detectionResult = piiDetection.detect(`Send to ${originalEmail}`);
      const emailToken = detectionResult.details[0].redactedValue;

      // Simulate stored mappings in cache
      mockCacheService.get.mockResolvedValue({
        mappings: [
          {
            placeholder: emailToken,
            originalValue: originalEmail,
            type: 'EMAIL',
            detectedAt: new Date().toISOString(),
          },
        ],
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString(),
      });

      // Simulate tool payload with token
      const payload = {
        to: emailToken,
        subject: 'Meeting Request',
        body: 'Let\'s meet tomorrow.',
      };

      const resolved = await piiResolution.resolvePayload(TENANT_ID, SESSION_ID, payload);

      expect(resolved.to).toBe(originalEmail);
      expect(resolved.subject).toBe('Meeting Request');
      expect(resolved.body).toBe('Let\'s meet tomorrow.');
    });

    it('should resolve PII tokens in nested payload', async () => {
      const originalEmail = 'user@domain.org';
      const detectionResult = piiDetection.detect(originalEmail);
      const emailToken = detectionResult.details[0].redactedValue;

      mockCacheService.get.mockResolvedValue({
        mappings: [
          {
            placeholder: emailToken,
            originalValue: originalEmail,
            type: 'EMAIL',
            detectedAt: new Date().toISOString(),
          },
        ],
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString(),
      });

      const payload = {
        recipients: {
          to: [emailToken],
          cc: ['other@example.com'],
        },
        content: {
          subject: 'Test',
        },
      };

      const resolved = await piiResolution.resolvePayload(TENANT_ID, SESSION_ID, payload);

      expect(resolved.recipients.to[0]).toBe(originalEmail);
      expect(resolved.recipients.cc[0]).toBe('other@example.com');
    });

    it('should resolve PII tokens in array payload', async () => {
      const email1 = 'first@domain.com';
      const email2 = 'second@domain.com';
      const detect1 = piiDetection.detect(email1);
      const detect2 = piiDetection.detect(email2);

      mockCacheService.get.mockResolvedValue({
        mappings: [
          {
            placeholder: detect1.details[0].redactedValue,
            originalValue: email1,
            type: 'EMAIL',
            detectedAt: new Date().toISOString(),
          },
          {
            placeholder: detect2.details[0].redactedValue,
            originalValue: email2,
            type: 'EMAIL',
            detectedAt: new Date().toISOString(),
          },
        ],
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString(),
      });

      const payload = {
        recipients: [
          detect1.details[0].redactedValue,
          detect2.details[0].redactedValue,
        ],
      };

      const resolved = await piiResolution.resolvePayload(TENANT_ID, SESSION_ID, payload);

      expect(resolved.recipients).toEqual([email1, email2]);
    });
  });

  describe('Round-trip integration', () => {
    it('should complete full PII round-trip: detect -> store -> resolve -> display', async () => {
      const userMessage = 'Send email to john.smith@company.com about the project';

      // Step 1: Detection (PII-01)
      const detectResult = piiDetection.detect(userMessage);
      expect(detectResult.hasPii).toBe(true);
      const emailToken = detectResult.details[0].redactedValue;
      const redactedMessage = detectResult.redactedText;
      expect(redactedMessage).not.toContain('john.smith@company.com');

      // Step 2: Storage (simulated via mock)
      await piiResolution.storeMapping(TENANT_ID, SESSION_ID, detectResult.details);
      expect(mockCacheService.set).toHaveBeenCalled();

      // Step 3: Simulate cache returning stored mappings
      mockCacheService.get.mockResolvedValue({
        mappings: [
          {
            placeholder: emailToken,
            originalValue: 'john.smith@company.com',
            type: 'EMAIL',
            detectedAt: new Date().toISOString(),
          },
        ],
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString(),
      });

      // Step 4: Tool resolution (PII-03)
      const toolPayload = {
        to: emailToken,
        subject: 'Project Update',
      };
      const resolvedPayload = await piiResolution.resolvePayload(TENANT_ID, SESSION_ID, toolPayload);
      expect(resolvedPayload.to).toBe('john.smith@company.com');

      // Step 5: Display restoration (PII-02)
      const aiResponse = `I've prepared an email to ${emailToken} about the project.`;
      const mappings = await piiResolution.getMappings(TENANT_ID, SESSION_ID);
      const displayText = applyPiiMappings(aiResponse, mappings);
      expect(displayText).toBe(`I've prepared an email to john.smith@company.com about the project.`);
    });
  });

  describe('Edge cases', () => {
    it('should handle empty input', () => {
      const result = piiDetection.detect('');
      expect(result.hasPii).toBe(false);
      expect(result.redactedText).toBe('');
    });

    it('should handle input with no PII', () => {
      const input = 'Please review the quarterly report';
      const result = piiDetection.detect(input);
      expect(result.hasPii).toBe(false);
      expect(result.redactedText).toBe(input);
    });

    it('should handle missing session gracefully', async () => {
      mockCacheService.get.mockResolvedValue(null);

      const payload = {
        to: '[EMAIL_REDACTED:missing]',
        subject: 'Test',
      };

      const resolved = await piiResolution.resolvePayload(TENANT_ID, SESSION_ID, payload);

      // Should return original payload when session not found
      expect(resolved.to).toBe('[EMAIL_REDACTED:missing]');
    });
  });
});

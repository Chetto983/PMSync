    ```typescript
    function convertDbMessagesToThreadMessageLike(
      messages: Array<{
        id: string;
        role: string;
        content: unknown;
        createdAt: string;
      }>
    ): ThreadMessageLike[] {
      return messages
        .filter((m) => m.role === 'user' || m.role === 'assistant')
        .map((m) => {
          const raw = m.content;
          let text = '';
          if (typeof raw === 'string') {
            text = raw;
          } else if (Array.isArray(raw)) {
            // AI SDK stores content as parts array: [{ type: 'text', text: '...' }]
            text = raw
              .filter((p: Record<string, unknown>) => p.type === 'text')
              .map((p: Record<string, unknown>) => String(p.text ?? ''))
              .join('');
          } else if (raw && typeof raw === 'object') {
            text = String((raw as Record<string, unknown>).text ?? '');
          }
          return {
            role: m.role as 'user' | 'assistant',
            content: text,
            id: m.id,
            createdAt: new Date(m.createdAt),
          };
        });
    }
    ```

    **Step 4:** Create the `ThreadHistoryProvider` component:

    ```typescript
    function ThreadHistoryProvider({ children }: { children: ReactNode }) {
      const remoteId = useAuiState((s) => s.threadListItem.remoteId);

      const history = useMemo<ThreadHistoryAdapter>(() => ({
        async load() {
          if (!remoteId) return { messages: [] };
          try {
            const { data } = await apiClient.get(`/ai/chat/conversations/${remoteId}`);
            const dbMessages = data?.conversation?.messages ?? [];
            const threadMessages = convertDbMessagesToThreadMessageLike(dbMessages);
            return ExportedMessageRepository.fromArray(threadMessages);
          } catch {
            return { messages: [] };
          }
        },
        async append() { /* no-op -- backend persists during SSE streaming */ },
      }), [remoteId]);

      return (
        <RuntimeAdapterProvider adapters={{ history }}>
          {children}
        </RuntimeAdapterProvider>
      );
    }
    ```

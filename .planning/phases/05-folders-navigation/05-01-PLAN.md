const bulkArchiveMutation = useMutation({

  onSuccess: () => {
    clearSelection();
    queryClient.invalidateQueries({ queryKey: ['emails'] });
    toast({ title: t.email.bulkActions?.archived || 'Emails archived' });
  },
  onError: () => {
    toast({ title: t.errors.genericError, variant: 'destructive' });
  },
});

const bulkDeleteMutation = useMutation({
  mutationFn: (ids: string[]) => emailsApi.bulkDelete(ids),
  onSuccess: () => {
    clearSelection();
    queryClient.invalidateQueries({ queryKey: ['emails'] });
    toast({ title: t.email.bulkActions?.deleted || 'Emails deleted' });
  },
});

const bulkMarkReadMutation = useMutation({
  mutationFn: (ids: string[]) => emailsApi.bulkMarkRead(ids, true),

});

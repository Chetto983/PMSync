case 'e':
  // Archive focused item
  if (focusedId && onArchive && !e.ctrlKey && !e.metaKey && !e.altKey) {
    e.preventDefault();
    onArchive(focusedId);
  }
  break;

case '#':
  // Delete focused item (Shift+3)
  if (focusedId && onDelete && !e.ctrlKey && !e.metaKey && !e.altKey) {
    e.preventDefault();
    onDelete(focusedId);
  }
  break;

case 'r':
  // Reply to focused item
  if (focusedId && onReply && !e.ctrlKey && !e.metaKey && !e.altKey) {
    e.preventDefault();
    onReply(focusedId);
  }
  break;

case 'a':
  // Reply all (only if not Ctrl+A for select all)
  if (focusedId && onReplyAll && !e.ctrlKey && !e.metaKey && !e.altKey) {
    e.preventDefault();
    onReplyAll(focusedId);
  }
  break;

case 'f':
  // Forward focused item
  if (focusedId && onForward && !e.ctrlKey && !e.metaKey && !e.altKey) {
    e.preventDefault();
    onForward(focusedId);
  }
  break;

case '/':
  // Focus search (only if not Shift+/ for help)
  if (onFocusSearch && !e.shiftKey && !e.ctrlKey && !e.metaKey && !e.altKey) {
    e.preventDefault();
    onFocusSearch();
  }
  break;

case '?':
  // Show help (Shift+/)
  if (onShowHelp) {
    e.preventDefault();
    onShowHelp();
  }
  break;
```

import {
  Dialog,
  DialogTitle,
  DialogContent,
  DialogActions,
  Button,
  Box,
  Typography,
  Chip,
} from '@mui/material';
import { Keyboard } from 'lucide-react';
import { useTranslations } from '@/lib/hooks/use-translations';

interface KeyboardShortcutsDialogProps {
  open: boolean;
  onClose: () => void;
}

function ShortcutRow({ keys, label }: { keys: string[]; label: string }) {
  return (
    <Box sx={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', py: 0.75 }}>
      <Typography variant="body2" color="text.secondary">{label}</Typography>
      <Box sx={{ display: 'flex', gap: 0.5 }}>
        {keys.map((key, i) => (
          <Chip
            key={i}
            label={key}
            size="small"
            sx={{
              fontFamily: 'monospace',
              minWidth: 28,
              height: 24,
            }}
          />
        ))}
      </Box>
    </Box>
  );
}

export function KeyboardShortcutsDialog({ open, onClose }: KeyboardShortcutsDialogProps) {
  const t = useTranslations();
  const shortcuts = t.email.keyboardShortcuts;

  return (
    <Dialog open={open} onClose={onClose} maxWidth="xs" fullWidth>
      <DialogTitle sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
        <Keyboard size={20} />
        {shortcuts?.title || 'Keyboard Shortcuts'}
      </DialogTitle>
      <DialogContent dividers>
        <Box sx={{ display: 'flex', flexDirection: 'column', gap: 0.5 }}>
          <Typography variant="overline" color="text.secondary" sx={{ mt: 1 }}>
            {shortcuts?.navigation || 'Navigation'}
          </Typography>
          <ShortcutRow keys={['j']} label={shortcuts?.moveDown || 'Next email'} />
          <ShortcutRow keys={['k']} label={shortcuts?.moveUp || 'Previous email'} />
          <ShortcutRow keys={['Enter']} label={shortcuts?.open || 'Open email'} />
          <ShortcutRow keys={['Esc']} label={shortcuts?.close || 'Close / Deselect'} />

          <Typography variant="overline" color="text.secondary" sx={{ mt: 2 }}>
            {shortcuts?.actions || 'Actions'}
          </Typography>
          <ShortcutRow keys={['e']} label={shortcuts?.archive || 'Archive'} />
          <ShortcutRow keys={['#']} label={shortcuts?.delete || 'Delete'} />
          <ShortcutRow keys={['r']} label={shortcuts?.reply || 'Reply'} />
          <ShortcutRow keys={['a']} label={shortcuts?.replyAll || 'Reply all'} />
          <ShortcutRow keys={['f']} label={shortcuts?.forward || 'Forward'} />
          <ShortcutRow keys={['x']} label={shortcuts?.toggleSelect || 'Toggle select'} />

          <Typography variant="overline" color="text.secondary" sx={{ mt: 2 }}>
            {shortcuts?.other || 'Other'}
          </Typography>
          <ShortcutRow keys={['c']} label={shortcuts?.compose || 'Compose new'} />
          <ShortcutRow keys={['/']} label={shortcuts?.search || 'Search'} />
          <ShortcutRow keys={['?']} label={shortcuts?.showHelp || 'Show shortcuts'} />
          <ShortcutRow keys={['Cmd', 'K']} label={shortcuts?.commandPalette || 'Command palette'} />
        </Box>
      </DialogContent>
      <DialogActions>
        <Button onClick={onClose}>{t.common.close || 'Close'}</Button>
      </DialogActions>
    </Dialog>
  );
}
```

```typescript
const handleDragEnd = (event: DragEndEvent) => {
  const { active, over } = event;

  if (!active || !over) return;

  const activeType = active.data.current?.type;
  const overType = over.data.current?.type;

  // Case 1: Email dropped on folder (existing logic)
  if (activeType === 'email' && overType === 'folder') {
    // ... existing email move logic
    return;
  }

  // Case 2: Folder dropped ON another folder (create nesting)
  if (activeType === 'folder' && overType === 'folder') {
    const sourceFolder = active.data.current?.folder;
    const targetFolder = over.data.current?.folder;

    // Prevent dropping folder into itself or its children
    if (sourceFolder.id === targetFolder.id) return;
    if (isDescendantOf(sourceFolder, targetFolder)) return;

    // Call API to move folder into target as child
    handleMoveToParent(sourceFolder.id, targetFolder.id);
    return;
  }

  // Case 3: Folder dropped in list (reorder - existing logic)
  if (activeType === 'folder' && !overType) {
    // ... existing reorder logic
  }
};
```

    ```typescript
    import { LANE_TO_ACTION } from './triage-override.service';

    async promoteOverrides(tenantId: string): Promise<{ promoted: number; skipped: boolean }> {
      // Idempotency guard (UNCHANGED from Plan 03)
      if (this.redis) {
        const last = await this.redis.get(IDEMPOTENCY_KEY(tenantId));
        if (last && Date.now() - Number(last) < MIN_INTERVAL_MS) {
          this.logger.debug(`Promotion skipped for tenant=${tenantId} (within 5min window)`);
          return { promoted: 0, skipped: true };
        }
      }

      const suggestions = await this.feedbackService.getSuggestions(tenantId);
      let promoted = 0;
      const patternsPromoted: Array<{ scope: 'sender' | 'domain'; pattern: string }> = [];

      // D-05a: label_correction suggestions are NOT promoted to TriageOverride.
      // They feed the few-shot path instead. Only lane_correction maps to an action.
      // Phase 147 has no lane_correction producers (D-02), so this loop is effectively
      // a no-op this phase. Phase 148+ wires lane_correction capture; this code then
      // starts creating overrides with zero additional changes.
      for (const s of suggestions) {
        const action = LANE_TO_ACTION[s.target];
        if (!action) continue;   // target isn't a lane — skip (label_correction case)

        await this.overrideService.addOverride(tenantId, s.scope, s.pattern, action);
        patternsPromoted.push({ scope: s.scope, pattern: s.pattern });
        promoted += 1;
        this.logger.log(`Promoted ${s.scope} override: ${s.pattern} -> ${action} (tenant=${tenantId})`);
      }

      // Cache invalidation for newly-promoted overrides — next triage picks them up.
      if (patternsPromoted.length > 0) {
        const affectedEmailIds: string[] = [];
        for (const p of patternsPromoted) {
          const rows = p.scope === 'sender'
            ? await this.prisma.email.findMany({
                where: { tenantId, from: { equals: p.pattern, mode: 'insensitive' } },
                select: { id: true },
              })
            : await this.prisma.email.findMany({
                where: { tenantId, from: { contains: `@${p.pattern}`, mode: 'insensitive' } },
                select: { id: true },
              });
          affectedEmailIds.push(...rows.map((r) => r.id));
        }
        if (affectedEmailIds.length > 0) {
          const invalidated = await this.analysisService.invalidateCache(tenantId, affectedEmailIds);
          this.logger.log(`Invalidated ${invalidated} cached analyses for newly-promoted overrides (tenant=${tenantId})`);
        }
      }

      if (this.redis) {
        await this.redis.setex(IDEMPOTENCY_KEY(tenantId), IDEMPOTENCY_TTL_SECONDS, String(Date.now()));
      }

      this.logger.log(`Promotion run complete for tenant=${tenantId}: promoted=${promoted}`);
      return { promoted, skipped: false };
    }
    ```

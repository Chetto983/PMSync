    /**
     * Detect follow-up type based on email folder.
     * WAITING_FOR: Email in SENT folder with waitingFor items
     * TODO: Email in INBOX/other folder with actionItems
     */
    private detectFollowUpType(email: { folderId: string | null }, extraction?: { waitingFor?: unknown[]; actionItems?: unknown[] }): FollowUpType {
      // If email was sent by user and has waitingFor items -> WAITING_FOR
      const isSentFolder = email.folderId?.toLowerCase().includes('sent') || false;
      const hasWaitingFor = extraction?.waitingFor && extraction.waitingFor.length > 0;

      if (isSentFolder && hasWaitingFor) {
        return FollowUpType.WAITING_FOR;
      }
      return FollowUpType.TODO;
    }

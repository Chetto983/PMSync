    ```typescript
    import * as chrono from 'chrono-node';

    /**
     * Extract deadline from email text using NLP (chrono-node).
     * Looks for patterns like "by Friday", "end of next week", "before March 15".
     * @returns Date or null if no deadline found
     */
    private extractDeadlineFromText(text: string, referenceDate: Date = new Date()): Date | null {
      // First try to find explicit deadline patterns
      const deadlinePatterns = [
        /\b(?:by|before|until|due|deadline[:]?)\s+(.+?)(?:\.|,|$)/gi,
        /\b(?:respond|reply|answer|get back)\s+(?:by|before)\s+(.+?)(?:\.|,|$)/gi,
        /\b(?:please|kindly)\s+(?:respond|reply)\s+by\s+(.+?)(?:\.|,|$)/gi,
      ];

      for (const pattern of deadlinePatterns) {
        const match = pattern.exec(text);
        if (match && match[1]) {
          const parsed = chrono.parseDate(match[1], referenceDate);
          if (parsed) return parsed;
        }
      }

      // Fallback: general date extraction from text
      const results = chrono.parse(text, referenceDate, { forwardDate: true });
      if (results.length > 0) {
        // Prefer dates in the future
        const futureResult = results.find(r => r.start.date() > referenceDate);
        if (futureResult) return futureResult.start.date();
        return results[0].start.date();
      }

      return null;
    }
    ```

    1. Add smart timing utility function to follow-up.service.ts:
    ```typescript
    interface SmartTimingRecommendation {
      suggestedDate: Date;
      reason: string;
      confidence: number; // 0-1
    }

    /**
     * Calculate optimal follow-up time based on research:
     * - 2-3 days after initial send (sweet spot)
     * - Tuesday-Thursday preferred (higher response rates)
     * - 9-11 AM optimal window
     *
     * Research source: FEATURES.md competitive analysis
     * - Single follow-up doubles response rate (16% -> 27%)
     */
    private calculateSmartTiming(createdAt: Date): SmartTimingRecommendation {
      const now = new Date();
      const daysSinceCreation = Math.floor((now.getTime() - createdAt.getTime()) / (1000 * 60 * 60 * 24));

      // Default: 2-3 days from creation or now, whichever is later
      let targetDate = new Date(createdAt);
      targetDate.setDate(targetDate.getDate() + 3); // 3 days after creation

      if (targetDate < now) {
        // If already past, suggest next business day
        targetDate = new Date(now);
        targetDate.setDate(targetDate.getDate() + 1);
      }

      // Adjust to Tuesday-Thursday if possible
      const dayOfWeek = targetDate.getDay(); // 0=Sun, 1=Mon, 2=Tue, 3=Wed, 4=Thu, 5=Fri, 6=Sat
      if (dayOfWeek === 0) targetDate.setDate(targetDate.getDate() + 2); // Sun -> Tue
      if (dayOfWeek === 6) targetDate.setDate(targetDate.getDate() + 3); // Sat -> Tue
      if (dayOfWeek === 5) targetDate.setDate(targetDate.getDate() + 4); // Fri -> Tue
      if (dayOfWeek === 1) targetDate.setDate(targetDate.getDate() + 1); // Mon -> Tue

      // Set to 10 AM (optimal window)
      targetDate.setHours(10, 0, 0, 0);

      const reason = daysSinceCreation < 2
        ? 'Recommended: 2-3 days after initial send'
        : 'Recommended: Tuesday-Thursday 10 AM';

      return {
        suggestedDate: targetDate,
        reason,
        confidence: 0.85, // Based on research averages
      };
    }
    ```

    2. Add endpoint GET /follow-ups/:id/smart-timing to controller:

    ```typescript
    @ApiOperation({ summary: 'Get smart timing recommendation' })
    async getSmartTiming(
      @Req() req: AuthenticatedRequest,
      @Param('id') id: string,
    ) {
      const followUp = await this.followUpService.getFollowUpById(
        req.user.tenantId,
        req.user.userId,
        id,
      );
      if (!followUp) throw new NotFoundException('Follow-up not found');

      const recommendation = this.followUpService.getSmartTimingRecommendation(followUp.createdAt);
      return {
        success: true,
        recommendation,
      };
    }
    ```

**2. Create frontend/middleware.ts:**

```typescript
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
  const sessionCookie = request.cookies.get('better-auth.session_token');
  const pathname = request.nextUrl.pathname;

  // Skip: auth pages, static, API, _next, onboarding itself
  const skipPaths = ['/auth', '/_next', '/api', '/onboarding', '/favicon', '/terms', '/privacy'];
  if (skipPaths.some(p => pathname.startsWith(p))) {
    return NextResponse.next();
  }

  // No session -> login
  if (!sessionCookie) {
    const loginUrl = new URL('/auth/login', request.url);
    loginUrl.searchParams.set('returnTo', pathname);
    return NextResponse.redirect(loginUrl);
  }

  // Check onboarding completion via session cookie cache
  // The cookie cache (if enabled) contains user fields including onboardingComplete
  // For lightweight Edge check: look for a separate onboarding cookie flag
  // OR check the session cache cookie value
  // Simplest approach: set a separate "pmsync:onboarding-complete" cookie after wizard finishes
  // The onboarding page itself sets this cookie
  const onboardingCookie = request.cookies.get('pmsync:onboarding-complete');
  if (!onboardingCookie && pathname.startsWith('/dashboard')) {
    return NextResponse.redirect(new URL('/onboarding', request.url));
  }

  return NextResponse.next();
}

export const config = {
  matcher: ['/dashboard/:path*', '/onboarding'],
};
```

NOTE: The middleware uses a lightweight cookie check. The actual onboardingComplete field validation happens in the onboarding page component via `useSession()`. The cookie is set when the wizard completes and cleared on signOut. This avoids DB calls in Edge middleware.

**3. Create frontend/pages/onboarding.tsx:**

```typescript
import { useEffect } from 'react';
import { useRouter } from 'next/router';
import { Box, CircularProgress } from '@mui/material';
import { authClient } from '@/lib/auth-client';
import { OnboardingWizard } from '@/components/onboarding/OnboardingWizard';

export default function OnboardingPage() {
  const { data: session, isPending } = authClient.useSession();
  const router = useRouter();

  useEffect(() => {
    if (!isPending && !session?.user) {
      router.replace('/auth/login');
    }
    // If already completed onboarding, go to dashboard
    if (!isPending && session?.user?.onboardingComplete) {
      document.cookie = 'pmsync:onboarding-complete=1; path=/; max-age=31536000; SameSite=Lax';
      router.replace('/dashboard');
    }
  }, [session, isPending, router]);

  if (isPending || !session?.user) {
    return <Box sx={{ display: 'flex', justifyContent: 'center', alignItems: 'center', minHeight: '100vh' }}><CircularProgress /></Box>;
  }

  return <OnboardingWizard user={session.user} />;
}
```

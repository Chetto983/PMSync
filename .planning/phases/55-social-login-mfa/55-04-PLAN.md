**3. Create `frontend/pages/api/onboarding/create-workspace.ts`** (Next.js API route):

This route replicates the workspace provisioning logic from `provisionWorkspaceForUser` in better-auth.instance.ts but is callable on-demand for social login users.

```typescript
import type { NextApiRequest, NextApiResponse } from 'next';
import { auth } from '@/lib/auth'; // Next.js better-auth instance for session validation

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'POST') return res.status(405).end();

  // Validate session using better-auth
  const session = await auth.api.getSession({ headers: req.headers as any });
  if (!session?.user) return res.status(401).json({ error: 'Unauthorized' });

  // Check user doesn't already have a tenant
  if (session.user.tenantId) return res.status(400).json({ error: 'Workspace already exists' });

  const { workspaceName } = req.body;
  if (!workspaceName || typeof workspaceName !== 'string' || workspaceName.trim().length < 2) {
    return res.status(400).json({ error: 'Workspace name must be at least 2 characters' });
  }

  // Import prisma from backend shared instance or create standalone
  // Use the same pattern as better-auth.instance.ts provisionWorkspaceForUser
  const { PrismaClient } = require('@prisma/client');
  const prisma = new PrismaClient();

  try {
    const slug = workspaceName.trim().toLowerCase().replace(/[^a-z0-9]/g, '-').slice(0, 30);
    const tenant = await prisma.tenant.create({
      data: {
        name: workspaceName.trim(),
        slug: `${slug}-${Date.now().toString(36)}`,
        settings: { create: {} },
      },
    });

    await prisma.user.update({
      where: { id: session.user.id },
      data: { tenantId: tenant.id },
    });

    // Create GDPR consent record
    await prisma.gdprConsent.create({
      data: {
        userId: session.user.id,
        tenantId: tenant.id,
        consentType: 'terms_of_service',
        granted: true,
        ipAddress: (req.headers['x-forwarded-for'] as string)?.split(',')[0] || req.socket.remoteAddress || '',
        userAgent: req.headers['user-agent'] || '',
      },
    });

    res.status(201).json({ tenantId: tenant.id });
  } catch (error) {
    console.error('Workspace creation failed:', error);
    res.status(500).json({ error: 'Failed to create workspace' });
  } finally {
    await prisma.$disconnect();
  }
}
```

@Patch(':sourceType/:id/snooze')
@RateTierDecorator('safe-write')
@ApiOperation({ summary: 'Snooze a task until a future date' })
async snooze(
  @Req() req: AuthenticatedRequest,
  @Param('sourceType') sourceType: TaskSourceType,
  @Param('id') id: string,
  @Body() dto: SnoozeTaskDto,
) {
  const dueDate = new Date(dto.dueDate);
  if (isNaN(dueDate.getTime()) || dueDate.getTime() <= Date.now()) {
    throw new BadRequestException('dueDate must be a valid future date');
  }
  if (!TASK_SOURCE_TYPES.includes(sourceType)) {
    throw new BadRequestException(`Invalid source type: ${sourceType}`);
  }
  await this.aggregationService.snoozeTask(
    req.user.tenantId,
    req.user.id,
    sourceType,
    id,
    dueDate,
  );
  return { success: true, dueDate: dueDate.toISOString() };
}

```typescript
async snoozeTask(
  tenantId: string,
  userId: string,
  sourceType: TaskSourceType,
  sourceId: string,
  dueDate: Date,
): Promise<void> {
  switch (sourceType) {
    case 'manual': {
      await this.manualTaskService.update(tenantId, sourceId, {
        dueDate: dueDate.toISOString(),
      });
      break;
    }
    case 'follow_up': {
      // Delegate to existing follow-up snooze pattern (ai.controller.ts:744-751).
      const result = await this.prisma.emailFollowUp.updateMany({
        where: {
          id: sourceId,
          tenantId,
          userId,
          status: { in: [FollowUpStatus.PENDING, FollowUpStatus.REMINDED] },
        },
        data: { dueDate, status: FollowUpStatus.PENDING, remindCount: 0 },
      });
      if (result.count === 0) {
        throw new NotFoundException(`Follow-up ${sourceId} not found or resolved`);
      }
      break;
    }
    case 'reminder': {
      // Reminder has its own dueDate — update directly, keep tenant guard.
      const record = await this.prisma.reminder.findUnique({ where: { id: sourceId } });
      if (!record || record.tenantId !== tenantId) {
        throw new NotFoundException(`Reminder ${sourceId} not found`);
      }
      await this.prisma.reminder.update({
        where: { id: sourceId },
        data: { dueDate },
      });
      break;
    }
    case 'deal_action':
    case 'ticket':
      throw new BadRequestException(
        `Snooze not supported for source type: ${sourceType}`,
      );
    default:
      throw new BadRequestException(`Unknown source type: ${sourceType}`);
  }
}
```

export function useCompleteFeedItem() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (item: TodayFeedItem) => {
      if (!item.canComplete || !item.taskSourceType || !item.taskSourceId) {
        throw new Error('Item is not completable');
      }
      await apiClient.post(`/tasks/${item.taskSourceType}/${item.taskSourceId}/complete`);
    },
    onMutate: async (item) => {
      await qc.cancelQueries({ queryKey: todayKeys.all });
      const queries = qc.getQueriesData<TodayFeedResponse>({ queryKey: todayKeys.all });
      const snapshots = queries.map(([k, d]) => [k, d] as const);
      for (const [key, data] of queries) {
        if (!data) continue;
        qc.setQueryData<TodayFeedResponse>(key, {
          ...data,
          items: data.items.filter((i) => i.id !== item.id),
        });
      }
      return { snapshots };
    },
    onError: (_err, _item, ctx) => {
      if (ctx?.snapshots) {
        for (const [k, d] of ctx.snapshots) qc.setQueryData(k, d);
      }
    },
    onSettled: () => qc.invalidateQueries({ queryKey: todayKeys.all }),
  });

onMutate: async (item) => {
  await qc.cancelQueries({ queryKey: todayKeys.all });
  const queries = qc.getQueriesData<TodayFeedResponse>({ queryKey: todayKeys.all });
  const snapshots = queries.map(([k, d]) => [k, d] as const);
  for (const [key, data] of queries) {
    if (!data) continue;
    qc.setQueryData<TodayFeedResponse>(key, { ...data, items: data.items.filter((i) => i.id !== item.id) });
  }
  return { snapshots };
},
onError: (_err, _item, ctx) => {
  if (ctx?.snapshots) for (const [k, d] of ctx.snapshots) qc.setQueryData(k, d);
},

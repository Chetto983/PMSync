    ```typescript
    'use client';

    import { Box, IconButton, Tooltip } from '@mui/material';
    import { Check, Clock, X } from 'lucide-react';
    import { useTranslations } from '@/lib/hooks/use-translations';
    import type { TodayFeedItem } from '@/lib/api/dashboard';

    interface Props {
      item: TodayFeedItem;
      onComplete: () => void;
      onSnooze: (anchor: HTMLElement) => void;
      onDismiss: () => void;
    }

    function stop(fn: () => void) {
      return (e: React.MouseEvent) => {
        e.stopPropagation();
        e.preventDefault();
        fn();
      };
    }

    export function TodayActions({ item, onComplete, onSnooze, onDismiss }: Props) {
      const t = useTranslations();
      const aria = t.dashboard.today.actions;
      const openSnooze = (e: React.MouseEvent<HTMLButtonElement>) => {
        e.stopPropagation();
        e.preventDefault();
        onSnooze(e.currentTarget);
      };
      return (
        <Box
          sx={{
            display: 'flex',
            alignItems: 'center',
            gap: 0.25,
            opacity: { xs: 1, sm: 0.5 },
            transition: 'opacity 120ms ease-out',
            // Reduced-motion fallback per UI-SPEC §Accessibility line 221:
            // opacity 0.5 → 1.0 reveal must be instant under prefers-reduced-motion.
            '@media (prefers-reduced-motion: reduce)': {
              transition: 'none',
            },
            '.feed-row:hover &': { opacity: 1 },
            '.feed-row:focus-within &': { opacity: 1 },
          }}
        >
          {item.canComplete && (
            <Tooltip title={aria.complete} enterDelay={800}>
              <IconButton
                size="small"
                aria-label={aria.complete}
                onClick={stop(onComplete)}
                sx={{ width: 24, height: 24 }}
              >
                <Check size={16} />
              </IconButton>
            </Tooltip>
          )}
          {item.canSnooze && (
            <Tooltip title={aria.snooze} enterDelay={800}>
              <IconButton
                size="small"
                aria-label={aria.snooze}
                onClick={openSnooze}
                sx={{ width: 24, height: 24 }}
              >
                <Clock size={16} />
              </IconButton>
            </Tooltip>
          )}
          <Tooltip title={aria.dismiss} enterDelay={800}>
            <IconButton
              size="small"
              aria-label={aria.dismiss}
              onClick={stop(onDismiss)}
              sx={{ width: 24, height: 24 }}
            >
              <X size={16} />
            </IconButton>
          </Tooltip>
        </Box>
      );
    }
    ```

    ```typescript
    import { useTranslations } from '@/lib/hooks/use-translations';

    import { TodayKpiChip } from './TodayKpiChip';

    interface Props {
      counts?: TodayFeedCounts;
      loading: boolean;

    function pickGreetingKey(): 'morning' | 'afternoon' | 'evening' {
      const h = new Date().getHours();
      if (h < 12) return 'morning';
      if (h < 18) return 'afternoon';
      return 'evening';
    }

    export function TodayHeader({ counts, loading }: Props) {
      const t = useTranslations();
      const { locale } = useLocale();
      const greeting = t.dashboard.today.greeting[pickGreetingKey()];
      const dateStr = format(new Date(), 'EEEE d MMMM', {
        locale: locale === 'it' ? itLocale : enUS,
      });
      return (
        <Stack spacing={1.5} component="header">
          <Box>
            <Typography sx={{ fontSize: 24, fontWeight: 600, lineHeight: 1.2, color: 'text.primary' }}>
              {greeting}
            </Typography>
            <Typography sx={{ fontSize: 14, fontWeight: 400, lineHeight: 1.5, color: 'text.secondary' }}>
              {dateStr}
            </Typography>
          </Box>
          <Stack direction="row" spacing={1} flexWrap="wrap" useFlexGap>
            <TodayKpiChip
              icon={Mail}
              count={counts?.kpi.inbox ?? 0}
              label={t.dashboard.today.kpi.inbox}
              href="/dashboard/email"
              loading={loading}
            />
            <TodayKpiChip
              icon={Calendar}
              count={counts?.kpi.calendar ?? 0}
              label={t.dashboard.today.kpi.calendar}

              label={t.dashboard.today.kpi.tasks}
              href="/dashboard/tasks"
              loading={loading}
            />
          </Stack>
        </Stack>
      );
    }
    ```

**2. Create `lib/api/compliance.ts`:**

Follow the pattern of `lib/api/ai.ts` (use the existing apiClient):

```typescript
export const complianceApi = {
  // Legal acceptance
  getLegalStatus: () => apiClient.get('/compliance/legal/status'),
  acceptLegalDocument: (data: { documentType: string; documentVersion: string }) =>
    apiClient.post('/compliance/legal/accept', data),

  // Data export
  requestExport: () => apiClient.post('/compliance/data-export'),
  listExports: () => apiClient.get('/compliance/data-export'),
  getExportStatus: (id: string) => apiClient.get(`/compliance/data-export/${id}`),
  downloadExport: (id: string) => apiClient.get(`/compliance/data-export/${id}/download`, { responseType: 'blob' }),

  // Account deletion
  requestDeletion: () => apiClient.post('/compliance/data-deletion'),
  getDeletionStatus: () => apiClient.get('/compliance/data-deletion/status'),
  cancelDeletion: () => apiClient.delete('/compliance/data-deletion/cancel'),
};
```

**3. Create `hooks/queries/use-data-rights.ts`:**

Follow existing TanStack Query hook patterns in the codebase:

- `dataRightsKeys` factory: `{ all: ['dataRights'], exports: ['dataRights', 'exports'], deletionStatus: ['dataRights', 'deletion'], legalStatus: ['dataRights', 'legal'] }`
- `useDataExports()`: `useQuery({ queryKey: dataRightsKeys.exports, queryFn: complianceApi.listExports, staleTime: 30_000 })`
- `useExportMutation()`: `useMutation({ mutationFn: complianceApi.requestExport, onSuccess: invalidate exports })`
- `useDeletionStatus()`: `useQuery({ queryKey: dataRightsKeys.deletionStatus, queryFn: complianceApi.getDeletionStatus })`
- `useDeletionMutation()`: `useMutation({ mutationFn: complianceApi.requestDeletion, onSuccess: invalidate deletion status })`
- `useCancelDeletionMutation()`: `useMutation({ mutationFn: complianceApi.cancelDeletion, onSuccess: invalidate deletion status })`
- `useLegalStatus()`: `useQuery({ queryKey: dataRightsKeys.legalStatus, queryFn: complianceApi.getLegalStatus })`
- `useLegalAcceptMutation()`: `useMutation({ mutationFn: complianceApi.acceptLegalDocument, onSuccess: invalidate legal status })`

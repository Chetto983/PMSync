function ToolStateIndicator({ state }: { state: string }) {
  switch (state) {
    case 'input-streaming':
    case 'output-streaming':
      return <CircularProgress size={14} sx={{ ml: 1 }} />;
    case 'output-available':
      return <Check size={14} color={theme.palette.success.main} style={{ marginLeft: 4 }} />;
    case 'output-error':
      return <AlertCircle size={14} color={theme.palette.error.main} style={{ marginLeft: 4 }} />;
    default:
      return null;
  }
}

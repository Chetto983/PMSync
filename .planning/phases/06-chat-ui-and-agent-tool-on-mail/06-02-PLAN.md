  <files>frontend/components/email/thread/EmailChatFab.tsx</files>
  <action>
Create a floating action button component for toggling the chat panel.

Implementation:
1. Use MUI Fab component with Sparkles icon from lucide-react
2. Position fixed in bottom-right (with safe area insets for iOS)
3. Accept props: isOpen: boolean, onClick: () => void, disabled?: boolean
4. When chat is open, show X icon instead of Sparkles
5. Add pulse animation when there's agent activity (optional prop: hasActivity)
6. Use theme.palette.primary for background

```typescript
'use client';

import { Fab, Zoom, Badge } from '@mui/material';
import { Sparkles, X } from 'lucide-react';

interface EmailChatFabProps {
  isOpen: boolean;
  onClick: () => void;
  disabled?: boolean;
  hasActivity?: boolean;
}

export function EmailChatFab({ isOpen, onClick, disabled, hasActivity }: EmailChatFabProps) {
  return (
    <Zoom in={!disabled}>
      <Fab
        color="primary"
        onClick={onClick}
        sx={{
          position: 'fixed',
          bottom: { xs: `calc(16px + env(safe-area-inset-bottom, 0px))`, md: 24 },
          right: { xs: 16, md: 24 },
          zIndex: 1300, // Above most content but below dialogs
        }}
        aria-label={isOpen ? 'Close AI assistant' : 'Open AI assistant'}
      >
        <Badge color="secondary" variant="dot" invisible={!hasActivity || isOpen}>
          {isOpen ? <X size={24} /> : <Sparkles size={24} />}
        </Badge>
      </Fab>
    </Zoom>
  );
}
```

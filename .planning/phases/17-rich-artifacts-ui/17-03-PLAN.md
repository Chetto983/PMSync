5. Add handler functions inside the component:
```typescript
const handleResolve = (followUpId: string) => {
  setResolvedIds(prev => new Set(prev).add(followUpId));
  onAction?.({
    type: 'api-call',
    fn: async () => {
      const { voltAgentApi } = await import('@/lib/api/ai');
      await voltAgentApi.resolveFollowUp(followUpId);
    },
  });
};

const handleSnooze = (followUpId: string, emailSubject: string) => {
  onAction?.({
    type: 'send-message',
    message: `Snooze the follow-up for "${emailSubject}" to tomorrow`,
  });
};
```

Note: Use dynamic import for `voltAgentApi` to avoid importing the full API module at the renderer level. The function is invoked lazily only when the user clicks.

6. For each follow-up item row, add a `relative group` class and an ActionBar positioned absolutely:
```tsx
<div key={fu.followUpId} className="relative group flex items-start gap-2.5 px-2 py-2 rounded-md hover:bg-muted/50 transition-colors">
  {/* existing content... */}

  {/* Action bar - shows on hover */}
  {!resolvedIds.has(fu.followUpId) && (
    <div className="absolute right-1 top-1/2 -translate-y-1/2 opacity-0 group-hover:opacity-100 transition-opacity">

        { icon: CheckIcon, label: 'Complete', onClick: () => handleResolve(fu.followUpId) },
```

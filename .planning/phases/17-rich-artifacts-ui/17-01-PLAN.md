3. Inside the `ChatMessages` component function, add a `useRouter()` call and create a `handleArtifactAction` callback:
```typescript
const router = useRouter();

const handleArtifactAction = useCallback((action: ArtifactAction) => {
  switch (action.type) {
    case 'navigate':
      router.push(action.route);
      break;
    case 'send-message':
      onSuggestedQuestionSelect?.(action.message);
      break;
    case 'api-call':
      action.fn().catch((err) => console.error('Artifact action failed:', err));
      break;
  }
}, [router, onSuggestedQuestionSelect]);
```

/**
 * Download object content as Buffer from S3.
 * Used for re-attaching files in forward operations.
 */
async getObject(key: string): Promise<Buffer> {
  const command = new GetObjectCommand({
    Bucket: this.bucket,
    Key: key,
  });
  const response = await this.client.send(command);
  const stream = response.Body;
  if (!stream) {
    throw new Error(`Empty response body for S3 key: ${key}`);
  }
  // Convert readable stream to Buffer
  const chunks: Uint8Array[] = [];
  for await (const chunk of stream as AsyncIterable<Uint8Array>) {
    chunks.push(chunk);
  }
  return Buffer.concat(chunks);
}
```

/**
 * Email Send/Reply/Forward Validation Orchestrator
 *
 * Usage:
 *   node scripts/validate-email-send.mjs --provider=gmail
 *   node scripts/validate-email-send.mjs --provider=outlook
 *   node scripts/validate-email-send.mjs --provider=imap
 *   node scripts/validate-email-send.mjs                  # all providers
 *
 * Tests: WRITE-01 (Gmail send), WRITE-02 (Outlook send), WRITE-03 (IMAP send),
 *        WRITE-04 (reply threading), WRITE-05 (forward with attachments)
 *
 * Requires Edge running with --remote-debugging-port=9222 and logged into PMSync.
 */
import { connectCDP, getAuthCookies, getCsrfToken, report } from './lib/validation-helpers.mjs';
import { validateEmailSend } from './lib/validate-email-send.mjs';

const args = process.argv.slice(2);
const providerFlag = args.find(a => a.startsWith('--provider='))?.split('=')[1] || 'all';

async function main() {
  console.log('=== Email Send/Reply/Forward Validation ===');
  console.log(`Provider: ${providerFlag}\n`);

  const { browser, context } = await connectCDP();
  const cookieHeader = await getAuthCookies(context);
  const { csrfToken, cookieHeader: finalCookies } = await getCsrfToken(context, cookieHeader);

  const results = await validateEmailSend(context, finalCookies, csrfToken, {
    provider: providerFlag,
  });

  const exitCode = report('Email Send/Reply/Forward Validation', results);
  await browser.close();
  process.exit(exitCode);
}

main().catch(err => { console.error('Fatal error:', err.message); process.exit(1); });
```

// --- WRITE-01/02/03: Send new email ---
async function validateSend(context, cookieHeader, csrfToken, provider, results) {
  const label = providerTypeLabel(provider);
  console.log(`\n  --- WRITE-01/02/03: Send new email (${label}) ---`);

  const uniqueSubject = `PMSync validation send ${Date.now()}`;

  // Send email to self (provider.email) to avoid external side effects
  const sendResult = await apiPost(context, cookieHeader, csrfToken, '/api/v1/emails/send', {
    providerId: provider.id,
    to: [provider.email],
    subject: uniqueSubject,
    bodyHtml: `<p>Validation test email sent at ${new Date().toISOString()}</p>`,
  });

  assertTrue(`${label} send returns success`, sendResult?.success === true, results);
  assertTrue(`${label} send returns messageId`, typeof sendResult?.messageId === 'string' && sendResult.messageId.length > 0, results);

  // Wait briefly for DB write, then verify email in Sent folder
  await new Promise(r => setTimeout(r, 2000));

  // Fetch emails to verify the sent email appears in DB
  const emailsRes = await apiGet(context, `/api/v1/emails?providerId=${provider.id}&limit=10`, cookieHeader);
  const emails = unwrapEmails(emailsRes);
  const sentEmail = emails.find(e => e.subject === uniqueSubject);
  assertTrue(`${label} sent email found in DB`, !!sentEmail, results);

  return { sentEmail, messageId: sendResult?.messageId };
}

// --- WRITE-04: Reply preserves thread ---
async function validateReply(context, cookieHeader, csrfToken, provider, results) {
  const label = providerTypeLabel(provider);
  console.log(`\n  --- WRITE-04: Reply preserves thread (${label}) ---`);

  // Find an existing email with messageId and threadId to reply to
  const emailsRes = await apiGet(context, `/api/v1/emails?providerId=${provider.id}&limit=20`, cookieHeader);
  const emails = unwrapEmails(emailsRes);
  const replyTarget = emails.find(e => e.messageId && e.externalId);

  if (!replyTarget) {
    console.log(`  SKIP: No suitable email found for reply test (${label})`);
    return;
  }

  console.log(`  Replying to email: ${replyTarget.subject} (id: ${replyTarget.id})`);

  const replyResult = await apiPost(context, cookieHeader, csrfToken, `/api/v1/emails/${replyTarget.id}/reply`, {
    bodyHtml: `<p>Validation reply at ${new Date().toISOString()}</p>`,
  });

  assertTrue(`${label} reply returns success`, replyResult?.success === true, results);
  assertTrue(`${label} reply returns messageId`, typeof replyResult?.messageId === 'string' && replyResult.messageId.length > 0, results);

  // Wait for DB write, then fetch the reply to check headers
  await new Promise(r => setTimeout(r, 2000));

  const afterEmails = unwrapEmails(await apiGet(context, `/api/v1/emails?providerId=${provider.id}&limit=20`, cookieHeader));
  const replyEmail = afterEmails.find(e => e.inReplyTo && e.inReplyTo === replyTarget.messageId);

  if (replyEmail) {
    assertEqual(`${label} reply inReplyTo matches original messageId`, replyEmail.inReplyTo, replyTarget.messageId, results);
    assertTrue(`${label} reply references contains original messageId`, (replyEmail.references || '').includes(replyTarget.messageId), results);
    // For Gmail: check threadId preserved
    if (provider.providerType === 'google' && replyTarget.threadId) {
      assertEqual(`${label} reply threadId matches original`, replyEmail.threadId, replyTarget.threadId, results);
    }
  } else {
    console.log(`  WARN: Could not find reply email in DB to verify headers`);
    assertTrue(`${label} reply email found in DB for header check`, false, results);
  }
}

// --- WRITE-05: Forward with attachments ---
async function validateForward(context, cookieHeader, csrfToken, provider, results) {
  const label = providerTypeLabel(provider);
  console.log(`\n  --- WRITE-05: Forward with attachments (${label}) ---`);

  // Find an email with attachments
  const emailsRes = await apiGet(context, `/api/v1/emails?providerId=${provider.id}&hasAttachments=true&limit=10`, cookieHeader);
  const emails = unwrapEmails(emailsRes);
  const fwdTarget = emails.find(e => e.attachments?.length > 0);

  if (!fwdTarget) {
    console.log(`  SKIP: No email with attachments found for forward test (${label})`);
    return;
  }

  const originalAttachmentCount = fwdTarget.attachments.length;
  const originalFilenames = fwdTarget.attachments.map(a => a.filename).sort();
  console.log(`  Forwarding email with ${originalAttachmentCount} attachments: ${originalFilenames.join(', ')}`);

  const fwdResult = await apiPost(context, cookieHeader, csrfToken, `/api/v1/emails/${fwdTarget.id}/forward`, {
    to: [provider.email], // Self-forward
  });

  assertTrue(`${label} forward returns success`, fwdResult?.success === true, results);
  assertTrue(`${label} forward returns messageId`, typeof fwdResult?.messageId === 'string' && fwdResult.messageId.length > 0, results);

  // Wait for DB write, then verify forwarded email has attachments
  await new Promise(r => setTimeout(r, 3000));

  const afterEmails = unwrapEmails(await apiGet(context, `/api/v1/emails?providerId=${provider.id}&limit=20`, cookieHeader));
  const fwdEmail = afterEmails.find(e => e.subject && e.subject.includes('Fwd:') && e.subject.includes(fwdTarget.subject.replace(/^(Fwd:\s*)+/, '')));

  if (fwdEmail) {
    const fwdAttachmentCount = fwdEmail.attachments?.length ?? 0;
    assertTrue(`${label} forward includes original attachments (expected >= ${originalAttachmentCount}, got ${fwdAttachmentCount})`,
      fwdAttachmentCount >= originalAttachmentCount, results);

    // Verify filenames match
    if (fwdEmail.attachments) {
      const fwdFilenames = fwdEmail.attachments.map(a => a.filename).sort();
      for (const origName of originalFilenames) {
        assertTrue(`${label} forward includes attachment "${origName}"`, fwdFilenames.includes(origName), results);
      }
    }
  } else {
    console.log(`  WARN: Could not find forwarded email in DB -- may need manual check`);
    // Don't fail here -- the email was sent successfully, DB record may take longer
  }
}

// --- Main export ---
export async function validateEmailSend(context, cookieHeader, csrfToken, opts = {}) {
  const results = [];
  setResultsCollector(results);

  const providers = await getProviders(context, cookieHeader);
  const filtered = providers.filter(p => matchesProviderFilter(p, opts.provider || 'all'));

  if (filtered.length === 0) {
    console.log('No matching providers found!');
    return results;
  }

  for (const provider of filtered) {
    const label = providerTypeLabel(provider);
    console.log(`\n=== Provider: ${label} (${provider.email}) ===`);

    // WRITE-01/02/03: Send
    await validateSend(context, cookieHeader, csrfToken, provider, results);

    // WRITE-04: Reply
    await validateReply(context, cookieHeader, csrfToken, provider, results);

    // WRITE-05: Forward with attachments
    await validateForward(context, cookieHeader, csrfToken, provider, results);
  }

  return results;
}

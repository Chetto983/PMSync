  title?: string;

      {error?.message ?? t.errors.fetchErrorDescription}
    </Alert>
  );
}

Create `frontend/scripts/i18n-audit.ts` -- a Node.js script that:

1. Imports both `en.ts` and `it.ts` translation objects
2. Recursively walks each object to collect all leaf key paths (e.g., `common.retry`, `email.emptyTitle`)
3. Computes the diff: keys in en but not in it, and keys in it but not in en
4. Prints results to stdout: "Missing in it.ts: N keys" with the list, and "Missing in en.ts: N keys" with the list
5. Exits with code 1 if any missing keys found, 0 if parity

The script should be runnable via `npx tsx frontend/scripts/i18n-audit.ts`.

Example implementation:
```typescript
import en from '../locales/en';
import it from '../locales/it';

function collectKeys(obj: Record<string, unknown>, prefix = ''): string[] {
  const keys: string[] = [];
  for (const [key, value] of Object.entries(obj)) {
    const path = prefix ? `${prefix}.${key}` : key;
    if (value !== null && typeof value === 'object' && !Array.isArray(value)) {
      keys.push(...collectKeys(value as Record<string, unknown>, path));
    } else {
      keys.push(path);
    }
  }
  return keys;
}

const enKeys = new Set(collectKeys(en));
const itKeys = new Set(collectKeys(it));

const missingInIt = [...enKeys].filter(k => !itKeys.has(k)).sort();
const missingInEn = [...itKeys].filter(k => !enKeys.has(k)).sort();

if (missingInIt.length > 0) {
  console.log(`\nMissing in it.ts (${missingInIt.length} keys):`);
  missingInIt.forEach(k => console.log(`  - ${k}`));
}
if (missingInEn.length > 0) {
  console.log(`\nMissing in en.ts (${missingInEn.length} keys):`);
  missingInEn.forEach(k => console.log(`  - ${k}`));
}
if (missingInIt.length === 0 && missingInEn.length === 0) {
  console.log('i18n key parity: OK (0 missing keys)');
}

process.exit(missingInIt.length + missingInEn.length > 0 ? 1 : 0);
```

NOTE: Check whether en.ts and it.ts export as default or named. Adjust imports accordingly. They likely use `export default { ... }` or `const en = { ... }; export default en;`.

**1. Modify MemoryService** (`memory.service.ts`):

Add new method `getMemoriesByEntity()`:
```typescript
async getMemoriesByEntity(
  tenantId: string,
  userId: string,
  entityId: string,
  entityType: 'contact' | 'email',
  limit: number = 10,
): Promise<MemoryResult[]> {
  // Direct FK lookup -- no vector search needed
  const where = entityType === 'contact'
    ? { tenantId, userId, contactId: entityId }
    : { tenantId, userId, emailId: entityId };

  const memories = await this.prisma.aIMemory.findMany({
    where,
    orderBy: { updatedAt: 'desc' },
    take: limit,
  });

  return memories.map(m => ({
    id: m.id,
    content: m.content,
    memoryType: m.memoryType,
    confidence: m.confidence,
    category: m.category,
    similarity: 1.0, // Direct match, max relevance
  }));
}
```

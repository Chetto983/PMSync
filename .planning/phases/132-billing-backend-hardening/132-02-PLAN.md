    ```typescript
    /**
     * Check for subscriptions that were cancelled and have passed their cancelAt date.
     * User retains access until cancelAt; after that, downgrade to FREE.
     */
    private async checkCancelledSubscriptions(): Promise<number> {
      const now = new Date();

      const cancelledSubs = await this.prisma.subscription.findMany({
        where: {
          status: SubscriptionStatus.ACTIVE,
          cancelAt: { lte: now },
          plan: { not: SubscriptionPlan.FREE },
        },
        select: { tenantId: true, id: true, plan: true, cancelAt: true },
      });

      if (cancelledSubs.length === 0) {
        return 0;
      }

      this.logger.log(`Found ${cancelledSubs.length} cancelled subscriptions past cancelAt date`);

      let processed = 0;
      for (const sub of cancelledSubs) {
        try {
          await this.subscriptionService.updatePlan(sub.tenantId, SubscriptionPlan.FREE);
          await this.subscriptionService.updateStatus(sub.tenantId, SubscriptionStatus.CANCELED);

          this.logger.log(
            `Cancelled subscription ${sub.tenantId} (${sub.plan}) downgraded to FREE after cancelAt ${sub.cancelAt?.toISOString()}`,
          );
          processed++;
        } catch (error) {
          const msg = error instanceof Error ? error.message : String(error);
          this.logger.error(`Failed to downgrade cancelled ${sub.tenantId}: ${msg}`);
        }
      }

      return processed;
    }
    ```

    ```typescript
    private async sendAlertEmails(
      alert: BillingAlert,
      tenantAdminEmail: string | null,
    ): Promise<void> {
      const { tenantId, level } = alert;

      if (tenantAdminEmail) {
        try {
          await this.emailSendService.queueSend({
            tenantId: 'system',
            providerId: 'system',
            to: [tenantAdminEmail],
            subject: this.formatEmailSubject(level),
            bodyHtml: this.formatEmailBody(alert),
          });
          this.logger.log(`Billing alert email queued for ${tenantAdminEmail} | Level: ${level} | Tenant: ${tenantId}`);
        } catch (err) {
          // Non-blocking: email failure should not break billing flow
          this.logger.error(`Failed to queue billing alert email for ${tenantAdminEmail}: ${err instanceof Error ? err.message : String(err)}`);
        }
      }

      // Admin notification for critical/blocked
      if ((level === 'critical' || level === 'blocked') && this.adminEmail) {
        try {
          await this.emailSendService.queueSend({
            tenantId: 'system',
            providerId: 'system',
            to: [this.adminEmail],
            subject: `[ADMIN] ${this.formatEmailSubject(level)}`,
            bodyHtml: this.formatEmailBody(alert),
          });
          this.logger.log(`Admin billing alert email queued for ${this.adminEmail} | Level: ${level} | Tenant: ${tenantId}`);
        } catch (err) {
          this.logger.error(`Failed to queue admin alert email: ${err instanceof Error ? err.message : String(err)}`);
        }
      }
    }
    ```

**Migration 2: Hybrid search SQL functions** (20260302_add_hybrid_search_functions/migration.sql):

```sql
-- RRF hybrid search for emails (vector + FTS)
CREATE OR REPLACE FUNCTION hybrid_email_search(
  p_tenant_id TEXT,
  p_query TEXT,
  p_embedding vector(1024),
  p_match_count INT DEFAULT 10,
  p_fts_weight FLOAT DEFAULT 1.0,
  p_semantic_weight FLOAT DEFAULT 1.0,
  p_rrf_k INT DEFAULT 50
)
RETURNS TABLE (
  email_id TEXT,
  subject TEXT,
  "from" TEXT,
  snippet TEXT,
  received_at TIMESTAMPTZ,
  rrf_score FLOAT
)
LANGUAGE sql STABLE AS $$
  WITH fts_results AS (
    SELECT e.id, ROW_NUMBER() OVER (ORDER BY ts_rank_cd(e.fts, websearch_to_tsquery('simple', p_query)) DESC) AS rank_ix
    FROM emails e
    WHERE e."tenantId" = p_tenant_id AND e.fts @@ websearch_to_tsquery('simple', p_query)
    ORDER BY rank_ix LIMIT LEAST(p_match_count * 2, 40)
  ),
  semantic_results AS (
    SELECT emb."emailId" AS id, ROW_NUMBER() OVER (ORDER BY emb.vector <-> p_embedding) AS rank_ix
    FROM embeddings emb
    WHERE emb."tenantId" = p_tenant_id AND emb."emailId" IS NOT NULL
    ORDER BY emb.vector <-> p_embedding LIMIT LEAST(p_match_count * 2, 40)
  )
  SELECT
    e.id AS email_id, e.subject, e."from", e.snippet,
    e."receivedAt" AS received_at,
    (COALESCE(1.0 / (p_rrf_k + fts.rank_ix), 0.0) * p_fts_weight +
     COALESCE(1.0 / (p_rrf_k + sem.rank_ix), 0.0) * p_semantic_weight) AS rrf_score
  FROM fts_results fts
  FULL OUTER JOIN semantic_results sem ON fts.id = sem.id
  JOIN emails e ON COALESCE(fts.id, sem.id) = e.id AND e."tenantId" = p_tenant_id
  ORDER BY rrf_score DESC
  LIMIT p_match_count;
$$;

-- RRF hybrid search for contacts (vector + FTS)
CREATE OR REPLACE FUNCTION hybrid_contact_search(
  p_tenant_id TEXT,
  p_query TEXT,
  p_embedding vector(1024),
  p_match_count INT DEFAULT 10,
  p_rrf_k INT DEFAULT 50
)
RETURNS TABLE (
  contact_id TEXT,
  display_name TEXT,
  company TEXT,
  job_title TEXT,
  rrf_score FLOAT
)
LANGUAGE sql STABLE AS $$
  WITH fts_results AS (
    SELECT c.id, ROW_NUMBER() OVER (ORDER BY ts_rank_cd(c.fts, websearch_to_tsquery('simple', p_query)) DESC) AS rank_ix
    FROM contacts c
    WHERE c."tenantId" = p_tenant_id AND c.fts @@ websearch_to_tsquery('simple', p_query)
    ORDER BY rank_ix LIMIT LEAST(p_match_count * 2, 40)
  ),
  semantic_results AS (
    SELECT emb."contactId" AS id, ROW_NUMBER() OVER (ORDER BY emb.vector <-> p_embedding) AS rank_ix
    FROM embeddings emb
    WHERE emb."tenantId" = p_tenant_id AND emb."contactId" IS NOT NULL
    ORDER BY emb.vector <-> p_embedding LIMIT LEAST(p_match_count * 2, 40)
  )
  SELECT
    c.id AS contact_id, c."displayName" AS display_name, c.company, c."jobTitle" AS job_title,
    (COALESCE(1.0 / (p_rrf_k + fts.rank_ix), 0.0) +
     COALESCE(1.0 / (p_rrf_k + sem.rank_ix), 0.0)) AS rrf_score
  FROM fts_results fts
  FULL OUTER JOIN semantic_results sem ON fts.id = sem.id
  JOIN contacts c ON COALESCE(fts.id, sem.id) = c.id AND c."tenantId" = p_tenant_id
  ORDER BY rrf_score DESC
  LIMIT p_match_count;
$$;
```

```typescript
semantic_search: tool({
  description: 'Search across emails, contacts, and calendar events using semantic and keyword matching. Finds relevant content by meaning, not just exact text matches.',
  inputSchema: z.object({
    query: z.string().describe('Natural language search query'),
    sources: z.array(z.enum(['emails', 'contacts', 'calendar'])).optional().describe('Sources to search. Omit to search all.'),
    limit: z.number().min(1).max(20).optional().describe('Maximum results (default 10)'),
  }),
  execute: async (args) => {
    const resolved = await piiResolve(args as Record<string, unknown>);
    const results = await this.hybridSearchService.search(
      tenantId,
      resolved.query as string,
      (resolved.sources as string[]) || ['emails', 'contacts', 'calendar'],
      (resolved.limit as number) ?? 10,
    );
    return { results, count: results.length };
  },
}),
```

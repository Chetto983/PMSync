```bash
cd backend
P95=$(jq -r '.aggregate.summaries["http.response_time"].p95' load-test/report-1000.json)

HTTP_4XX=$(jq -r '[.aggregate.counters | to_entries[] | select(.key | test("^http\\.codes\\.4\\d\\d$")) | .value] | add // 0' load-test/report-1000.json)
HTTP_5XX=$(jq -r '[.aggregate.counters | to_entries[] | select(.key | test("^http\\.codes\\.5\\d\\d$")) | .value] | add // 0' load-test/report-1000.json)

TIMEOUT_RATE=$(awk -v t="$TIMEOUTS" -v r="$RESETS" -v tot="$TOTAL_REQ" 'BEGIN { printf "%.4f", (t+r)/tot*100 }')

# Read replica memory limit recorded in Task 1

echo "p95=$P95 p99=$P99 requests=$TOTAL_REQ http_errors=$HTTP_ERRORS timeouts=$TIMEOUTS resets=$RESETS error_rate=${ERROR_RATE}% timeout_rate=${TIMEOUT_RATE}% sio_ok=$SIO_OK sio_err=$SIO_ERR replica_mem_gb=$RAILWAY_REPLICA_MEMORY_GB pattern_d_rss_trigger_gb=$PATTERN_D_RSS_TRIGGER_GB"
```

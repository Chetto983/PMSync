**File 6 — `backend/load-test/processors/auth.js`** (CommonJS — NOT TypeScript):

```javascript
// backend/load-test/processors/auth.js — Artillery processor (CommonJS).
// Artillery loads processors as plain Node — ts-node is NOT available.
// See 157-RESEARCH.md §Security V7 — cookie redaction is mandatory.

'use strict';

module.exports = {
  // Redact Set-Cookie and authorization-bearing headers in response artifacts.
  // Artillery writes `response.headers` into report JSON; without redaction,
  // session cookies end up in CI artifacts.
  redactCookie(requestParams, response, context, ee, next) {
    if (response && response.headers) {
      if (response.headers['set-cookie']) {
        response.headers['set-cookie'] = '<redacted>';
      }
      if (response.headers['authorization']) {
        response.headers['authorization'] = '<redacted>';
      }
    }
    return next();
  },

  // Rotate through pre-seeded user pool deterministically based on VU id.
  // Exposes `context.vars.loadtestEmail` for scenarios that want a stable
  // per-VU identity instead of $randomNumber.
  pickUser(requestParams, context, ee, next) {
    const raw = context.vars.$uuid || context.$uuid || Math.random() * 1000;
    const idx = Math.floor(Number(raw)) % 1000;
    context.vars.loadtestEmail = `loadtest+${idx}@pmsync.local`;
    return next();
  },
};
```

```typescript
/**
 * SCALE-04 — Seed 1000 synthetic users into the loadtest-tenant.
 * Idempotent: skips users that already exist.
 * Run via: npm run load-test:seed
 *
 * Env vars required (see .env.loadtest.example):
 *   LOADTEST_TENANT_ID (default: loadtest-tenant)
 *   LOADTEST_PASSWORD
 *   LOADTEST_USER_COUNT (default: 1000)
 *
 * RESEARCH.md §Open Question 6: better-auth bcrypt-ish hashing is ~100ms/user.
 * Uses chunked Promise.all (50 parallel × 20 chunks) to keep total under 5 min.
 */
import { PrismaClient } from '@prisma/client';

const prisma = new PrismaClient();

const TENANT_ID = process.env.LOADTEST_TENANT_ID || 'loadtest-tenant';
const PASSWORD = process.env.LOADTEST_PASSWORD;
const USER_COUNT = Number(process.env.LOADTEST_USER_COUNT || 1000);
const CHUNK_SIZE = 50;

async function main(): Promise<void> {
  if (!PASSWORD) {
    console.error('FATAL: LOADTEST_PASSWORD must be set');
    process.exit(1);
  }
  if (USER_COUNT < 1 || USER_COUNT > 10_000) {
    console.error(`FATAL: LOADTEST_USER_COUNT out of range: ${USER_COUNT}`);
    process.exit(1);
  }

  console.log(`[seed] target tenant: ${TENANT_ID}`);
  console.log(`[seed] target user count: ${USER_COUNT}`);

  // Ensure tenant exists
  const tenant = await prisma.tenant.upsert({
    where: { id: TENANT_ID },
    update: {},
    create: {
      id: TENANT_ID,
      name: 'Load Test Tenant',
      // NOTE: other required fields per current schema — adapt based on actual Tenant model
    },
  });
  console.log(`[seed] tenant ready: ${tenant.id}`);

  const emails = Array.from({ length: USER_COUNT }, (_, i) => `loadtest+${i}@pmsync.local`);
  let created = 0;
  let skipped = 0;

  // Chunked parallel creation
  for (let start = 0; start < emails.length; start += CHUNK_SIZE) {
    const chunk = emails.slice(start, start + CHUNK_SIZE);
    const results = await Promise.allSettled(
      chunk.map(async (email) => {
        const existing = await prisma.user.findFirst({
          where: { email, tenantId: TENANT_ID },
          select: { id: true },
        });
        if (existing) return { email, status: 'skipped' as const };
        // Create via better-auth's server-side user creation flow.
        // NOTE: this is a placeholder — actual implementation must use the
        // project's user-creation helper (e.g., AuthService.createUser or
        // better-auth's admin.createUser). Adapt based on actual API.
        await prisma.user.create({
          data: {
            email,
            tenantId: TENANT_ID,
            // Add required fields per schema; password hashing handled by auth helper
            name: `Load Test User ${email}`,
          },
        });
        return { email, status: 'created' as const };
      }),
    );

    for (const r of results) {
      if (r.status === 'fulfilled') {
        if (r.value.status === 'created') created++;
        else skipped++;
      } else {
        console.error(`[seed] error:`, r.reason);
      }
    }
    const done = Math.min(start + CHUNK_SIZE, emails.length);
    console.log(`[seed] progress: ${done}/${emails.length} (created=${created} skipped=${skipped})`);
  }

  console.log(`[seed] done — created=${created} skipped=${skipped}`);
  await prisma.$disconnect();
}

main().catch((err) => {
  console.error('[seed] fatal:', err);
  process.exit(1);
});
```

```typescript
 * SCALE-04 — Delete all users + tenant row for loadtest-tenant.
 * Run via: npm run load-test:cleanup

}

main().catch((err) => {
```

## Reading the Report

After `npm run load-test:report`, open `load-test/report.html`. Key sections:

| Section | What to look for |
|---|---|
| **Summary** | `http.codes.200` count ≈ `http.requests` count; `errors.*` should be near-zero. |
| **Latency histograms** | `http.response_time.p95` per endpoint. The aggregate p95 is a bad metric — check per-endpoint (plugin `metrics-by-endpoint`). |
| **Phases** | Each phase should show sustained throughput matching its `arrivalRate`. If phase 5 (hold 1000) shows a drop to 700, the server is rejecting requests. |
| **errors.ECONNREFUSED / ETIMEDOUT** | The load generator's network is overwhelmed, not the server. Re-run from a bigger machine. |
| **errors.EHOSTUNREACH** | DNS / Railway service down. Check `LOADTEST_TARGET`. |

## Known Pitfalls (from RESEARCH.md)

1. **Tenant leak (Pitfall 5):** If `load-test:cleanup` is skipped, `loadtest-tenant` rows accumulate and pollute aggregate dashboards. ALWAYS run cleanup after the test. The npm script includes a safety guard that refuses to delete tenants whose ID doesn't start with `loadtest-`.

2. **LRU access-gated TTL (Pitfall 6):** Not relevant to this runbook (it's a backend-code concern for SCALE-02).

3. **Per-user RAM dominated by buffer churn (Pitfall 7):** Take RAM measurements at STEADY STATE — 5+ minutes into the 1000-VU hold phase — not during the ramp. Ramping allocates transient Buffer objects that skew `rss`; the 15-min hold gives V8's GC time to drain them.

4. **k6 for Socket.IO:** Don't try it. k6's `ws` module speaks raw WebSocket; Socket.IO's framing/heartbeat/ack protocol will not be exercised. Metrics lie.

5. **Load-testing from home Wi-Fi:** Your uplink is ~10 Mbps. Artillery at 1000 VUs × 50 KB/req needs ≥400 Mbps. Use GitHub Actions or an EC2 t3.large.

6. **Missing baseline:** SCALE-05 (Plan 04) runs the CURRENT topology's baseline BEFORE the split. Do not interpret the first post-split run in isolation — always compare before/after.

## Rollback Procedure

If a load test uncovers a regression severe enough to halt the phase:

1. `npm run load-test:cleanup` (delete loadtest-tenant).
2. Revert the most recent phase commits that affect the system under test.
3. Re-run `load-test:smoke` against the reverted state to confirm recovery.
4. Document the regression in a new `.planning/phases/XXX-regression/` directory before resuming phase 157.

The load-test rig itself is reversible: `git revert` the commits from phase 157 plan 03 removes all load-test files. The only out-of-tree artifact is the seeded `loadtest-tenant` — cleanup handles it.

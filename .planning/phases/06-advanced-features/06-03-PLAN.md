```typescript
/**
 * Search Query Parser
 *
 * Parses Gmail-style filter syntax:
 * - from:john@example.com
 * - to:jane@example.com
 * - subject:meeting
 * - has:attachment
 * - after:2024-01-01
 * - before:2024-12-31
 * - "quoted phrase" for exact match
 */

export interface ParsedFilters {
  query: string;
  from?: string;
  to?: string;
  subject?: string;
  hasAttachment?: boolean;
  startDate?: string; // ISO date string
  endDate?: string;   // ISO date string
}

/**
 * Parse search input into structured filters
 * Handles quoted strings and multiple filters
 */
export function parseSearchQuery(input: string): ParsedFilters {
  const filters: ParsedFilters = { query: '' };

  // Match quoted strings or non-space tokens
  const parts = input.match(/(?:[^\s"]+|"[^"]*")+/g) || [];
  const remainingParts: string[] = [];

  for (const part of parts) {
    const colonIndex = part.indexOf(':');
    if (colonIndex === -1) {
      remainingParts.push(part);
      continue;
    }

    const prefix = part.slice(0, colonIndex).toLowerCase();
    const value = part.slice(colonIndex + 1).replace(/^"|"$/g, '');

    switch (prefix) {
      case 'from':
        filters.from = value;
        break;
      case 'to':
        filters.to = value;
        break;
      case 'subject':
        filters.subject = value;
        break;
      case 'has':
        if (value === 'attachment') filters.hasAttachment = true;
        break;
      case 'after':
        filters.startDate = parseDate(value);
        break;
      case 'before':
        filters.endDate = parseDate(value);
        break;
      default:
        // Unknown prefix, keep as plain text
        remainingParts.push(part);
    }
  }

  filters.query = remainingParts.join(' ');
  return filters;
}

/**
 * Parse date string (YYYY-MM-DD or natural language)
 */
function parseDate(value: string): string | undefined {
  // Try ISO format first
  if (/^\d{4}-\d{2}-\d{2}$/.test(value)) {
    return value;
  }
  // Could add natural language parsing here (today, yesterday, last week)
  return undefined;
}

/**
 * Reconstruct search string from parsed filters
 */
export function filtersToSearchString(filters: ParsedFilters): string {
  const parts: string[] = [];

  if (filters.from) parts.push(`from:${filters.from}`);
  if (filters.to) parts.push(`to:${filters.to}`);
  if (filters.subject) parts.push(`subject:"${filters.subject}"`);
  if (filters.hasAttachment) parts.push('has:attachment');
  if (filters.startDate) parts.push(`after:${filters.startDate}`);
  if (filters.endDate) parts.push(`before:${filters.endDate}`);
  if (filters.query) parts.push(filters.query);

  return parts.join(' ');
}
```

```typescript
import { Box, Chip } from '@mui/material';
import { Mail, AtSign, FileText, Paperclip, Calendar } from 'lucide-react';
import { useTranslations } from '@/lib/hooks/use-translations';
import type { ParsedFilters } from '@/lib/utils/search-parser';

interface SearchFilterChipsProps {
  filters: ParsedFilters;
  onRemoveFilter: (key: keyof ParsedFilters) => void;
}

export function SearchFilterChips({ filters, onRemoveFilter }: SearchFilterChipsProps) {
  const t = useTranslations();

  const chips: { key: keyof ParsedFilters; label: string; icon: React.ReactNode }[] = [];

  if (filters.from) {
    chips.push({
      key: 'from',
      label: `${t.email?.search?.filters?.from || 'From'}: ${filters.from}`,
      icon: <Mail size={14} />,
    });
  }
  if (filters.to) {
    chips.push({
      key: 'to',
      label: `${t.email?.search?.filters?.to || 'To'}: ${filters.to}`,
      icon: <AtSign size={14} />,
    });
  }
  if (filters.subject) {
    chips.push({
      key: 'subject',
      label: `${t.email?.search?.filters?.subject || 'Subject'}: ${filters.subject}`,
      icon: <FileText size={14} />,
    });
  }
  if (filters.hasAttachment) {
    chips.push({
      key: 'hasAttachment',
      label: t.email?.search?.filters?.hasAttachment || 'Has attachment',
      icon: <Paperclip size={14} />,
    });
  }
  if (filters.startDate) {
    chips.push({
      key: 'startDate',
      label: `${t.email?.search?.filters?.after || 'After'}: ${filters.startDate}`,
      icon: <Calendar size={14} />,
    });
  }
  if (filters.endDate) {
    chips.push({
      key: 'endDate',
      label: `${t.email?.search?.filters?.before || 'Before'}: ${filters.endDate}`,
      icon: <Calendar size={14} />,
    });
  }

  if (chips.length === 0) return null;

  return (
    <Box sx={{ display: 'flex', gap: 0.5, mt: 0.5, flexWrap: 'wrap' }}>
      {chips.map((chip) => (
        <Chip
          key={chip.key}
          size="small"
          icon={chip.icon as React.ReactElement}
          label={chip.label}
          onDelete={() => onRemoveFilter(chip.key)}
          sx={{ height: 24, '& .MuiChip-label': { fontSize: '0.75rem' } }}
        />
      ))}
    </Box>
  );
}
```

1. Create ViewModeToggle.tsx (~50 lines):
```typescript
import { ToggleButton, ToggleButtonGroup, Tooltip } from '@mui/material';
import { List, Columns } from 'lucide-react';
import { useEmailUIStore } from '@/stores/email-ui-store';
import { useTranslations } from '@/lib/hooks/use-translations';

export function ViewModeToggle() {
  const viewMode = useEmailUIStore((s) => s.viewMode);
  const setViewMode = useEmailUIStore((s) => s.setViewMode);
  const t = useTranslations();

  return (
    <ToggleButtonGroup
      value={viewMode === 'thread' ? 'list' : viewMode}
      exclusive
      onChange={(_, value) => value && setViewMode(value)}
      size="small"
    >
      <Tooltip title={t.email?.viewMode?.list || 'List view'}>
        <ToggleButton value="list" aria-label="List view">
          <List size={18} />
        </ToggleButton>
      </Tooltip>
      <Tooltip title={t.email?.viewMode?.kanban || 'Kanban view'}>
        <ToggleButton value="kanban" aria-label="Kanban view">
          <Columns size={18} />
        </ToggleButton>
      </Tooltip>
    </ToggleButtonGroup>
  );
}
```

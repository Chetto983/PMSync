Pattern from existing research:
```typescript
import { useThemePersistence } from '@/hooks/use-theme-persistence';
import { IconButton, Tooltip, Menu, MenuItem, ListItemIcon, ListItemText } from '@mui/material';
import { Sun, Moon, Monitor, Check } from 'lucide-react';
import { useTranslations } from '@/lib/hooks/use-translations';

export function ThemeToggle() {
  const { theme, setTheme, resolvedTheme } = useThemePersistence();
  const t = useTranslations();
  const [anchorEl, setAnchorEl] = useState<null | HTMLElement>(null);

  const icon = resolvedTheme === 'dark' ? <Moon size={20} /> : <Sun size={20} />;

  const options = [
    { value: 'light', label: t.theme?.light || 'Light', icon: Sun },
    { value: 'dark', label: t.theme?.dark || 'Dark', icon: Moon },
    { value: 'system', label: t.theme?.system || 'System', icon: Monitor },
  ] as const;

  return (
    <>
      <Tooltip title={t.theme?.tooltip || 'Theme'}>
        <IconButton onClick={(e) => setAnchorEl(e.currentTarget)} size="small">
          {icon}
        </IconButton>
      </Tooltip>
      <Menu
        anchorEl={anchorEl}
        open={Boolean(anchorEl)}
        onClose={() => setAnchorEl(null)}
      >
        {options.map((opt) => (
          <MenuItem
            key={opt.value}
            onClick={() => { setTheme(opt.value); setAnchorEl(null); }}
            selected={theme === opt.value}
          >
            <ListItemIcon>
              <opt.icon size={18} />
            </ListItemIcon>
            <ListItemText>{opt.label}</ListItemText>
            {theme === opt.value && <Check size={16} />}
          </MenuItem>
        ))}
      </Menu>
    </>
  );
}
```

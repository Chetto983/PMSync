export interface TreeItem {
  id: string;

  parentId: string | null;
  children: TreeItem[];

  level: number;
  orderIndex: number;

export interface FlattenedItem extends TreeItem {

Functions:
- `buildTree(flatLabels: Label[], collapsedIds: Set<string>): TreeItem[]` — Build nested tree from flat API response. Sort children by orderIndex. Compute `aggregateCount` recursively (sum of own _count.emailLabels + all descendants). Set `collapsed` from collapsedIds (invert: items NOT in expandedIds are collapsed).
- `flattenTree(items: TreeItem[], parentId?: string | null, depth?: number): FlattenedItem[]` — Flatten tree into display-order array. Skip children of collapsed nodes.
- `getProjection(items: FlattenedItem[], activeId: string, overId: string, dragOffset: number, indentationWidth: number): { depth: number; parentId: string | null }` — Calculate projected drop position. Clamp depth to MAX_DEPTH - 1 (0-indexed). Use the pattern from RESEARCH.md code examples.
- `getDescendantIds(items: TreeItem[], parentId: string): string[]` — Return all descendant IDs recursively. Used for subtree delete warning count.
- `getAncestorIds(flatLabels: Label[], targetId: string): string[]` — Walk up parentId chain, return all ancestor IDs. Used for search filtering (show ancestors of matches).

Import `arrayMove` from `@dnd-kit/sortable` for getProjection.

**4. Query hooks additions (frontend/hooks/queries/use-labels.ts)**

Add a `useMoveLabel` mutation hook:

```typescript
export function useMoveLabel() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: ({ labelId, dto }: { labelId: string; dto: UpdateLabelDto }) =>
      labelsApi.updateLabel(labelId, dto),
    onMutate: async ({ labelId, dto }) => {
      await queryClient.cancelQueries({ queryKey: labelKeys.lists() });
      const previous = queryClient.getQueryData<Label[]>(labelKeys.list());
      // Optimistic: update parentId and orderIndex in cache
      if (previous) {
        queryClient.setQueryData(
          labelKeys.list(),
          previous.map((l) =>
            l.id === labelId
              ? { ...l, parentId: dto.parentId ?? l.parentId, orderIndex: dto.orderIndex ?? l.orderIndex }
              : l
          )
        );
      }
      return { previous };
    },
    onError: (_, __, context) => {
      if (context?.previous) {
        queryClient.setQueryData(labelKeys.list(), context.previous);
      }
    },
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: labelKeys.lists() });
    },
  });
}
```

private async archiveDescendants(tenantId: string, parentId: string) {
  const children = await this.prisma.userLabel.findMany({
    where: { tenantId, parentId, isArchived: false },
    select: { id: true, slug: true },
  });
  for (const child of children) {
    await this.prisma.userLabel.update({
      where: { id: child.id },
      data: {
        isArchived: true,
        slug: `${child.slug}-deleted-${Date.now()}`,
      },
    });
    await this.archiveDescendants(tenantId, child.id);
  }
}

const { expandedIds, toggleExpanded, pruneStaleIds } = useLabelTreeStore();

// Build tree from flat labels, using inverted expandedIds as collapsed state
const tree = useMemo(() => {

  const collapsedIds = new Set(
    userLabels.filter(l => !expandedIds.has(l.id)).map(l => l.id)

```typescript
/**
 * Check if moving a folder to a new parent would create a circular reference.
 * Walks up the parent chain from newParentId to ensure folderId is not an ancestor.
 *
 * @param folderId - The folder being moved
 * @param newParentId - The proposed new parent
 * @returns true if circular reference would be created, false otherwise
 */
private async wouldCreateCircularReference(
  folderId: string,
  newParentId: string,
): Promise<boolean> {
  // Self-reference is circular
  if (folderId === newParentId) {
    return true;
  }

  // Walk up parent chain from newParentId
  let currentParentId: string | null | undefined = newParentId;
  while (currentParentId) {
    if (currentParentId === folderId) {
      return true; // Found folderId in ancestor chain
    }

    const parent: { parentId: string | null } | null =
      await this.prisma.folder.findUnique({
        where: { id: currentParentId },
        select: { parentId: true },
      });

    currentParentId = parent?.parentId;
  }

  return false;
}
```

Place this method at the end of the FolderSyncService class, following the existing method structure.

Why this approach:
- Matches proven pattern from labels.service.ts (lines 413-428)
- Iterative algorithm prevents stack overflow on deep hierarchies
- Only selects parentId field for efficiency
- Handles null parent (root folders) correctly

```typescript
/**
 * Move a folder to a new parent or to root level.
 * Updates folder hierarchy (parentId, level) and recursively updates children levels.
 *
 * @param folderId - The folder to move
 * @param parentId - New parent ID (null = move to root)
 * @param tenantId - Tenant ID for isolation
 * @throws NotFoundException if folder or parent not found
 * @throws ConflictException if circular reference or system folder
 */
async moveFolder(
  folderId: string,
  parentId: string | null,
  tenantId: string,
): Promise<void> {
  // 1. Validate folder exists and belongs to tenant
  const folder = await this.prisma.folder.findUnique({
    where: { id: folderId },
  });

  if (!folder || folder.tenantId !== tenantId) {
    throw new NotFoundException('Folder not found');
  }

  // 2. Protect system folders
  if (folder.isSystem) {
    throw new ConflictException('Cannot move system folders');
  }

  // 3. Handle move to root
  if (parentId === null) {
    await this.prisma.folder.update({
      where: { id: folderId },
      data: {
        parentId: null,
        level: 0,
      },
    });

    // Recursively update children levels
    await this.updateChildrenLevels(folderId, 0);
    return;
  }

  // 4. Validate new parent exists and belongs to tenant
  const parent = await this.prisma.folder.findUnique({
    where: { id: parentId },
  });

  if (!parent || parent.tenantId !== tenantId) {
    throw new NotFoundException('Parent folder not found');
  }

  // 5. Check circular reference
  if (await this.wouldCreateCircularReference(folderId, parentId)) {
    throw new ConflictException('Cannot create circular folder hierarchy');
  }

  // 6. Update folder parent and level
  const newLevel = parent.level + 1;
  await this.prisma.folder.update({
    where: { id: folderId },
    data: {
      parentId,
      level: newLevel,
    },
  });

  // 7. Recursively update children levels
  await this.updateChildrenLevels(folderId, newLevel);
}

/**
 * Recursively update levels of all descendant folders.
 *
 * @param parentFolderId - The folder whose children to update
 * @param parentLevel - The parent's new level
 */
private async updateChildrenLevels(
  parentFolderId: string,
  parentLevel: number,
): Promise<void> {
  const children = await this.prisma.folder.findMany({
    where: { parentId: parentFolderId },
    select: { id: true },
  });

  for (const child of children) {
    const childLevel = parentLevel + 1;
    await this.prisma.folder.update({
      where: { id: child.id },
      data: { level: childLevel },
    });

    // Recursively update grandchildren
    await this.updateChildrenLevels(child.id, childLevel);
  }
}
```

Why this approach:
- Follows labels.service.ts pattern (lines 151-166)
- Validates tenant isolation before any mutation
- Prevents moving system folders (INBOX, SENT, etc.)
- Handles both "move to root" and "move to parent" cases
- Recursively updates children levels to maintain hierarchy integrity
- Uses ConflictException for business rule violations (matches existing code)

```typescript
@Patch(':folderId')
@Throttle({ default: { ttl: 60000, limit: 10 } }) // SEC-4: 10 requests per minute
@ApiOperation({ summary: 'Rename or move a folder' })
@ApiBody({
  description: 'Rename folder (newName) or move folder (parentId) or both',
  schema: {
    type: 'object',
    properties: {
      newName: { type: 'string', minLength: 1, maxLength: 100 },
      parentId: { type: 'string', nullable: true },
    },
  },
})
@ApiQuery({ name: 'wait', required: false, description: 'Wait for operation to complete (default: true)' })
@ApiResponse({ status: 200, description: 'Folder operation successful' })
@ApiResponse({ status: 202, description: 'Folder operation job queued' })
@ApiResponse({ status: 404, description: 'Folder not found' })
async updateFolder(
  @Request() req: AuthenticatedRequest,
  @Param('folderId') folderId: string,
  @Query('wait') wait?: string,
  @Body() renameDto?: RenameFolderDto,
  @Body() moveDto?: MoveFolderDto,
) {
  const tenantId = req.user.tenantId;
  const shouldWait = wait !== 'false'; // Default: true

  // Handle move operation (synchronous, no queue)
  if (moveDto?.parentId !== undefined) {
    await this.folderSyncService.moveFolder(
      folderId,
      moveDto.parentId,
      tenantId,
    );
  }

  // Handle rename operation (existing queue logic)
  if (renameDto?.newName) {
    const jobId = await this.folderOperationsQueue.enqueueRename(
      tenantId,
      folderId,
      renameDto.newName,
    );

    if (shouldWait) {
      await this.folderOperationsQueue.waitForCompletion(jobId, 30000);
      return { message: 'Folder renamed successfully' };
    }

    return {
      message: 'Folder rename job queued',
      jobId,
    };
  }

  // If only move was performed
  if (moveDto?.parentId !== undefined) {
    return { message: 'Folder moved successfully' };
  }

  throw new BadRequestException('No operation specified (newName or parentId required)');
}
```

Why this approach:
- Extends existing endpoint instead of creating new one (RESTful)
- Move operation is synchronous (no provider sync needed, just DB update)
- Rename operation remains async with queue (requires provider API call)
- Both operations can happen in single request (move THEN rename)
- Backward compatible: existing rename-only calls still work

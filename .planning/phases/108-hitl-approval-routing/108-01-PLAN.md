     try {
       await this.prisma.auditLog.create({
         data: {
           tenantId: data.tenantId,
           userId: data.userId,
           action: `HITL:${data.decision.toUpperCase()}`,
           entity: 'AiChat',
           entityId: data.toolCallId,
           changes: { toolName: data.toolName, conversationId: data.conversationId },
         },
       });

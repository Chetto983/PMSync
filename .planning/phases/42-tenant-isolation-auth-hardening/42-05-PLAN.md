    Add a new describe block that imports `withTenantContext` and tests the AsyncLocalStorage propagation end-to-end through the mock, proving the full chain from `withTenantContext('tenant-a', ...)` -> `getTenantIdFromContext()` returns 'tenant-a'.    
    ```typescript
    describe('Cross-tenant isolation via withTenantContext (TSEC-01 — real context chain)', () => {
      it('withTenantContext sets correct tenantId for the callback scope', () => {
        const result = withTenantContext('tenant-a', () => {
          return getTenantIdFromContext();
        });
        expect(result).toBe('tenant-a');
      });

      it('nested withTenantContext does not leak between tenants', () => {
        const results: (string | undefined)[] = [];

        withTenantContext('tenant-a', () => {
          results.push(getTenantIdFromContext());

          // Verify tenant-b context is isolated
          withTenantContext('tenant-b', () => {
            results.push(getTenantIdFromContext());
          });

          // After tenant-b scope exits, tenant-a is restored
          results.push(getTenantIdFromContext());
        });

        expect(results).toEqual(['tenant-a', 'tenant-b', 'tenant-a']);
      });

      it('simulated cross-tenant query proves WHERE injection isolates tenants', () => {
        // Simulate what the Prisma extension does: inject tenantId into WHERE
        function simulateWhereInjection(model: string): { tenantId: string } | undefined {
          if (!TENANT_SCOPED_MODELS.has(model)) return undefined;
          const tenantId = getTenantIdFromContext();
          if (!tenantId) throw new TenantContextRequiredError(model, 'findMany');
          return { tenantId };
        }

        // Tenant A context
        const tenantAWhere = withTenantContext('tenant-a', () =>
          simulateWhereInjection('Email'),
        );
        expect(tenantAWhere).toEqual({ tenantId: 'tenant-a' });

        // Tenant B context — same query, different result
        const tenantBWhere = withTenantContext('tenant-b', () =>
          simulateWhereInjection('Email'),
        );
        expect(tenantBWhere).toEqual({ tenantId: 'tenant-b' });

        // Proves: if DB has both tenant-a and tenant-b emails,
        // the WHERE clause ensures each tenant only sees their own
        expect(tenantAWhere!.tenantId).not.toBe(tenantBWhere!.tenantId);
      });
    });
    ```

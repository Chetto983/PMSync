**3. Create `use-ai-draft.ts` (NEW file, ~80 lines):**

```typescript
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { voltAgentApi } from '@/lib/api/ai';

export function useAiDraft(
  emailId?: string,
  options?: { tone?: string; locale?: string; enabled?: boolean },
) {
  return useQuery({
    queryKey: ['ai', 'draft', emailId, options?.tone],
    queryFn: async () => {
      const { data } = await voltAgentApi.generateReplyDraft({
        emailId: emailId!,
        tone: options?.tone,
        locale: options?.locale,
      });
      return data;
    },
    enabled: !!emailId && (options?.enabled !== false),
    staleTime: 0, // Always fresh draft
    gcTime: 5 * 60 * 1000, // Keep 5 min in cache
    retry: 1,
  });
}

export function useRegenerateDraft() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async (params: {
      emailId: string;
      tone?: string;
      instructions?: string;
      locale?: string;
    }) => {
      const { data } = await voltAgentApi.generateReplyDraft(params);
      return data;
    },
    onSuccess: (_data, variables) => {
      // Invalidate the query cache for this email's draft
      queryClient.invalidateQueries({ queryKey: ['ai', 'draft', variables.emailId] });
    },
  });
}

export function useRecordDraftFeedback() {
  return useMutation({
    mutationFn: (params: {
      draftId: string;
      action: 'accepted' | 'edited' | 'rejected' | 'regenerated';
      sentBody?: string;
    }) => voltAgentApi.recordDraftFeedback(params),
  });
}
```

```typescript
import { Box, Chip, Stack } from '@mui/material';
import { useTranslations } from '@/lib/hooks/use-translations';

export type DraftTone = 'formal' | 'casual' | 'shorter' | 'detailed';

interface ToneSelectorProps {
  selectedTone?: DraftTone;
  onToneChange: (tone: DraftTone | undefined) => void;
  disabled?: boolean;
}

export function ToneSelector({ selectedTone, onToneChange, disabled }: ToneSelectorProps) {
  const t = useTranslations();

  const tones: { key: DraftTone; label: string }[] = [
    { key: 'formal', label: t.email.aiDraft.toneFormal },
    { key: 'casual', label: t.email.aiDraft.toneCasual },
    { key: 'shorter', label: t.email.aiDraft.toneShorter },
    { key: 'detailed', label: t.email.aiDraft.toneDetailed },
  ];

  return (
    <Stack direction="row" spacing={0.5} sx={{ flexWrap: 'wrap', gap: 0.5 }}>
      {tones.map(({ key, label }) => (
        <Chip
          key={key}
          label={label}
          size="small"
          variant={selectedTone === key ? 'filled' : 'outlined'}
          color={selectedTone === key ? 'primary' : 'default'}
          onClick={() => onToneChange(selectedTone === key ? undefined : key)}
          disabled={disabled}
          sx={{ height: 24, fontSize: '0.75rem' }}
        />
      ))}
    </Stack>
  );
}
```

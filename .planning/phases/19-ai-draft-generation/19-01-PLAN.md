```prisma
model AiDraftFeedback {
  id            String   @id @default(cuid())
  tenantId      String
  userId        String
  emailId       String
  draftBody     String   // The AI-generated draft
  sentBody      String?  // What the user actually sent (null if rejected)
  action        String   @default("pending") // pending | accepted | edited | rejected | regenerated
  editDistance   Float?   // Levenshtein distance ratio (0 = identical, 1 = completely different)
  corrections   Json?    // Extracted correction patterns
  tone          String?
  locale        String?
  providerId    String?
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  @@index([tenantId, userId])
  @@index([tenantId, emailId])
  @@map("ai_draft_feedback")
}
```

**2. Create `draft-generation.service.ts` (NEW file, ~200 lines):**

```typescript
@Injectable()
export class DraftGenerationService {
  private readonly logger = new Logger(DraftGenerationService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly mistralService: MistralService,
    private readonly writingStyleService: WritingStyleService,
    @Optional() @Inject(PiiDetectionService)
    private readonly piiDetectionService: PiiDetectionService | null,
  ) {}

  async generateDraft(
    tenantId: string,
    userId: string,
    emailId: string,
    options: { tone?: string; instructions?: string; locale?: string; providerId?: string },
  ): Promise<{ draftId: string; body: string; subject: string }>
```

Implementation steps:
1. Fetch email via `prisma.email.findFirst({ where: { id: emailId, tenantId }, select: { id, subject, from, to, bodyText, snippet, providerId } })`
2. Throw `NotFoundException` if not found
3. Get writing style: `writingStyleService.getStylePromptSection(tenantId, userId, locale)` — pass the email's `providerId` if the per-provider method exists, else fall back to user-level
4. Detect incoming email language using a simple heuristic: check for Italian/English keywords or use the email's existing locale metadata. Create a private `detectLanguage(text: string): string` method that returns 'en' or 'it' based on common word patterns (Italian: "ciao", "buongiorno", "saluti", "cordiali", "gentile"; English default)
5. PII redaction: if `piiDetectionService` exists, call `piiDetectionService.detect(bodyText)` for matches and `piiDetectionService.redact(bodyText)` for safe text
6. Build system prompt:
   - Include writing style section
   - Set language instruction: "Write in {detectedLanguage}"
   - Add: "Do NOT include a signature or sign-off line. The email signature will be appended automatically."
   - Include tone if provided
   - Include user instructions if provided
   - Instruct JSON output: `{ "subject": "Re: ...", "body": "..." }`
7. Call `mistralService.completePrompt({ systemPrompt, userPrompt, temperature: 0.35, maxTokens: 800, jsonMode: true })`
8. Rehydrate PII tokens if piiDetectionService exists
9. Parse JSON response, extract subject and body
10. Create `AiDraftFeedback` record with `action: 'pending'`
11. Return `{ draftId, body, subject }`

Handle errors gracefully: if JSON parsing fails, wrap raw text as body with original subject prefixed with "Re: ".

**1. Create `ai-draft.controller.ts` (NEW file, ~120 lines):**

Follow the existing controller pattern from `voltagent.controller.ts`:

```typescript
import { Controller, Post, Body, Req, Get, UseGuards } from '@nestjs/common';
import { ApiTags, ApiBearerAuth, ApiOperation } from '@nestjs/swagger';
import { JwtAuthGuard } from '../../../auth/guards/jwt-auth.guard';
import { TenantGuard } from '../../../auth/guards/tenant.guard';
import { QuotaGuard, RequiresQuota } from '../../../subscription/guards/quota.guard';
import { RequiresConsent } from '../../../compliance/decorators/requires-consent.decorator';
import { DraftGenerationService } from '../services/draft-generation.service';
import { DraftFeedbackService } from '../services/draft-feedback.service';
import type { AuthenticatedRequest } from '../../../auth/types';
import { IsString, IsOptional } from 'class-validator';

// DTO classes (inline, small file)
class GenerateReplyDraftDto {
  @IsString() emailId: string;
  @IsOptional() @IsString() tone?: string;
  @IsOptional() @IsString() instructions?: string;
  @IsOptional() @IsString() locale?: string;
}

class RecordDraftFeedbackDto {
  @IsString() draftId: string;
  @IsString() action: 'accepted' | 'edited' | 'rejected' | 'regenerated';
  @IsOptional() @IsString() sentBody?: string;
}

@ApiTags('AI Drafts')
@ApiBearerAuth()
@UseGuards(JwtAuthGuard, TenantGuard, QuotaGuard)
@Controller('api/v1/voltagent')
export class AiDraftController {
  constructor(
    private readonly draftGenerationService: DraftGenerationService,
    private readonly draftFeedbackService: DraftFeedbackService,
  ) {}

  @Post('generate-reply-draft')
  @RequiresConsent('ai_email_processing')
  @RequiresQuota('ai')
  @ApiOperation({ summary: 'Generate AI reply draft for an email' })
  async generateReplyDraft(
    @Req() req: AuthenticatedRequest,
    @Body() dto: GenerateReplyDraftDto,
  ) {
    return this.draftGenerationService.generateDraft(
      req.user.tenantId,
      req.user.userId,
      dto.emailId,
      { tone: dto.tone, instructions: dto.instructions, locale: dto.locale },
    );
  }

  @Post('draft-feedback')
  @ApiOperation({ summary: 'Record feedback for an AI draft' })
  async recordDraftFeedback(
    @Req() req: AuthenticatedRequest,
    @Body() dto: RecordDraftFeedbackDto,
  ) {
    await this.draftFeedbackService.recordFeedback({
      draftId: dto.draftId,
      tenantId: req.user.tenantId,
      action: dto.action,
      sentBody: dto.sentBody,
    });
    return { success: true };
  }

  @Get('draft-stats')
  @ApiOperation({ summary: 'Get AI draft quality statistics' })
  async getDraftStats(@Req() req: AuthenticatedRequest) {
    const stats = await this.draftFeedbackService.getDraftStats(
      req.user.tenantId,
      req.user.userId,
    );
    return { success: true, stats };
  }
}
```

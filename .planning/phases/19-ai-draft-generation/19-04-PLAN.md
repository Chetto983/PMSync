**2. Create `use-writing-style.ts` (NEW file, ~70 lines):**

```typescript
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { voltAgentApi } from '@/lib/api/ai';

export function useWritingStyle(providerId?: string) {
  return useQuery({
    queryKey: ['ai', 'writing-style', providerId],
    queryFn: async () => {
      // Use existing endpoint: GET /api/v1/voltagent/stats/writing-style
      // TODO: If per-provider support is added to the endpoint, pass providerId as query param
      const { data } = await voltAgentApi.getWritingStyle();
      return data;
    },
    staleTime: 5 * 60 * 1000,
    gcTime: 30 * 60 * 1000,
  });
}

export function useUpdateManualSamples() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async (params: { samples: string; providerId?: string }) => {
      // Endpoint: POST /api/v1/voltagent/writing-style/manual-samples
      // This needs to be added to the controller — if not available yet,
      // use the existing update endpoint or note it for gap closure
      const { data } = await voltAgentApi.updateManualSamples(params);
      return data;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['ai', 'writing-style'] });
    },
  });
}

export function useDraftStats() {
  return useQuery({
    queryKey: ['ai', 'draft-stats'],
    queryFn: async () => {
      const { data } = await voltAgentApi.getDraftStats();
      return data.stats;
    },
    staleTime: 60 * 1000,
    gcTime: 5 * 60 * 1000,
  });
}
```

import { Injectable, Logger } from '@nestjs/common';
import { IEmailDriver } from '../drivers/interfaces/email-driver.interface';

/**
 * Callback type for getting fresh driver credentials
 * Avoids circular dependency with provider-specific services
 */
export type GetDriverCredentialsCallback = (providerId: string) => Promise<{
  provider: any;
  credentials: any;
  driverType: 'google' | 'microsoft' | 'imap';
}>;

/**
 * Callback type for creating connected driver
 */
export type CreateDriverCallback = (
  driverType: string,
  credentials: any,
) => Promise<IEmailDriver>;

/**
 * Result from driver refresh operation
 */
export interface RefreshResult {
  driver: IEmailDriver;
  tokenExpiresAt: Date | null | undefined;
}

@Injectable()
export class SyncTokenService {
  private readonly logger = new Logger(SyncTokenService.name);

  // CRIT-3 FIX: Token refresh threshold (refresh if less than 2 min remaining)
  private readonly TOKEN_REFRESH_THRESHOLD_MS = 2 * 60 * 1000;

  /**
   * CRIT-3 FIX: Check if token is about to expire and needs refresh
   * Compares current time against token expiry with threshold buffer
   */
  isTokenExpiringSoon(tokenExpiresAt: Date | null | undefined): boolean {
    if (!tokenExpiresAt) {
      return false; // Can't check without expiry time
    }
    const now = Date.now();
    const expiresAt = new Date(tokenExpiresAt).getTime();
    return now >= expiresAt - this.TOKEN_REFRESH_THRESHOLD_MS;
  }

  /**
   * CRIT-3 FIX: Refresh driver with new credentials if token is expiring
   * Returns new driver if refreshed, or existing driver if still valid
   *
   * @param currentDriver - Current connected driver
   * @param providerId - Provider ID for credential lookup
   * @param tokenExpiresAt - Current token expiry time
   * @param getDriverCredentials - Callback to get fresh credentials
   * @param createDriver - Callback to create new driver
   */
  async refreshDriverIfNeeded(
    currentDriver: IEmailDriver,
    providerId: string,
    tokenExpiresAt: Date | null | undefined,
    getDriverCredentials: GetDriverCredentialsCallback,
    createDriver: CreateDriverCallback,
  ): Promise<RefreshResult> {
    if (!this.isTokenExpiringSoon(tokenExpiresAt)) {
      return { driver: currentDriver, tokenExpiresAt };
    }

    this.logger.log(`CRIT-3: Token expiring soon, refreshing driver for provider ${providerId}`);

    try {
      // Disconnect old driver
      await currentDriver.disconnect();

      // Get fresh credentials (will trigger token refresh via ProviderTokenService)
      const { provider, credentials, driverType } = await getDriverCredentials(providerId);

      // Create new driver with fresh token
      const newDriver = await createDriver(driverType, credentials);

      if (!newDriver.isConnected()) {
        throw new Error('New driver failed to connect');
      }

      this.logger.log(`CRIT-3: Driver refreshed successfully for provider ${providerId}`);
      return { driver: newDriver, tokenExpiresAt: provider.tokenExpiresAt };
    } catch (error) {
      this.logger.error(
        `CRIT-3: Failed to refresh driver: ${error instanceof Error ? error.message : String(error)}`,
      );
      // FIX Phase 1.3: Throw error instead of returning expired driver
      throw new Error(
        `Token refresh failed for provider ${providerId}: ${error instanceof Error ? error.message : String(error)}`
      );
    }
  }
}

import { Injectable, Logger } from '@nestjs/common';
import { IEmailDriver } from '../drivers/interfaces/email-driver.interface';
import { CanonicalMessage } from '../drivers/interfaces/canonical.types';
import { SyncMessageProcessorService } from './sync-message-processor.service';
import { SyncTokenService, GetDriverCredentialsCallback, CreateDriverCallback } from './sync-token.service';
import { MessageProcessorContext } from '../interfaces/sync-context.interface';

/**
 * Configuration for full sync operation
 */
export interface FullSyncConfig {
  fullSyncMaxMessages: number;
  batchSize: number;
  getListMessagesOptions: () => Record<string, any>;
}

/**
 * Context for sync orchestration (token refresh callbacks)
 */
export interface OrchestrationContext {
  providerId: string;
  tenantId: string;
  tokenExpiresAt?: Date | null;
  userId?: string;
  getDriverCredentials: GetDriverCredentialsCallback;
  createDriver: CreateDriverCallback;
  messageProcessorContext: MessageProcessorContext;
}

/**
 * Result from full sync operation
 */
export interface FullSyncResult {
  messagesProcessed: number;
  newMessages: number;
  syncToken: string;
  driver: IEmailDriver;
}

/**
 * Result from incremental sync operation
 */
export interface IncrementalSyncResult {
  messagesProcessed: number;
  newMessages: number;
  newSyncToken: string;
  driver: IEmailDriver;
}

/**
 * Hook result for provider-specific incremental handling
 */
export interface IncrementalHookResult {
  skipProcessing: boolean;
  logMessage?: string;
}

@Injectable()
export class SyncOrchestrationService {
  private readonly logger = new Logger(SyncOrchestrationService.name);

  constructor(
    private readonly messageProcessor: SyncMessageProcessorService,
    private readonly tokenService: SyncTokenService,
  ) {}

  /**
   * Full sync using driver - shared implementation
   * FOLDER-FIX: Syncs ALL important folders, not just INBOX
   * DEDUP-FIX: Cross-folder deduplication to prevent duplicate emails
   * CRIT-3 FIX: Refreshes driver mid-sync if token expiring
   */
  async executeFullSync(
    initialDriver: IEmailDriver,
    config: FullSyncConfig,
    context: OrchestrationContext,
  ): Promise<FullSyncResult> {
    // FOLDER-FIX: Sync all important folders, not just INBOX
    const foldersToSync = ['INBOX', 'SENT', 'DRAFTS', 'TRASH', 'ARCHIVE'];
    const messagesPerFolder = Math.ceil(config.fullSyncMaxMessages / foldersToSync.length);

    this.logger.debug(
      `Full sync with driver - fetching ${messagesPerFolder} messages from each folder: ${foldersToSync.join(', ')}`,
    );

    let messagesProcessed = 0;
    let newMessages = 0;

    // CRIT-3 FIX: Track mutable driver and token expiry for mid-sync refresh
    let driver = initialDriver;
    let tokenExpiresAt = context.tokenExpiresAt;

    // DEDUP-FIX: Track processed externalIds across ALL folders
    const processedExternalIds = new Set<string>();

    for (const folder of foldersToSync) {
      let pageToken: string | undefined;
      let fetchedCount = 0;

      this.logger.debug(`Syncing folder: ${folder}`);

      try {
        do {
          // CRIT-3 FIX: Check if token needs refresh before each batch
          const refreshResult = await this.tokenService.refreshDriverIfNeeded(
            driver,
            context.providerId,
            tokenExpiresAt,
            context.getDriverCredentials,
            context.createDriver,
          );
          driver = refreshResult.driver;
          tokenExpiresAt = refreshResult.tokenExpiresAt;

          const result = await driver.listMessages({
            folder,
            pageSize: Math.min(config.batchSize, messagesPerFolder - fetchedCount),
            pageToken,
            ...config.getListMessagesOptions(),
          });

          // DEDUP-FIX: Filter out already-processed messages
          const dedupedMessages = result.messages.filter((msg) => {
            if (processedExternalIds.has(msg.id)) {
              this.logger.debug(
                `DEDUP-FIX: Skipping duplicate message ${msg.id}`,
              );
              return false;
            }
            processedExternalIds.add(msg.id);
            return true;
          });

          const batchResult = await this.messageProcessor.processBatch(
            driver,
            dedupedMessages,
            context.providerId,
            context.tenantId,
            context.messageProcessorContext,
          );
          messagesProcessed += batchResult.processed;
          newMessages += batchResult.newCount;

          fetchedCount += result.messages.length;
          pageToken = result.nextPageToken;

          if (fetchedCount >= messagesPerFolder || !pageToken) {
            break;
          }
        } while (pageToken);

        this.logger.debug(`Folder ${folder}: synced ${fetchedCount} messages`);
      } catch (folderError) {
        this.logger.warn(
          `Failed to sync folder ${folder}: ${folderError instanceof Error ? folderError.message : String(folderError)}`,
        );
      }
    }

    if (!driver.getInitialSyncToken) {
      throw new Error('Driver does not support getInitialSyncToken');
    }
    const syncToken = await driver.getInitialSyncToken();

    this.logger.debug(
      `Full sync complete: ${messagesProcessed} processed, ${newMessages} new across ${foldersToSync.length} folders`,
    );

    return { messagesProcessed, newMessages, syncToken, driver };
  }

  /**
   * Incremental sync using driver - shared implementation
   * CRIT-3 FIX: Refreshes driver mid-sync if token expiring
   */
  async executeIncrementalSync(
    initialDriver: IEmailDriver,
    lastSyncToken: string,
    context: OrchestrationContext,
    onIncrementalChanges: (
      changesResult: { changes: Array<{ type: string; messageId: string }>; newSyncToken: string },
      lastSyncToken: string,
    ) => IncrementalHookResult,
  ): Promise<IncrementalSyncResult> {
    this.logger.debug(`Incremental sync from token: ${lastSyncToken.substring(0, 20)}...`);

    let messagesProcessed = 0;
    let newMessages = 0;
    const processedIds = new Set<string>();

    let driver = initialDriver;
    let tokenExpiresAt = context.tokenExpiresAt;

    if (!driver.getChangesSince) {
      throw new Error('Driver does not support getChangesSince');
    }
    const changesResult = await driver.getChangesSince(lastSyncToken);

    const hookResult = onIncrementalChanges(changesResult, lastSyncToken);
    if (hookResult.skipProcessing) {
      if (hookResult.logMessage) {
        this.logger.debug(hookResult.logMessage);
      }
      return {
        messagesProcessed: 0,
        newMessages: 0,
        newSyncToken: changesResult.newSyncToken,
        driver,
      };
    }

    this.logger.debug(`Found ${changesResult.changes.length} changes`);

    let batchCounter = 0;
    const TOKEN_CHECK_INTERVAL = 50;

    for (const change of changesResult.changes) {
      if (processedIds.has(change.messageId)) continue;
      processedIds.add(change.messageId);

      batchCounter++;
      if (batchCounter % TOKEN_CHECK_INTERVAL === 0) {
        const refreshResult = await this.tokenService.refreshDriverIfNeeded(
          driver,
          context.providerId,
          tokenExpiresAt,
          context.getDriverCredentials,
          context.createDriver,
        );
        driver = refreshResult.driver;
        tokenExpiresAt = refreshResult.tokenExpiresAt;
      }

      try {
        if (change.type === 'deleted') {
          await this.messageProcessor.handleDeletion(
            change.messageId,
            context.providerId,
            context.tenantId,
          );
          messagesProcessed++;
        } else if (change.type === 'added' || change.type === 'modified') {
          const message = await driver.getMessage(change.messageId);
          if (message) {
            const upsertResult = await this.messageProcessor.upsertMessage(
              driver,
              message,
              context.providerId,
              context.tenantId,
              context.messageProcessorContext,
            );
            messagesProcessed++;
            if (upsertResult.wasNew) {
              newMessages++;
            }
          }
        }
      } catch (error) {
        this.logger.warn(
          `Failed to process change for ${change.messageId}: ${error instanceof Error ? error.message : String(error)}`,
        );
      }
    }

    this.logger.debug(
      `Incremental sync complete: ${messagesProcessed} processed, ${newMessages} new`,
    );

    return {
      messagesProcessed,
      newMessages,
      newSyncToken: changesResult.newSyncToken,
      driver,
    };
  }
}

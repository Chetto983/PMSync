import { Test, TestingModule } from '@nestjs/testing';
import { SyncTokenService } from '../sync-token.service';
import { IEmailDriver } from '../../drivers/interfaces/email-driver.interface';

describe('SyncTokenService', () => {
  let service: SyncTokenService;

  beforeEach(async () => {
    const module: TestingModule = await Test.createTestingModule({
      providers: [SyncTokenService],
    }).compile();

    service = module.get<SyncTokenService>(SyncTokenService);
  });

  describe('isTokenExpiringSoon', () => {
    it('should return false when tokenExpiresAt is null', () => {
      expect(service.isTokenExpiringSoon(null)).toBe(false);
    });

    it('should return false when tokenExpiresAt is undefined', () => {
      expect(service.isTokenExpiringSoon(undefined)).toBe(false);
    });

    it('should return false when token expires in more than 2 minutes', () => {
      const futureDate = new Date(Date.now() + 5 * 60 * 1000); // 5 min in future
      expect(service.isTokenExpiringSoon(futureDate)).toBe(false);
    });

    it('should return true when token expires in less than 2 minutes', () => {
      const soonDate = new Date(Date.now() + 1 * 60 * 1000); // 1 min in future
      expect(service.isTokenExpiringSoon(soonDate)).toBe(true);
    });

    it('should return true when token is already expired', () => {
      const pastDate = new Date(Date.now() - 60 * 1000); // 1 min ago
      expect(service.isTokenExpiringSoon(pastDate)).toBe(true);
    });

    it('should return true at exactly 2 minute threshold', () => {
      const thresholdDate = new Date(Date.now() + 2 * 60 * 1000); // exactly 2 min
      expect(service.isTokenExpiringSoon(thresholdDate)).toBe(true);
    });

    it('should return false just above 2 minute threshold', () => {
      const justAbove = new Date(Date.now() + 2 * 60 * 1000 + 1000); // 2 min + 1 sec
      expect(service.isTokenExpiringSoon(justAbove)).toBe(false);
    });
  });

  describe('refreshDriverIfNeeded', () => {
    let mockDriver: jest.Mocked<IEmailDriver>;
    let mockNewDriver: jest.Mocked<IEmailDriver>;
    let getCredentialsFn: jest.Mock;
    let createDriverFn: jest.Mock;

    beforeEach(() => {
      mockDriver = {
        disconnect: jest.fn().mockResolvedValue(undefined),
        isConnected: jest.fn().mockReturnValue(true),
      } as unknown as jest.Mocked<IEmailDriver>;

      mockNewDriver = {
        disconnect: jest.fn().mockResolvedValue(undefined),
        isConnected: jest.fn().mockReturnValue(true),
      } as unknown as jest.Mocked<IEmailDriver>;

      getCredentialsFn = jest.fn().mockResolvedValue({
        provider: { tokenExpiresAt: new Date(Date.now() + 60 * 60 * 1000) },
        credentials: { accessToken: 'new-token' },
        driverType: 'google',
      });

      createDriverFn = jest.fn().mockResolvedValue(mockNewDriver);
    });

    it('should return same driver when token is not expiring', async () => {
      const futureDate = new Date(Date.now() + 10 * 60 * 1000); // 10 min

      const result = await service.refreshDriverIfNeeded(
        mockDriver,
        'provider-123',
        futureDate,
        getCredentialsFn,
        createDriverFn,
      );

      expect(result.driver).toBe(mockDriver);
      expect(mockDriver.disconnect).not.toHaveBeenCalled();
      expect(getCredentialsFn).not.toHaveBeenCalled();
    });

    it('should refresh driver when token is expiring soon', async () => {
      const soonDate = new Date(Date.now() + 1 * 60 * 1000); // 1 min

      const result = await service.refreshDriverIfNeeded(
        mockDriver,
        'provider-123',
        soonDate,
        getCredentialsFn,
        createDriverFn,
      );

      expect(result.driver).toBe(mockNewDriver);
      expect(mockDriver.disconnect).toHaveBeenCalled();
      expect(getCredentialsFn).toHaveBeenCalledWith('provider-123');
      expect(createDriverFn).toHaveBeenCalledWith('google', { accessToken: 'new-token' });
    });

    it('should throw error when new driver fails to connect', async () => {
      const soonDate = new Date(Date.now() + 1 * 60 * 1000);
      mockNewDriver.isConnected.mockReturnValue(false);

      await expect(
        service.refreshDriverIfNeeded(
          mockDriver,
          'provider-123',
          soonDate,
          getCredentialsFn,
          createDriverFn,
        ),
      ).rejects.toThrow('New driver failed to connect');
    });

    it('should throw error when credential fetch fails', async () => {
      const soonDate = new Date(Date.now() + 1 * 60 * 1000);
      getCredentialsFn.mockRejectedValue(new Error('Auth failed'));

      await expect(
        service.refreshDriverIfNeeded(
          mockDriver,
          'provider-123',
          soonDate,
          getCredentialsFn,
          createDriverFn,
        ),
      ).rejects.toThrow('Token refresh failed for provider provider-123: Auth failed');
    });
  });
});

import { Test, TestingModule } from '@nestjs/testing';
import { SyncMessageProcessorService } from '../sync-message-processor.service';
import { PrismaService } from '../../../../prisma/prisma.service';
import { RealtimeEventsService } from '../../../realtime/services/realtime-events.service';
import { RedisLockService } from '../../../../common/services/redis-lock.service';

describe('SyncMessageProcessorService', () => {
  let service: SyncMessageProcessorService;
  let mockPrisma: jest.Mocked<PrismaService>;
  let mockRealtimeEvents: jest.Mocked<RealtimeEventsService>;
  let mockRedisLock: jest.Mocked<RedisLockService>;

  beforeEach(async () => {
    mockPrisma = {
      email: {
        findFirst: jest.fn(),
        findUnique: jest.fn(),
        update: jest.fn(),
      },
    } as unknown as jest.Mocked<PrismaService>;

    mockRealtimeEvents = {
      emitEmailDelete: jest.fn(),
      emitEmailNew: jest.fn(),
    } as unknown as jest.Mocked<RealtimeEventsService>;

    mockRedisLock = {
      withLock: jest.fn((key, ttl, fn) => fn()),
    } as unknown as jest.Mocked<RedisLockService>;

    const module: TestingModule = await Test.createTestingModule({
      providers: [
        SyncMessageProcessorService,
        { provide: PrismaService, useValue: mockPrisma },
        { provide: RealtimeEventsService, useValue: mockRealtimeEvents },
      ],
    }).compile();

    service = module.get<SyncMessageProcessorService>(SyncMessageProcessorService);
    service.setRedisLockService(mockRedisLock);
  });

  describe('parseImapExternalId', () => {
    it('should parse legacy format (just UID)', () => {
      const result = service.parseImapExternalId('12345');
      expect(result).toEqual({
        folder: null,
        uid: 12345,
        isSpecialMarker: false,
        markerType: null,
      });
    });

    it('should parse folder:uid format', () => {
      const result = service.parseImapExternalId('INBOX:12345');
      expect(result).toEqual({
        folder: 'INBOX',
        uid: 12345,
        isSpecialMarker: false,
        markerType: null,
      });
    });

    it('should parse folder with colon in name', () => {
      const result = service.parseImapExternalId('Work:Projects:99');
      expect(result).toEqual({
        folder: 'Work:Projects',
        uid: 99,
        isSpecialMarker: false,
        markerType: null,
      });
    });

    it('should detect NEW special marker', () => {
      const result = service.parseImapExternalId('INBOX:__NEW__123');
      expect(result).toEqual({
        folder: 'INBOX',
        uid: null,
        isSpecialMarker: true,
        markerType: 'NEW',
      });
    });

    it('should detect EXPUNGE special marker', () => {
      const result = service.parseImapExternalId('SENT:__EXPUNGE__456');
      expect(result).toEqual({
        folder: 'SENT',
        uid: null,
        isSpecialMarker: true,
        markerType: 'EXPUNGE',
      });
    });

    it('should detect FLAGS special marker', () => {
      const result = service.parseImapExternalId('DRAFTS:__FLAGS__789');
      expect(result).toEqual({
        folder: 'DRAFTS',
        uid: null,
        isSpecialMarker: true,
        markerType: 'FLAGS',
      });
    });

    it('should return null uid for invalid number', () => {
      const result = service.parseImapExternalId('INBOX:abc');
      expect(result).toEqual({
        folder: 'INBOX',
        uid: null,
        isSpecialMarker: false,
        markerType: null,
      });
    });

    it('should return null uid for legacy invalid format', () => {
      const result = service.parseImapExternalId('notanumber');
      expect(result).toEqual({
        folder: null,
        uid: null,
        isSpecialMarker: false,
        markerType: null,
      });
    });
  });

  describe('handleDeletion', () => {
    it('should return needsReconciliation true for EXPUNGE marker', async () => {
      const result = await service.handleDeletion(
        'INBOX:__EXPUNGE__123',
        'provider-1',
        'tenant-1',
      );

      expect(result).toEqual({
        needsReconciliation: true,
        folder: 'INBOX',
      });
      expect(mockPrisma.email.findFirst).not.toHaveBeenCalled();
    });

    it('should mark email as deleted when found', async () => {
      mockPrisma.email.findFirst.mockResolvedValue({
        id: 'email-123',
        folder: 'INBOX',
        externalId: '12345',
      });
      mockPrisma.email.update.mockResolvedValue({} as any);

      const result = await service.handleDeletion(
        '12345',
        'provider-1',
        'tenant-1',
      );

      expect(result).toEqual({ needsReconciliation: false });
      expect(mockPrisma.email.update).toHaveBeenCalledWith({
        where: { id_tenantId: { id: 'email-123', tenantId: 'tenant-1' } },
        data: { isDeleted: true, folder: 'TRASH' },
      });
      expect(mockRealtimeEvents.emitEmailDelete).toHaveBeenCalledWith('tenant-1', {
        emailId: 'email-123',
        providerId: 'provider-1',
        folder: 'INBOX',
        reason: 'message-deleted',
      });
    });

    it('should handle email not found gracefully', async () => {
      mockPrisma.email.findFirst.mockResolvedValue(null);

      const result = await service.handleDeletion(
        'nonexistent:999',
        'provider-1',
        'tenant-1',
      );

      expect(result).toEqual({ needsReconciliation: false });
      expect(mockPrisma.email.update).not.toHaveBeenCalled();
      expect(mockRealtimeEvents.emitEmailDelete).not.toHaveBeenCalled();
    });
  });

  describe('upsertMessage', () => {
    it('should throw error when RedisLockService not set', async () => {
      const serviceWithoutLock = new SyncMessageProcessorService(
        mockPrisma,
        mockRealtimeEvents,
      );

      await expect(
        serviceWithoutLock.upsertMessage(
          {} as any,
          { id: '1', from: { email: 'test@test.com' } } as any,
          'provider-1',
          'tenant-1',
          {
            tenantId: 'tenant-1',
            providerId: 'provider-1',
            getEmbeddingQueue: () => ({} as any),
            getBatchIndexingService: () => ({} as any),
            processAttachments: async () => {},
            followUpService: {} as any,
            writingStyleService: {} as any,
          },
        ),
      ).rejects.toThrow('RC-9: RedisLockService not available');
    });
  });
});

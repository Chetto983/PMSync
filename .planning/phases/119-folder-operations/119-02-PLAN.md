   ```typescript
   // Move emails to TRASH before folder deletion
   const trashFolder = await this.prisma.folder.findFirst({
     where: { providerId: folder.providerId, specialUse: 'TRASH' },
   });
   if (trashFolder) {
     await this.prisma.email.updateMany({
       where: { folderId: folder.id },
       data: { folderId: trashFolder.id, folder: trashFolder.path },
     });
   }
   ```

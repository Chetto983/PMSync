       import { NestFactory } from '@nestjs/core';
       import { AppModule } from '../src/app.module';
       import { PrismaService } from '../src/prisma/prisma.service';
       import { LabelAccuracyService } from '../src/modules/ai/services/label-accuracy.service';
       import { getSharedRedisConnection, releaseSharedRedisConnection } from '../src/common/config/queue.config';
       import { TP_IMPLICIT_WINDOW_DAYS } from '../src/modules/ai/utils/triage-correction-weight';

       interface BackfillOptions {
         tenantId?: string;
         dryRun?: boolean;
       }

       interface BackfillStats {
         tenantsProcessed: number;
         feedbackApplied: number;
         feedbackSkipped: number;
         tpCredited: number;
         tpSkipped: number;
         beforeCount: number;
         afterCount: number;
       }

       const TP_CREDITED  = (t: string) => `triage:accuracy-backfill:credited:${t}`;
       const DEDUP_TTL_SECONDS = 7 * 24 * 3600;

       // BACKFILL_LOOKBACK_DAYS is the INDEPENDENT 30-day window bound for the
       // feedback pass; the composed 44 is `TP_IMPLICIT_WINDOW_DAYS + 30`.
       const BACKFILL_LOOKBACK_DAYS = 30;
       const LOOKBACK_FEEDBACK_DAYS = BACKFILL_LOOKBACK_DAYS;
       const LOOKBACK_TP_UPPER_DAYS = TP_IMPLICIT_WINDOW_DAYS;
       const LOOKBACK_TP_LOWER_DAYS = TP_IMPLICIT_WINDOW_DAYS + BACKFILL_LOOKBACK_DAYS;

       export async function runBackfill(options: BackfillOptions = {}): Promise<BackfillStats> {
         const app = await NestFactory.createApplicationContext(AppModule, { bufferLogs: false });
         const prisma = app.get(PrismaService);
         const accuracy = app.get(LabelAccuracyService);
         const redis = getSharedRedisConnection();
         const stats: BackfillStats = {
           tenantsProcessed: 0,
           feedbackApplied: 0,
           feedbackSkipped: 0,
           tpCredited: 0,
           tpSkipped: 0,
           beforeCount: 0,
           afterCount: 0,
         };

         try {
           const tenants = options.tenantId
             ? [{ tenantId: options.tenantId }]
             : await prisma.emailTriageFeedback.findMany({
                 where: { createdAt: { gte: new Date(Date.now() - LOOKBACK_FEEDBACK_DAYS * 24 * 3600 * 1000) } },
                 distinct: ['tenantId'],
                 select: { tenantId: true },
               });

           stats.beforeCount = await prisma.labelAccuracyProfile.count(
             options.tenantId ? { where: { tenantId: options.tenantId } } : undefined,
           );

           for (const { tenantId } of tenants) {
             stats.tenantsProcessed++;
             const labels = await prisma.userLabel.findMany({
               where: { tenantId },
               select: { id: true, slug: true, name: true },
             });
             const bySlug = new Map(labels.map((l) => [l.slug, l.id]));
             const byName = new Map(labels.map((l) => [l.name.toLowerCase(), l.id]));

             // Pass 1: feedback rows
             const cutoff = new Date(Date.now() - LOOKBACK_FEEDBACK_DAYS * 24 * 3600 * 1000);
             const feedback = await prisma.emailTriageFeedback.findMany({
               where: { tenantId, type: 'label_correction', createdAt: { gte: cutoff } },
               select: { id: true, emailId: true, signal: true, oldValue: true, newValue: true, createdAt: true },
             });
             for (const row of feedback) {
               const seen = await redis.sismember(FEEDBACK_SEEN(tenantId), row.id);
               if (seen === 1) {
                 stats.feedbackSkipped++;
                 continue;
               }
               for (const [labelName, kind] of [
                 [row.oldValue, 'fp' as const],
                 [row.newValue, 'fn' as const],
               ]) {
                 if (!labelName) continue;
                 const labelId = bySlug.get(labelName) ?? byName.get(labelName.toLowerCase());
                 if (!labelId) continue;

                   await accuracy.applyDelta({
                     tenantId,
                     labelId,
                     kind,
                     delta: 1,
                     signal: row.signal as 'edit' | 'correction',
                   });
                 }
               }
               if (!options.dryRun) {
                 await redis.sadd(FEEDBACK_SEEN(tenantId), row.id);
                 await redis.expire(FEEDBACK_SEEN(tenantId), DEDUP_TTL_SECONDS);
               }
               stats.feedbackApplied++;
             }

             // Pass 2: TP credits for emails with clean aiLabels (processedAt 14-44d ago — B-1 derived)
             const tpUpper = new Date(Date.now() - LOOKBACK_TP_UPPER_DAYS * 24 * 3600 * 1000);
             const tpLower = new Date(Date.now() - LOOKBACK_TP_LOWER_DAYS * 24 * 3600 * 1000);
             const emails = await prisma.email.findMany({
               where: { tenantId, processedAt: { gte: tpLower, lte: tpUpper }, aiLabels: { isEmpty: false } },
               select: { id: true, processedAt: true, aiLabels: true },
             });
             for (const email of emails) {
               for (const labelName of email.aiLabels ?? []) {
                 const labelId = bySlug.get(labelName) ?? byName.get(labelName.toLowerCase());
                 if (!labelId) continue;
                 const key = `${email.id}:${labelId}`;
                 const already = await redis.sismember(TP_CREDITED(tenantId), key);
                 if (already === 1) { stats.tpSkipped++; continue; }
                 const hasCorrection = await prisma.emailTriageFeedback.count({
                   where: {
                     tenantId,
                     emailId: email.id,
                     OR: [{ oldValue: labelName }, { newValue: labelName }],
                   },
                 });
                 if (hasCorrection > 0) { stats.tpSkipped++; continue; }
                 if (!options.dryRun) {
                   await accuracy.applyDelta({
                     tenantId, labelId, kind: 'tp', delta: 1, signal: 'system',
                   });
                   await redis.sadd(TP_CREDITED(tenantId), key);
                   await redis.expire(TP_CREDITED(tenantId), DEDUP_TTL_SECONDS);
                 }
                 stats.tpCredited++;
               }
             }
           }

           stats.afterCount = await prisma.labelAccuracyProfile.count(
             options.tenantId ? { where: { tenantId: options.tenantId } } : undefined,
           );
           if (stats.afterCount < stats.beforeCount) {
             throw new Error(
               `Backfill integrity failure: row count shrank (before=${stats.beforeCount}, after=${stats.afterCount})`,
             );
           }
           return stats;
         } finally {
           releaseSharedRedisConnection();
           await app.close();
         }
       }

       if (require.main === module) {
         const args = process.argv.slice(2);
         const tenantArg = args.find((a) => a.startsWith('--tenant='));
         const dryRun = args.includes('--dry-run');
         runBackfill({
           tenantId: tenantArg ? tenantArg.split('=')[1] : undefined,
           dryRun,
         })
           .then((stats) => {
             // eslint-disable-next-line no-console
             console.log('Backfill complete:', stats);
             process.exit(0);
           })
           .catch((err) => {
             // eslint-disable-next-line no-console
             console.error('Backfill failed:', err);
             process.exit(1);
           });
       }

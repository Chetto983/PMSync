1. **Event contract file (D-15).** Declare `AccuracyDeltaEvent` (emitted by the service after every mutation) and `LabelAccuracyEvent` (the inbound listener contract consumed from Phase-147 feedback hooks). Single file, zero logic — the public contract.

2. **LabelAccuracyService.** Owns all counter math: `applyDelta` (inline per-feedback FP/FN math + Wilson recompute + post-transaction event emit per W-2), `sweepTps` (nightly 1-day slice per D-06, day offsets derived from `TP_IMPLICIT_WINDOW_DAYS` per B-1), `rolloverWindows` (per D-08, window-aging FP/FN + NEW TP-aging decrement per B-6 for `windowKey='7d'` and `'30d'`, append-only for `'all'`), `read` (internal, used by Plan 05 controller). Every query filters by `tenantId`. Counter injection forbidden — only this service mutates counters. Pure derivation helpers live in a sibling `label-accuracy.compute.ts` per W-6 to keep the service under 500 lines after Plan 04's gauge-refresh block lands. Metric DI tokens are `@Optional()` (B-4) — Plan 03 spec wires null, Plan 04 wires real mocks.

3. **Queue + worker + listener + sweeper-handler registration.** `LabelAccuracyQueueService` owns the `'accuracy-update'` BullMQ queue. `LabelAccuracyListener` listens to `'triage.accuracy.applied'` events emitted by Phase-147 `TriageFeedbackService` (wired in Plan 04) and enqueues one job per `(tenantId, labelId, kind)` mutation. Listener separator guard covers EVERY component including the resolved labelId (B-7). `LabelAccuracyService` registers the `TRIAGE_ACCURACY_SWEEP` cron handler in its constructor.

Plan 04 then wires the upstream emit (TriageFeedbackService → `'triage.accuracy.applied'`), the Prometheus metric providers, and the backfill script. Plan 05 wires the read API. Plan 06 wires the frontend.

Purpose: keep the "counter engine" and "wiring to Phase 147 / Prometheus / HTTP / UI" in separate plans so each can be reviewed in isolation and Prometheus instrumentation (Plan 04) can be added without rewriting this plan's queue worker.         * Phase 148 D-07 / D-08 / D-11 pure-math helpers.
        *
        * Extracted from `label-accuracy.service.ts` so the service file stays below
        * the CLAUDE.md 500-line cap after Plan 04 appends the runSweep gauge-refresh

       import { wilsonInterval } from '../utils/wilson-interval';

       export interface RawCounters {
         tp: number;
         fp: number;
         fn: number;
       }

       export interface DerivedCounters {
         sampleSize: number;
         precision: number | null;
         recall: number | null;
         f1: number | null;
         wilsonPrecLow: number | null;
         wilsonPrecHigh: number | null;
         wilsonRecLow: number | null;
         wilsonRecHigh: number | null;
       }

       /**
        * D-07 materialized derivation. F1 is null when either precision or recall
        * is null OR when p + r === 0 (which catches the all-zero case). Wilson
        * bounds are null when the denominator is 0 (no data points).
        */
       export function computeDerived(counters: RawCounters): DerivedCounters {
         const { tp, fp, fn } = counters;
         const sampleSize = tp + fp + fn;
         const precision = tp + fp > 0 ? tp / (tp + fp) : null;
         const recall = tp + fn > 0 ? tp / (tp + fn) : null;
         const f1 =
           precision != null && recall != null && precision + recall > 0
             ? (2 * precision * recall) / (precision + recall)
             : null;
         const wPrec = wilsonInterval(tp, tp + fp);
         const wRec = wilsonInterval(tp, tp + fn);
         return {
           sampleSize,
           precision,
           recall,
           f1,
           wilsonPrecLow: tp + fp > 0 ? wPrec.low : null,
           wilsonPrecHigh: tp + fp > 0 ? wPrec.high : null,
           wilsonRecLow: tp + fn > 0 ? wRec.low : null,
           wilsonRecHigh: tp + fn > 0 ? wRec.high : null,
         };
       }

          *   1. LabelAccuracyQueueService worker (per-feedback FP/FN)
          *   2. sweepTps (TP credit)
          *   3. rolloverWindows (window-aging decrement)
          *   4. Plan 04 backfill script
          * NEVER called by the read controller (Plan 05).
          *
          * W-2: EventEmitter2.emit is dispatched AFTER `await $transaction` resolves.
          * Phase 147 D-01 invariant preserved: handlers cannot corrupt the counter write.

         async applyDelta(input: ApplyDeltaInput): Promise<void> {
           const { tenantId, labelId, kind, delta, triageSource, signal } = input;
           if (!tenantId || !labelId || !Number.isFinite(delta)) return;

           for (const windowKey of ['7d', '30d', 'all'] as const) {
             const event = await this.prisma.$transaction(async (tx) => {
               const existing = await tx.labelAccuracyProfile.findUnique({
                 where: {
                   tenantId_labelId_windowKey: { tenantId, labelId, windowKey },
                 },
               });
               const prev = existing ?? {
                 tp: 0,
                 fp: 0,
                 fn: 0,
                 precision: null as number | null,
                 recall: null as number | null,
                 f1: null as number | null,
                 sampleSize: 0,
               };

               const next = {
                 tp: kind === 'tp' ? Math.max(0, prev.tp + delta) : prev.tp,
                 fp: kind === 'fp' ? Math.max(0, prev.fp + delta) : prev.fp,
                 fn: kind === 'fn' ? Math.max(0, prev.fn + delta) : prev.fn,
               };
               const derived = computeDerived(next);

               await tx.labelAccuracyProfile.upsert({
                 where: {
                   tenantId_labelId_windowKey: { tenantId, labelId, windowKey },
                 },
                 create: { tenantId, labelId, windowKey, ...next, ...derived },
                 update: { ...next, ...derived },
               });

               if (triageSource) {
                 const brk = await tx.labelAccuracySourceBreakdown.findUnique({
                   where: {
                     tenantId_labelId_triageSource_windowKey: {
                       tenantId,
                       labelId,
                       triageSource,
                       windowKey,
                     },
                   },
                 });

                 const bNext = {
                   tp: kind === 'tp' ? Math.max(0, bPrev.tp + delta) : bPrev.tp,
                   fp: kind === 'fp' ? Math.max(0, bPrev.fp + delta) : bPrev.fp,
                   fn: kind === 'fn' ? Math.max(0, bPrev.fn + delta) : bPrev.fn,
                 };
                 const bDerived = computeDerived(bNext);
                 await tx.labelAccuracySourceBreakdown.upsert({
                   where: {
                     tenantId_labelId_triageSource_windowKey: {
                       tenantId, labelId, triageSource, windowKey,
                     },
                   },
                   create: { tenantId, labelId, triageSource, windowKey, ...bNext, ...bDerived },
                   update: { ...bNext, ...bDerived },
                 });

               const accEvent: AccuracyDeltaEvent = {
                 tenantId,
                 labelId,
                 windowKey,

                   recall: prev.recall ?? null,

          * D-06 TP sweeper: 1-day slice of aiLabels with no correction = TP credit.
          * Day offsets DERIVED from TP_IMPLICIT_WINDOW_DAYS (B-1 single source of truth).

         async sweepTps(tenantId: string): Promise<{ credited: number }> {
           const upper = new Date(Date.now() - TP_SLICE_UPPER * 24 * 3600 * 1000);
           const lower = new Date(Date.now() - TP_SLICE_LOWER * 24 * 3600 * 1000);
           const emails = await this.prisma.email.findMany({
             where: {
               tenantId,
               processedAt: { gte: lower, lte: upper },
               aiLabels: { isEmpty: false },
             },
             select: { id: true, processedAt: true, aiLabels: true },
           });
           if (emails.length === 0) return { credited: 0 };

           const labels = await this.prisma.userLabel.findMany({
             where: { tenantId },
             select: { id: true, slug: true, name: true },
           });
           const bySlug = new Map(labels.map((l) => [l.slug, l.id]));
           const byName = new Map(labels.map((l) => [l.name.toLowerCase(), l.id]));

           let credited = 0;
           for (const email of emails) {
             for (const labelName of email.aiLabels ?? []) {
               const labelId = bySlug.get(labelName) ?? byName.get(labelName.toLowerCase());
               if (!labelId) continue;

               const hasCorrection = await this.prisma.emailTriageFeedback.count({
                 where: {
                   tenantId,
                   emailId: email.id,
                   OR: [{ oldValue: labelName }, { newValue: labelName }],
                   createdAt: { gte: email.processedAt ?? lower, lte: new Date() },
                 },
               });
               if (hasCorrection > 0) continue;

               await this.applyDelta({
                 tenantId, labelId, kind: 'tp', delta: 1, signal: 'system',
               });
               credited++;
             }
           }
           return { credited };
         }

          * '`all`' is monotonic (no-op).

         async rolloverWindows(tenantId: string): Promise<{ decremented: number }> {
           let decremented = 0;
           const redis = this.getRedis();

           for (const windowKey of ['7d', '30d'] as const) {
             const windowDays = WINDOW_DAYS[windowKey]!;
             const setKey = ROLLOVER_SET_KEY(tenantId, windowKey);

             // ── Pass 1: FP/FN aging ────────────────────────────────────────
             const fpLower = new Date(Date.now() - (windowDays + ROLLOVER_SLICE_DAYS + 1) * 24 * 3600 * 1000);
             const fpUpper = new Date(Date.now() - (windowDays + ROLLOVER_SLICE_DAYS) * 24 * 3600 * 1000);
             const agedOut = await this.prisma.emailTriageFeedback.findMany({
               where: {
                 tenantId,
                 type: 'label_correction',
                 createdAt: { gte: fpLower, lt: fpUpper },
               },
               select: { id: true, oldValue: true, newValue: true },
             });

             if (agedOut.length > 0) {
               const labels = await this.prisma.userLabel.findMany({
                 where: { tenantId },
                 select: { id: true, slug: true, name: true },
               });
               const bySlug = new Map(labels.map((l) => [l.slug, l.id]));
               const byName = new Map(labels.map((l) => [l.name.toLowerCase(), l.id]));

               for (const row of agedOut) {
                 if (row.oldValue) {
                   const labelId = bySlug.get(row.oldValue) ?? byName.get(row.oldValue.toLowerCase());
                   if (labelId) {
                     const dedupKey = `${row.id}:${labelId}:fp`;
                     const alreadyDone = redis
                       ? await redis.sismember(setKey, dedupKey)
                       : 0;
                     if (alreadyDone === 0) {
                       await this.decrementSingleWindow(tenantId, labelId, windowKey, 'fp');
                       if (redis) {
                         await redis.sadd(setKey, dedupKey);
                         await redis.expire(setKey, ROLLOVER_SET_TTL_SECONDS);
                       }
                       decremented++;
                     }
                   }
                 }
                 if (row.newValue) {
                   const labelId = bySlug.get(row.newValue) ?? byName.get(row.newValue.toLowerCase());
                   if (labelId) {
                     const dedupKey = `${row.id}:${labelId}:fn`;
                     const alreadyDone = redis
                       ? await redis.sismember(setKey, dedupKey)
                       : 0;
                     if (alreadyDone === 0) {
                       await this.decrementSingleWindow(tenantId, labelId, windowKey, 'fn');
                       if (redis) {
                         await redis.sadd(setKey, dedupKey);
                         await redis.expire(setKey, ROLLOVER_SET_TTL_SECONDS);
                       }
                       decremented++;
                     }
                   }
                 }
               }
             }

             // ── Pass 2: B-6 TP-aging ───────────────────────────────────────
             decremented += await this.decrementTpAging(tenantId, windowKey, redis);
           }
           return { decremented };
         }

          * issued by the sweeper at `processedAt + TP_IMPLICIT_WINDOW_DAYS` now falls
          * outside the window, and decrements TP by 1 in that window only.

          * Scan slice: processedAt ∈ [now - (windowDays + TP_IMPLICIT_WINDOW_DAYS + ROLLOVER_SLICE),
          *                            now - (windowDays + TP_IMPLICIT_WINDOW_DAYS))
          *
          * Idempotent via Redis set `triage:accuracy-rollover:decremented:${tenantId}:${windowKey}`
          * with per-pair key `${emailId}:${labelId}:tp`. TTL 7d.
          */
         private async decrementTpAging(
           tenantId: string,
           windowKey: '7d' | '30d',
           redis: Redis | null,
         ): Promise<number> {
           const windowDays = WINDOW_DAYS[windowKey]!;
           const tpLower = new Date(
             Date.now() -
               (windowDays + TP_IMPLICIT_WINDOW_DAYS + ROLLOVER_SLICE_DAYS) * 24 * 3600 * 1000,
           );
           const tpUpper = new Date(
             Date.now() - (windowDays + TP_IMPLICIT_WINDOW_DAYS) * 24 * 3600 * 1000,
           );
           const emails = await this.prisma.email.findMany({
             where: {
               tenantId,
               processedAt: { gte: tpLower, lt: tpUpper },
               aiLabels: { isEmpty: false },
             },
             select: { id: true, processedAt: true, aiLabels: true },
           });
           if (emails.length === 0) return 0;

           const labels = await this.prisma.userLabel.findMany({
             where: { tenantId },
             select: { id: true, slug: true, name: true },
           });
           const bySlug = new Map(labels.map((l) => [l.slug, l.id]));
           const byName = new Map(labels.map((l) => [l.name.toLowerCase(), l.id]));
           const setKey = ROLLOVER_SET_KEY(tenantId, windowKey);
           let decremented = 0;

           for (const email of emails) {
             for (const labelName of email.aiLabels ?? []) {
               const labelId = bySlug.get(labelName) ?? byName.get(labelName.toLowerCase());
               if (!labelId) continue;
               // Was this pair TP-credited? (i.e., no correction feedback exists)
               const hasCorrection = await this.prisma.emailTriageFeedback.count({
                 where: {
                   tenantId,
                   emailId: email.id,
                   OR: [{ oldValue: labelName }, { newValue: labelName }],
                 },
               });
               if (hasCorrection > 0) continue;

               const dedupKey = `${email.id}:${labelId}:tp`;
               const alreadyDone = redis ? await redis.sismember(setKey, dedupKey) : 0;
               if (alreadyDone === 1) continue;

               await this.decrementSingleWindow(tenantId, labelId, windowKey, 'tp');
               if (redis) {
                 await redis.sadd(setKey, dedupKey);
                 await redis.expire(setKey, ROLLOVER_SET_TTL_SECONDS);
               }
               decremented++;
             }
           }
           return decremented;
         }

         /** windowKey-scoped delta (not the public 3-window loop). Used by rolloverWindows. */
         private async decrementSingleWindow(
           tenantId: string,
           labelId: string,
           windowKey: AccuracyWindowKey,
           kind: LabelAccuracyKind,
         ): Promise<void> {
           const event = await this.prisma.$transaction(async (tx) => {
             const existing = await tx.labelAccuracyProfile.findUnique({
               where: { tenantId_labelId_windowKey: { tenantId, labelId, windowKey } },
             });
             if (!existing) return null;
             const next = {
               tp: kind === 'tp' ? Math.max(0, existing.tp - 1) : existing.tp,
               fp: kind === 'fp' ? Math.max(0, existing.fp - 1) : existing.fp,
               fn: kind === 'fn' ? Math.max(0, existing.fn - 1) : existing.fn,
             };
             const derived = computeDerived(next);
             await tx.labelAccuracyProfile.update({
               where: { tenantId_labelId_windowKey: { tenantId, labelId, windowKey } },
               data: { ...next, ...derived },
             });
             const accEvent: AccuracyDeltaEvent = {
               tenantId,
               labelId,
               windowKey,
               before: {
                 precision: existing.precision,
                 recall: existing.recall,
                 f1: existing.f1,
                 sampleSize: existing.sampleSize,
               },
               after: {
                 precision: derived.precision,
                 recall: derived.recall,
                 f1: derived.f1,
                 sampleSize: derived.sampleSize,
               },
               deltaF1:
                 existing.sampleSize < LEARNING_THRESHOLD ||
                 derived.sampleSize < LEARNING_THRESHOLD ||
                 existing.f1 == null ||
                 derived.f1 == null
                   ? null
                   : derived.f1 - existing.f1,
               at: new Date().toISOString(),
             };
             return accEvent;
           });
           // W-2: emit OUTSIDE the transaction.
           if (event) {
             this.eventEmitter.emit(ACCURACY_CHANGED_TOPIC, event);
           }
         }

         /** Plan 05 controller consumer; shape matches D-09. Implementation body inline below. */
         async read(tenantId: string, options: ReadOptions = {}): Promise<ReadResult> {
           const windows: AccuracyWindowKey[] = options.windows ?? ['7d', '30d', 'all'];
           const includeBreakdown = options.includeBreakdown === true;

           const profiles = await this.prisma.labelAccuracyProfile.findMany({
             where: { tenantId, windowKey: { in: windows } },
           });
           const breakdowns = includeBreakdown
             ? await this.prisma.labelAccuracySourceBreakdown.findMany({
                 where: { tenantId, windowKey: { in: windows } },
               })
             : [];

           const labelIds = Array.from(
             new Set([
               ...profiles.map((p) => p.labelId),
               ...breakdowns.map((b) => b.labelId),
             ]),
           );
           const labels = labelIds.length
             ? await this.prisma.userLabel.findMany({
                 where: { tenantId, id: { in: labelIds } },
                 select: { id: true, name: true, slug: true },
               })
             : [];
           const labelMap = new Map(
             labels.map((l) => [l.id, { name: l.name, slug: l.slug ?? null }]),
           );

           const byLabel = new Map<
             string,
             {
               windows: Partial<Record<AccuracyWindowKey, ReadWindowCounters>>;
               breakdown: Partial<
                 Record<
                   AccuracyTriageSource,
                   Pick<ReadWindowCounters, 'precision' | 'recall' | 'f1' | 'sampleSize'> | null
                 >
               >;
               lastUpdatedAt: string | null;
             }
           >();

           for (const p of profiles) {
             let bucket = byLabel.get(p.labelId);
             if (!bucket) {
               bucket = { windows: {}, breakdown: {}, lastUpdatedAt: null };
               byLabel.set(p.labelId, bucket);
             }
             const counters: ReadWindowCounters = {
               precision: p.precision,
               recall: p.recall,
               f1: p.f1,
               tp: p.tp,
               fp: p.fp,
               fn: p.fn,
               sampleSize: p.sampleSize,
               wilsonPrecLow: p.wilsonPrecLow,
               wilsonPrecHigh: p.wilsonPrecHigh,
               wilsonRecLow: p.wilsonRecLow,
               wilsonRecHigh: p.wilsonRecHigh,
               state: p.sampleSize < LEARNING_THRESHOLD ? 'learning' : 'ready',
             };
             bucket.windows[p.windowKey as AccuracyWindowKey] = counters;
             if (
               bucket.lastUpdatedAt == null ||
               new Date(p.lastUpdatedAt) > new Date(bucket.lastUpdatedAt)
             ) {
               bucket.lastUpdatedAt = p.lastUpdatedAt.toISOString();
             }
           }

           if (includeBreakdown) {
             for (const b of breakdowns) {
               let bucket = byLabel.get(b.labelId);
               if (!bucket) {
                 bucket = { windows: {}, breakdown: {}, lastUpdatedAt: null };
                 byLabel.set(b.labelId, bucket);
               }
               const ts = b.triageSource as AccuracyTriageSource;
               bucket.breakdown[ts] =
                 b.sampleSize === 0
                   ? null
                   : {
                       precision: b.precision,
                       recall: b.recall,
                       f1: b.f1,
                       sampleSize: b.sampleSize,
                     };
             }
           }

           return {
             labels: Array.from(byLabel.entries()).map(([labelId, v]) => ({
               labelId,
               labelName: labelMap.get(labelId)?.name ?? null,
               labelSlug: labelMap.get(labelId)?.slug ?? null,
               windows: v.windows,
               triageSourceBreakdown: includeBreakdown ? v.breakdown : undefined,
               lastUpdatedAt: v.lastUpdatedAt,
             })),
             computedAt: new Date().toISOString(),
           };
         }

         /** D-06 cron handler. Fans out per-tenant with lease + watermark. */
         async runSweep(): Promise<void> {
           const lookback = new Date(Date.now() - 24 * 3600 * 1000);
           const rows = await this.prisma.emailTriageFeedback.findMany({
             where: { createdAt: { gte: lookback } },
             distinct: ['tenantId'],
             select: { tenantId: true },
           });
           this.logger.log(
             `accuracy-sweep fan-out: ${rows.length} tenant(s) with recent feedback`,
           );
           const redis = this.getRedis();
           for (const { tenantId } of rows) {
             await new Promise((r) =>
               setTimeout(r, Math.floor(Math.random() * 30_000)),
             );
             if (redis) {
               const acquired = await redis.set(
                 LEASE_KEY(tenantId),
                 String(Date.now()),
                 'EX',
                 LEASE_TTL_SECONDS,
                 'NX',
               );
               if (acquired !== 'OK') {
                 this.logger.debug(
                   `accuracy-sweep: lease held elsewhere for tenant=${tenantId}, skipping`,
                 );
                 continue;
               }
             }
             try {
               const tp = await this.sweepTps(tenantId);
               const dec = await this.rolloverWindows(tenantId);
               if (redis) {
                 await redis.set(WATERMARK_KEY(tenantId), new Date().toISOString());
               }
               this.logger.log(
                 `accuracy-sweep tenant=${tenantId}: credited=${tp.credited} decremented=${dec.decremented}`,
               );
             } catch (err) {
               this.logger.error(
                 `accuracy-sweep failed tenant=${tenantId}: ${String(err)}`,
               );
             }
           }
           // Plan 04 appends gauge-refresh block here. Plan 03 safeSet is a no-op.
         }
       }

         Logger,
         OnApplicationBootstrap,
         OnModuleDestroy,
         OnModuleInit,
       } from '@nestjs/common';

         tenantId: string;
         feedbackId: string;
         labelId: string;
         kind: LabelAccuracyKind;
         delta: number;
         triageSource?: AccuracyTriageSource;
         signal?: 'edit' | 'correction' | 'system';

             defaultJobOptions: {
               ...getDefaultJobOptions('NORMAL'),
               attempts: 3,
               backoff: { type: 'exponential', delay: 5_000 },
             },
           });
           this.logger.log('LabelAccuracyQueue initialized');

           const jobsEnabled =
             (process.env.JOBS_ENABLED || 'true').toLowerCase() !== 'false';

               concurrency: 2,
               lockDuration: 30_000,
             },

             this.logger.log('LabelAccuracy worker READY (concurrency: 2, dedup 5min)');
           });

           await this.queue?.close();
           if (this.redisConnection) {
             releaseSharedRedisConnection();
             this.redisConnection = null;
           }
           if (this.workerConnection) {
             releaseSharedWorkerConnection();
             this.workerConnection = null;
           }

       /**
        * Phase 148 D-05 listener. Subscribes to `'triage.accuracy.applied'` events
        * emitted by Phase-147 TriageFeedbackService (wired in Plan 04). Translates
        * each event into one or two enqueues on the accuracy-update queue.
        *
        * Label-name → labelId resolution uses UserLabel.slug (preferred) then
        * case-insensitive UserLabel.name fallback (for labels created pre-slug).
        *
        * B-7 hardening: containsSeparator() is invoked on EVERY dedup-key component
        * including the post-resolution labelId on each of the 2 enqueue branches.
        */
       import { Injectable, Logger } from '@nestjs/common';
       import { OnEvent } from '@nestjs/event-emitter';
       import { PrismaService } from '../../../prisma/prisma.service';
       import { LabelAccuracyQueueService } from '../services/label-accuracy.queue';
       import {
         ACCURACY_APPLIED_TOPIC,
         type LabelAccuracyEvent,
       } from '../services/label-accuracy.events';

       @Injectable()
       export class LabelAccuracyListener {
         private readonly logger = new Logger(LabelAccuracyListener.name);

         constructor(
           private readonly prisma: PrismaService,
           private readonly queue: LabelAccuracyQueueService,
         ) {}

         @OnEvent(ACCURACY_APPLIED_TOPIC, { async: true })
         async handleLabelAccuracyEvent(event: LabelAccuracyEvent): Promise<void> {
           try {
             // B-7: pre-resolution component checks.
             if (
               this.containsSeparator(event.tenantId) ||
               this.containsSeparator(event.feedbackId) ||
               this.containsSeparator(event.emailId)
             ) {
               this.logger.warn(
                 `Rejecting malformed accuracy event: component contains ':' — tenantId=${event.tenantId} feedbackId=${event.feedbackId}`,
               );
               return;
             }

             const names = [event.oldLabelName, event.newLabelName].filter(
               (n): n is string => typeof n === 'string' && n.length > 0,
             );
             if (names.length === 0) return;

             const labels = await this.prisma.userLabel.findMany({
               where: {
                 tenantId: event.tenantId,
                 OR: [
                   { slug: { in: names } },
                   { name: { in: names, mode: 'insensitive' } },
                 ],
               },
               select: { id: true, name: true, slug: true },
             });
             const bySlug = new Map(labels.map((l) => [l.slug, l.id]));
             const byName = new Map(labels.map((l) => [l.name.toLowerCase(), l.id]));

             // Old label → FP+1 (AI label was removed by the user)
             if (event.oldLabelName) {
               const labelId =
                 bySlug.get(event.oldLabelName) ??
                 byName.get(event.oldLabelName.toLowerCase());
               // B-7: per-branch post-resolution labelId guard.
               if (labelId && !this.containsSeparator(labelId)) {
                 await this.queue.enqueueAccuracyUpdate({
                   tenantId: event.tenantId,
                   feedbackId: event.feedbackId,
                   labelId,
                   kind: 'fp',
                   delta: 1,
                   triageSource: event.triageSource,
                   signal: event.signal,
                 });
               }
             }
             // New label → FN+1 (AI missed this label; user added it)
             if (event.newLabelName) {
               const labelId =
                 bySlug.get(event.newLabelName) ??
                 byName.get(event.newLabelName.toLowerCase());
               // B-7: per-branch post-resolution labelId guard.
               if (labelId && !this.containsSeparator(labelId)) {
                 await this.queue.enqueueAccuracyUpdate({
                   tenantId: event.tenantId,
                   feedbackId: event.feedbackId,
                   labelId,
                   kind: 'fn',
                   delta: 1,
                   triageSource: event.triageSource,
                   signal: event.signal,
                 });
               }
             }
           } catch (err) {
             this.logger.warn(
               `Accuracy listener failed: ${err instanceof Error ? err.message : String(err)}`,
             );
           }
         }

         private containsSeparator(s: string): boolean {
           return typeof s === 'string' && s.includes(':');
         }
       }

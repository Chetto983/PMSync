       ```typescript
       /**
        * Phase 148 D-09 frontend API module for triage accuracy read endpoint.
        */
       import { apiClient } from '../api-client';

       export type AccuracyWindowKey = '7d' | '30d' | 'all';
       export type AccuracyTriageSource = 'llm' | 'rule' | 'llm+rule';

       export interface LabelAccuracyWindow {
         precision: number | null;
         recall: number | null;
         f1: number | null;
         tp: number;
         fp: number;
         fn: number;
         sampleSize: number;
         wilsonPrecLow: number | null;
         wilsonPrecHigh: number | null;
         wilsonRecLow: number | null;
         wilsonRecHigh: number | null;
         state: 'learning' | 'ready';
       }

       export interface LabelAccuracyEntry {
         labelId: string;
         labelName: string | null;
         labelSlug: string | null;
         windows: Partial<Record<AccuracyWindowKey, LabelAccuracyWindow>>;
         triageSourceBreakdown?: Partial<
           Record<
             AccuracyTriageSource,
             Pick<LabelAccuracyWindow, 'precision' | 'recall' | 'f1' | 'sampleSize'> | null
           >
         >;
         lastUpdatedAt: string | null;
       }

       export interface LabelAccuracyResponse {
         labels: LabelAccuracyEntry[];
         computedAt: string;
       }

       export interface GetLabelAccuracyParams {
         windows?: AccuracyWindowKey[];
         includeBreakdown?: boolean;
       }

       export const triageAccuracyApi = {
         async getLabelAccuracy(
           params?: GetLabelAccuracyParams,
         ): Promise<LabelAccuracyResponse> {
           const query: Record<string, string> = {};
           if (params?.windows && params.windows.length > 0) {
             query.windows = params.windows.join(',');
           }
           if (params?.includeBreakdown) {
             query.includeBreakdown = 'true';
           }
           const response = await apiClient.get('/ai/triage/label-accuracy', {
             params: query,
           });
           return response.data;
         },
       };

       export function getLabelAccuracy(
         params?: GetLabelAccuracyParams,
       ): Promise<LabelAccuracyResponse> {
         return triageAccuracyApi.getLabelAccuracy(params);
       }
       ```

       ```typescript
       /**
        * Phase 148 D-10/D-11 TanStack Query hook for label accuracy.
        *
        * staleTime 5min, gcTime 30min, retry 1 per CLAUDE.md Frontend Procedure.
        */
       import { useQuery, queryOptions } from '@tanstack/react-query';
       import {
         triageAccuracyApi,
         type GetLabelAccuracyParams,
         type LabelAccuracyResponse,
       } from '@/lib/api/triage-accuracy';

       export const labelAccuracyKeys = {
         all: ['label-accuracy'] as const,
         byParams: (params?: GetLabelAccuracyParams) =>
           [...labelAccuracyKeys.all, params ?? {}] as const,
       } as const;

       export function labelAccuracyOptions(params?: GetLabelAccuracyParams) {
         return queryOptions({
           queryKey: labelAccuracyKeys.byParams(params),
           queryFn: async (): Promise<LabelAccuracyResponse> =>
             triageAccuracyApi.getLabelAccuracy(params),
           staleTime: 5 * 60 * 1000,
           gcTime: 30 * 60 * 1000,
           retry: 1,
         });
       }

       export function useLabelAccuracy(params?: GetLabelAccuracyParams) {
         return useQuery(labelAccuracyOptions(params));
       }
       ```

       ```tsx
       /**
        * Phase 148 D-10 / D-11 / D-12: traffic-light accuracy pill.
        *
        * Extracted as a standalone component so Phase 155 dashboard can reuse it
        * (chart axis tick, leaderboard cell). Props intentionally minimal.
        */
       import React from 'react';
       import { Chip, Tooltip } from '@mui/material';
       import { useLabelAccuracy } from '@/hooks/queries/use-label-accuracy';
       import {
         type AccuracyWindowKey,
         type LabelAccuracyWindow,
       } from '@/lib/api/triage-accuracy';
       import { useLocale } from '@/lib/hooks/use-locale';
       import { useTranslations } from '@/locales';

       // D-10 thresholds + D-11 cold-start threshold
       const F1_GREEN = 0.85;
       const F1_YELLOW = 0.6;
       const LEARNING_THRESHOLD = 20;
       const WINDOW_DAYS: Record<AccuracyWindowKey, number> = {
         '7d': 7,
         '30d': 30,
         all: 0, // "all" doesn't report a window-days figure; badge uses 30d by default
       };

       interface LabelAccuracyBadgeProps {
         labelId: string;
         /** Window used to compute the pill + tooltip content. Default '30d'. */
         window?: AccuracyWindowKey;
       }

       /** Format a template string like `"Precision: {value}%"` with a map of substitutions. */
       function fmt(template: string, values: Record<string, string | number>): string {
         return template.replace(/\{(\w+)\}/g, (_m, k) =>
           k in values ? String(values[k]) : `{${k}}`,
         );
       }

       export function LabelAccuracyBadge({
         labelId,
         window = '30d',
       }: LabelAccuracyBadgeProps): React.ReactElement | null {
         const { data, isLoading, isError } = useLabelAccuracy();
         const t = useTranslations();
         useLocale(); // consume for re-render on locale flip

         if (isLoading || isError || !data) return null;

         const entry = data.labels.find((l) => l.labelId === labelId);
         if (!entry) return null;

         const w: LabelAccuracyWindow | undefined = entry.windows[window];
         const learning = !w || w.sampleSize < LEARNING_THRESHOLD || w.state === 'learning';

         if (learning) {
           const current = w?.sampleSize ?? 0;
           const tooltipContent = fmt(t.email.labels.accuracy.tooltip.learning, {
             current,
             threshold: LEARNING_THRESHOLD,
           });
           return (
             <Tooltip title={tooltipContent} arrow enterDelay={400}>
               <Chip
                 size="small"
                 variant="outlined"
                 label={t.email.labels.accuracy.pill.learning}
                 sx={{
                   color: 'text.secondary',
                   borderColor: 'divider',
                   height: 20,
                   fontSize: '0.6875rem',
                   ml: 0.5,
                   flexShrink: 0,
                 }}
               />
             </Tooltip>
           );
         }

         const f1 = w.f1 ?? 0;
         const color =
           f1 >= F1_GREEN
             ? 'success.main'
             : f1 >= F1_YELLOW
               ? 'warning.main'
               : 'error.main';

         const precisionPct = Math.round((w.precision ?? 0) * 100);
         const recallPct = Math.round((w.recall ?? 0) * 100);
         const precisionCi = Math.round(
           (((w.wilsonPrecHigh ?? 0) - (w.wilsonPrecLow ?? 0)) / 2) * 100,
         );
         const recallCi = Math.round(
           (((w.wilsonRecHigh ?? 0) - (w.wilsonRecLow ?? 0)) / 2) * 100,
         );
         const windowDays = WINDOW_DAYS[window] || 30;

         const tooltipContent = (
           <>
             <div>
               {fmt(t.email.labels.accuracy.tooltip.precision, {
                 value: precisionPct,
                 ci: precisionCi,
               })}
             </div>
             <div>
               {fmt(t.email.labels.accuracy.tooltip.recall, {
                 value: recallPct,
                 ci: recallCi,
               })}
             </div>
             <div>
               {fmt(t.email.labels.accuracy.tooltip.sampleSize, {
                 count: w.sampleSize,
                 days: windowDays,
               })}
             </div>
           </>
         );

         const percent = Math.round(f1 * 100);
         const label = fmt(t.email.labels.accuracy.pill.ready, { percent });

         return (
           <Tooltip title={tooltipContent} arrow enterDelay={400}>
             <Chip
               size="small"
               variant="outlined"
               label={label}
               sx={{
                 color,
                 borderColor: color,
                 height: 20,
                 fontSize: '0.6875rem',
                 fontWeight: 600,
                 ml: 0.5,
                 flexShrink: 0,
               }}
             />
           </Tooltip>
         );
       }

       export default LabelAccuracyBadge;
       ```

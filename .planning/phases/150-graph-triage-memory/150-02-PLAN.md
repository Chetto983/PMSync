**Entry point signature + body (exact — includes B-3 label resolution):**
```typescript
export async function processAffinityPromotionForTenant(
  tenantId: string,
  deps: AffinityPromotionDeps,
  jobId: string,
  jobTimestamp: Date,
): Promise<AffinityPromotionTenantResult> {
  const { prisma, promotionCounter, activeEdgesGauge, halfLifeDays } = deps;
  let sendersPromoted = 0, sendersUpdated = 0, domainsPromoted = 0, domainsUpdated = 0;

  // Step 1: Fetch TriageOverride rows for this tenant of type 'sender'.
  const overrides = await prisma.triageOverride.findMany({
    where: { tenantId, type: 'sender' },
  });
  if (overrides.length === 0) {
    return { sendersPromoted: 0, sendersUpdated: 0, domainsPromoted: 0, domainsUpdated: 0 };
  }

  // Step 2: Fetch reinforcing feedback rows for this tenant.
  const feedbackRows = await prisma.emailTriageFeedback.findMany({
    where: { tenantId },
  });

  // Step 3: Join emails to resolve sender addresses.
  const emailIds = feedbackRows.map((f: any) => f.emailId).filter(Boolean);
  const emails = emailIds.length > 0
    ? await prisma.email.findMany({ where: { tenantId, id: { in: emailIds } } })
    : [];
  const emailFromById = new Map(emails.map((e: any) => [e.id, (e.from ?? '').trim().toLowerCase()]));

    const key = `${sender}|${fb.newValue}`;
    const ageDays = Math.max(0, (now - new Date(fb.createdAt).getTime()) / 86_400_000);
    const w = weightCorrection(ageDays, { halfLifeDays });
    if (!senderAgg.has(key)) {
      senderAgg.set(key, { count: 0, weightedSum: 0, overrideIds: new Set(), senderEmail: sender, labelRaw: fb.newValue });
    }
    const entry = senderAgg.get(key)!;
    entry.count += 1;
    entry.weightedSum += w;
    entry.overrideIds.add(matchingOverride.id);

    if (entry.weightedSum < AFFINITY_MIN_CONFIDENCE) continue;

    // B-3: resolve fb.newValue to canonical Label.id. newValue may be stored
    // as either Label.id or Label.name depending on the callsite that wrote
    // the feedback row — tolerate both. Skip if no matching (non-archived) label.
    const label = await prisma.userLabel.findFirst({
      where: {
        tenantId,
        OR: [{ id: entry.labelRaw }, { name: entry.labelRaw }],
        isArchived: false,
      },
      select: { id: true, name: true },
    });
    if (!label) continue; // orphan newValue — skip silently
    const labelId = label.id;
    const labelName = label.name;

    const senderNode = await prisma.graphNode.upsert({
      where: { tenantId_nodeType_externalId: { tenantId, nodeType: 'sender', externalId: entry.senderEmail } },
      update: { label: entry.senderEmail },
      create: { tenantId, nodeType: 'sender', externalId: entry.senderEmail, label: entry.senderEmail },
    });

    // Use the SAME externalId here so node families align.

      where: { tenantId_nodeType_externalId: { tenantId, nodeType: 'label', externalId: labelId } },
      update: { label: labelName },
      create: { tenantId, nodeType: 'label', externalId: labelId, label: labelName },
    });

    const existing = await prisma.graphEdge.count({
      where: { tenantId, sourceId: senderNode.id, targetId: labelNode.id, edgeType: 'SENDER_LABEL_AFFINITY' },
    });
    const computedConfidence = Math.min(1, entry.weightedSum / Math.max(entry.count, 1));
    const derivedFromOverrideIds = Array.from(entry.overrideIds);
    const metadata = {
      derivedFromOverrideIds,
      promotionAudit: {
        jobId,
        jobTimestamp: jobTimestamp.toISOString(),
        computedConfidence,
        feedbackRowCount: entry.count,
        halfLifeDays,
      },
    };

    await prisma.graphEdge.upsert({
      where: {
        tenantId_sourceId_targetId_edgeType: {
          tenantId,
          sourceId: senderNode.id,
          targetId: labelNode.id,
          edgeType: 'SENDER_LABEL_AFFINITY',
        },
      },
      update: { confidence: computedConfidence, validFrom: jobTimestamp, metadata },
      create: {
        tenantId,
        sourceId: senderNode.id,
        targetId: labelNode.id,
        edgeType: 'SENDER_LABEL_AFFINITY',
        confidence: computedConfidence,
        validFrom: jobTimestamp,
        metadata,
      },
    });

    const action: 'created' | 'updated' = existing > 0 ? 'updated' : 'created';
    if (action === 'created') sendersPromoted++; else sendersUpdated++;
    promotionCounter?.inc({ tenant_id: tenantId, edge_type: 'SENDER_LABEL_AFFINITY', action });
  }

  // Step 6: Domain aggregation. Inherit Phase 147 D-04: >= 3 distinct non-role senders per (domain, labelId).
  const domainAgg = new Map<string, { count: number; weightedSum: number; overrideIds: Set<string>; distinctSenders: Set<string>; distinctNonRoleSenders: Set<string>; domain: string; labelRaw: string }>();

    const domain = extractDomain(entry.senderEmail);
    if (!domain) continue;
    const localPart = entry.senderEmail.split('@')[0];
    const isRole = ROLE_ACCOUNT_LOCAL_PARTS.has(localPart);
    const dKey = `${domain}|${entry.labelRaw}`;
    if (!domainAgg.has(dKey)) {
      domainAgg.set(dKey, { count: 0, weightedSum: 0, overrideIds: new Set(), distinctSenders: new Set(), distinctNonRoleSenders: new Set(), domain, labelRaw: entry.labelRaw });
    }
    const d = domainAgg.get(dKey)!;
    d.count += entry.count;
    d.weightedSum += entry.weightedSum;
    for (const oid of entry.overrideIds) d.overrideIds.add(oid);
    d.distinctSenders.add(entry.senderEmail);
    if (!isRole) d.distinctNonRoleSenders.add(entry.senderEmail);
  }

  for (const [_key, d] of domainAgg.entries()) {
    if (d.distinctNonRoleSenders.size < MIN_DISTINCT_SENDERS_FOR_DOMAIN) continue;
    if (d.count < AFFINITY_MIN_FEEDBACK_ROWS) continue;
    if (d.weightedSum < AFFINITY_MIN_CONFIDENCE) continue;

    // B-3: domain aggregation also resolves labelRaw -> Label.id
    const label = await prisma.userLabel.findFirst({
      where: {
        tenantId,
        OR: [{ id: d.labelRaw }, { name: d.labelRaw }],
        isArchived: false,
      },
      select: { id: true, name: true },
    });
    if (!label) continue;
    const labelId = label.id;
    const labelName = label.name;

    const domainNode = await prisma.graphNode.upsert({
      where: { tenantId_nodeType_externalId: { tenantId, nodeType: 'email_domain', externalId: d.domain } },
      update: { label: d.domain },
      create: { tenantId, nodeType: 'email_domain', externalId: d.domain, label: d.domain },
    });
    const labelNode = await prisma.graphNode.upsert({
      where: { tenantId_nodeType_externalId: { tenantId, nodeType: 'label', externalId: labelId } },
      update: { label: labelName },
      create: { tenantId, nodeType: 'label', externalId: labelId, label: labelName },
    });

    const existing = await prisma.graphEdge.count({
      where: { tenantId, sourceId: domainNode.id, targetId: labelNode.id, edgeType: 'DOMAIN_LABEL_AFFINITY' },
    });
    const computedConfidence = Math.min(1, d.weightedSum / Math.max(d.count, 1));
    const derivedFromOverrideIds = Array.from(d.overrideIds);
    const metadata = {
      derivedFromOverrideIds,
      promotionAudit: {
        jobId,
        jobTimestamp: jobTimestamp.toISOString(),
        computedConfidence,
        feedbackRowCount: d.count,
        halfLifeDays,
      },
    };
    await prisma.graphEdge.upsert({
      where: {
        tenantId_sourceId_targetId_edgeType: {
          tenantId,
          sourceId: domainNode.id,
          targetId: labelNode.id,
          edgeType: 'DOMAIN_LABEL_AFFINITY',
        },
      },
      update: { confidence: computedConfidence, validFrom: jobTimestamp, metadata },
      create: {
        tenantId, sourceId: domainNode.id, targetId: labelNode.id,
        edgeType: 'DOMAIN_LABEL_AFFINITY',
        confidence: computedConfidence, validFrom: jobTimestamp, metadata,
      },
    });
    const action: 'created' | 'updated' = existing > 0 ? 'updated' : 'created';
    if (action === 'created') domainsPromoted++; else domainsUpdated++;
    promotionCounter?.inc({ tenant_id: tenantId, edge_type: 'DOMAIN_LABEL_AFFINITY', action });
  }

  // Step 7: Sample active-edges gauge at end of run.
  if (activeEdgesGauge) {
    for (const edgeType of ['SENDER_LABEL_AFFINITY', 'DOMAIN_LABEL_AFFINITY'] as const) {
      const active = await prisma.graphEdge.count({
        where: { tenantId, edgeType, OR: [{ validUntil: null }, { validUntil: { gt: jobTimestamp } }] },
      });
      activeEdgesGauge.set({ tenant_id: tenantId, edge_type: edgeType }, active);
    }
  }

  return { sendersPromoted, sendersUpdated, domainsPromoted, domainsUpdated };
}
```

```typescript
  const activeTenants = await this.findActiveTenants(job.data.tenantId);
```

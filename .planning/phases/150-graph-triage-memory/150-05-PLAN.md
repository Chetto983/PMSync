```typescript
// Phase 150 D-04 + Landmine §1 Option B: GraphNode lacks @@unique, so
// concurrent create races must be serialized with SELECT FOR UPDATE.
private async upsertLabelNode(
  tenantId: string,
  labelId: string,
  labelName: string,
): Promise<void> {
  await this.prisma.$transaction(async (tx) => {
    // Acquire row-level lock on any existing matching node to prevent concurrent create race.
    const existing = await tx.$queryRaw<Array<{ id: string }>>`
      SELECT id FROM graph_nodes
      WHERE "tenantId" = ${tenantId}
        AND "nodeType" = 'label'
        AND "externalId" = ${labelId}
      FOR UPDATE
      LIMIT 1
    `;
    if (existing.length > 0) {
      await tx.graphNode.update({
        where: { id: existing[0].id },
        data: { label: labelName, updatedAt: new Date() },
      });
    } else {
      await tx.graphNode.create({
        data: {
          tenantId,
          nodeType: 'label',
          externalId: labelId,
          label: labelName,
        },
      });
    }
  });
}
```

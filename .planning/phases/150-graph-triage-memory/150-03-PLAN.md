```typescript
/**
 * Phase 150 D-06: third decay-gc store. Sweeps affinity edges whose
 * recomputed weighted-sum confidence drops below AFFINITY_EDGE_DECAY_CUTOFF.
 *
 * Called AFTER processDecayGcForTenant's Redis + TriageOverride sweeps so
 * recomputed override weights drive affinity-edge confidence recomputation
 * (Phase 149 D-09 ordering invariant extended).
 *
 * Defensive semantics:
 *   - Edge with empty/missing `metadata.derivedFromOverrideIds` -> treat as
 *     stale (no traceable evidence) -> delete. This handles legacy/corrupt
 *     rows from future phases that bypass the Phase 150 D-01a provenance
 *     contract.
 *   - Edge whose overrides all deleted (Phase 149 TriageOverride sweep ran
 *     first) -> recompute falls through to the feedback-row path using the
 *     cached feedback rows; if none remain, weightedSum = 0 -> delete.
 *
 * Multi-tenant invariant: every Prisma call includes `tenantId` in `where`.
 */
export async function processAffinityEdgeDecayForTenant(
  tenantId: string,
  deps: DecayGcDeps,
): Promise<GraphEdgeDecayResult> {
  const { prisma, gcCounter, halfLifeDays } = deps;
  const now = new Date();
  let deleted = 0;

  const edges = await prisma.graphEdge.findMany({
    where: {
      tenantId,
      edgeType: { in: ['SENDER_LABEL_AFFINITY', 'DOMAIN_LABEL_AFFINITY'] },
      OR: [{ validUntil: null }, { validUntil: { gt: now } }],
    },
  });

  if (edges.length === 0) {
    return { graphEdgeDeleted: 0 };
  }

  // Collect all derivedFromOverrideIds across edges to batch feedback fetch.
  const allOverrideIds = new Set<string>();
  for (const e of edges) {
    const ids = ((e.metadata as any)?.derivedFromOverrideIds ?? []) as string[];
    for (const id of ids) allOverrideIds.add(id);
  }

  // Fetch feedback rows for those overrides, tenant-scoped.
  const feedbackByOverrideId = new Map<string, Array<{ createdAt: Date }>>();
  if (allOverrideIds.size > 0) {
    const feedback = await prisma.emailTriageFeedback.findMany({
      where: { tenantId, overrideId: { in: Array.from(allOverrideIds) } },
    });
    for (const fb of feedback) {
      const oid = (fb as any).overrideId;
      if (!oid) continue;
      if (!feedbackByOverrideId.has(oid)) feedbackByOverrideId.set(oid, []);
      feedbackByOverrideId.get(oid)!.push({ createdAt: fb.createdAt });
    }
  }

  const nowMs = now.getTime();
  for (const edge of edges) {
    const overrideIds = ((edge.metadata as any)?.derivedFromOverrideIds ?? []) as string[];
    let weightedSum = 0;
    for (const oid of overrideIds) {
      const rows = feedbackByOverrideId.get(oid) ?? [];
      for (const r of rows) {
        const ageDays = Math.max(0, (nowMs - r.createdAt.getTime()) / 86_400_000);
        weightedSum += weightCorrection(ageDays, { halfLifeDays });
      }
    }

    if (weightedSum < AFFINITY_EDGE_DECAY_CUTOFF) {
      await prisma.graphEdge.deleteMany({
        where: { tenantId, id: edge.id },
      });
      gcCounter?.inc({ store: 'graph_edge', tenant_id: tenantId });
      deleted += 1;
    }
  }

  return { graphEdgeDeleted: deleted };
}
```

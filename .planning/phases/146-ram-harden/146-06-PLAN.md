```typescript
/**
 * D-15: Check Redis maxmemory-policy is set to 'noeviction'.
 * BullMQ requires noeviction — other policies cause silent queue corruption.
 * Logs warning only, does not block startup (Railway may restrict CONFIG GET).
 */
async function checkRedisMaxMemoryPolicy(redis: Redis): Promise<void> {
  try {
    const config = await redis.config('GET', 'maxmemory-policy');
    // config returns ['maxmemory-policy', 'noeviction'] or similar
    const policy = Array.isArray(config) ? config[1] : null;
    if (policy && policy !== 'noeviction') {
      Logger.warn(
        `Redis maxmemory-policy is "${policy}" — BullMQ requires "noeviction". ` +
        `Queues may silently corrupt if Redis evicts keys.`,
        'Bootstrap',
      );
    } else if (policy === 'noeviction') {
      Logger.log('Redis maxmemory-policy: noeviction (correct)', 'Bootstrap');
    }
  } catch (error) {
    // Railway managed Redis may restrict CONFIG command
    Logger.warn(
      `Could not check Redis maxmemory-policy (CONFIG GET may be restricted): ${error instanceof Error ? error.message : String(error)}`,
      'Bootstrap',
    );
  }
}
```

```typescript
import { QueueHealthService } from '../queue-health.service';

// Mock the config module
jest.mock('../../config', () => ({
  getSharedRedisConnection: jest.fn(),
  releaseSharedRedisConnection: jest.fn(),
  QUEUE_NAMES: {
    EMAIL_SYNC_HIGH: 'email-sync-high',
    EMAIL_SYNC_NORMAL: 'email-sync-normal',
    DEAD_LETTER: 'dead-letter-queue',
  },
}));

import {
  getSharedRedisConnection,
  releaseSharedRedisConnection,
} from '../../config';

describe('QueueHealthService (Direct Redis)', () => {
  let service: QueueHealthService;
  let mockRedis: any;
  let mockPipeline: any;

  beforeEach(async () => {
    // Setup mock Redis pipeline
    mockPipeline = {
      llen: jest.fn().mockReturnThis(),
      zcard: jest.fn().mockReturnThis(),
      hget: jest.fn().mockReturnThis(),
      exec: jest.fn(),
    };

    mockRedis = {
      pipeline: jest.fn(() => mockPipeline),
      ping: jest.fn().mockResolvedValue('PONG'),
      info: jest.fn().mockResolvedValue('used_memory:1048576\nused_memory_peak:2097152'),
      status: 'ready',
    };

    (getSharedRedisConnection as jest.Mock).mockReturnValue(mockRedis);

    service = new QueueHealthService();
    await service.onModuleInit();
  });

  afterEach(async () => {
    await service.onModuleDestroy();
  });

  it('should initialize with queue names (not Queue objects)', () => {
    // Verify no Queue constructor was called
    expect((service as any).queueNames).toEqual([
      'email-sync-high',
      'email-sync-normal',
      'dead-letter-queue',
    ]);
  });

  it('should get queue status via Redis pipeline', async () => {
    // Mock pipeline results: [error, value] pairs
    mockPipeline.exec.mockResolvedValue([
      [null, 5],    // wait (LLEN)
      [null, 2],    // active (LLEN)
      [null, 100],  // completed (ZCARD)
      [null, 3],    // failed (ZCARD)
      [null, 10],   // delayed (ZCARD)
      [null, null],  // meta paused (HGET) — not paused
    ]);

    const status = await service.getQueueStatus('email-sync-high');

    expect(status).toEqual({
      name: 'email-sync-high',
      waiting: 5,
      active: 2,
      completed: 100,
      failed: 3,
      delayed: 10,
      paused: false,
    });

    expect(mockPipeline.llen).toHaveBeenCalledWith('bull:email-sync-high:wait');
    expect(mockPipeline.llen).toHaveBeenCalledWith('bull:email-sync-high:active');
    expect(mockPipeline.zcard).toHaveBeenCalledWith('bull:email-sync-high:completed');
    expect(mockPipeline.zcard).toHaveBeenCalledWith('bull:email-sync-high:failed');
    expect(mockPipeline.zcard).toHaveBeenCalledWith('bull:email-sync-high:delayed');
    expect(mockPipeline.hget).toHaveBeenCalledWith('bull:email-sync-high:meta', 'paused');
  });

  it('should detect paused queue via integer value', async () => {
    mockPipeline.exec.mockResolvedValue([
      [null, 0], [null, 0], [null, 0], [null, 0], [null, 0],
      [null, 1],  // paused = 1 (integer)
    ]);

    const status = await service.getQueueStatus('email-sync-high');
    expect(status?.paused).toBe(true);
  });

  it('should detect paused queue via string value', async () => {
    mockPipeline.exec.mockResolvedValue([
      [null, 0], [null, 0], [null, 0], [null, 0], [null, 0],
      [null, '1'],  // paused = '1' (string — defensive guard)
    ]);

    const status = await service.getQueueStatus('email-sync-high');
    expect(status?.paused).toBe(true);
  });

  it('should return null for unknown queue', async () => {
    const status = await service.getQueueStatus('nonexistent-queue');
    expect(status).toBeNull();
  });

  it('should handle Redis pipeline errors gracefully', async () => {
    mockPipeline.exec.mockRejectedValue(new Error('Redis timeout'));

    const status = await service.getQueueStatus('email-sync-high');
    expect(status).toBeNull();
  });

  it('should get all queue statuses in parallel', async () => {
    mockPipeline.exec.mockResolvedValue([
      [null, 1], [null, 0], [null, 0], [null, 0], [null, 0], [null, null],
    ]);

    const statuses = await service.getQueueStatuses();
    expect(statuses).toHaveLength(3); // 3 queues in mock QUEUE_NAMES
    expect(mockRedis.pipeline).toHaveBeenCalledTimes(3);
  });

  it('should release shared connection on destroy', async () => {
    await service.onModuleDestroy();
    expect(releaseSharedRedisConnection).toHaveBeenCalled();
  });
});
```

```typescript
import { Injectable, Logger } from '@nestjs/common';
import { OnEvent } from '@nestjs/event-emitter';
import { Queue } from 'bullmq';
import { EmailIndexedEvent } from '../../common/events/email.events';
import { PrismaService } from '../../prisma/prisma.service';
import { PushService } from './push.service';
import {
  QUEUE_NAMES,
  getRedisConnectionOptions,
  getDefaultJobOptions,
} from '../../common/config';
import { withoutTenantScopingAsync } from '../../prisma/tenant-scoping/tenant-context';

export interface PushDispatchJob {
  tenantId: string;
  userId: string;
  emailId: string;
  subject: string;
  from: string;
  threadId: string | null;
  category: 'email' | 'calendar' | 'ai';
}

@Injectable()
export class PushDispatchHandler implements OnModuleInit, OnModuleDestroy {
  private readonly logger = new Logger(PushDispatchHandler.name);
  private queue!: Queue<PushDispatchJob>;

  constructor(
    private readonly prisma: PrismaService,
    private readonly pushService: PushService,
  ) {}

  async onModuleInit(): Promise<void> {
    const connection = getRedisConnectionOptions();
    this.queue = new Queue<PushDispatchJob>(QUEUE_NAMES.PUSH_DISPATCH, {
      connection,
      defaultJobOptions: getDefaultJobOptions('HIGH'),
    });
  }

  async onModuleDestroy(): Promise<void> {
    await this.queue?.close();
  }

  @OnEvent(EmailIndexedEvent.eventName)
  async handleEmailIndexed(event: EmailIndexedEvent): Promise<void> {
    try {
      // Look up userId from providerId (event doesn't carry userId)
      const result = await withoutTenantScopingAsync(() =>
        this.prisma.providerConfig.findUnique({
          where: { id: event.providerId },
          select: { userId: true },
        }),
      );

      if (!result?.userId) {
        this.logger.warn(`No userId for provider ${event.providerId}, skipping push`);
        return;
      }

      const userId = result.userId;

      // Check user push preferences
      const prefs = await withoutTenantScopingAsync(() =>
        this.pushService.getUserPushPreferences(event.tenantId, userId),
      );

      if (!prefs.pushEmailEnabled) {
        return;
      }

      // Look up threadId for grouping
      const email = await withoutTenantScopingAsync(() =>
        this.prisma.email.findFirst({
          where: { id: event.emailId, tenantId: event.tenantId },
          select: { threadId: true },
        }),
      );

      // Thread grouping via Redis TTL (PUSH-07)
      const threadId = email?.threadId;
      if (threadId) {
        const dedupKey = `push:thread:${event.tenantId}:${userId}:${threadId}`;
        const redis = this.queue.client;
        // Wait for client to be ready
        const client = await redis;
        const count = await client.incr(dedupKey);
        if (count === 1) {
          // First email in thread window -- set TTL and schedule delayed job
          await client.expire(dedupKey, 60);
          await this.queue.add('push-email-grouped', {
            tenantId: event.tenantId,
            userId,
            emailId: event.emailId,
            subject: event.subject,
            from: event.from,
            threadId,
            category: 'email',
          }, {
            delay: 60_000, // 60-second grouping window
            jobId: dedupKey, // Prevent duplicate delayed jobs
          });
        }
        // count > 1 means another email in same thread within window -- skip, grouped job will handle it
        return;
      }

      // No threadId -- send immediately
      await this.queue.add('push-email', {
        tenantId: event.tenantId,
        userId,
        emailId: event.emailId,
        subject: event.subject,
        from: event.from,
        threadId: null,
        category: 'email',
      });
    } catch (error) {
      const message = error instanceof Error ? error.message : String(error);
      this.logger.error(`Failed to enqueue push for email ${event.emailId}: ${message}`);
    }
  }
}
```

```typescript
import { Injectable, Logger, OnModuleInit, OnModuleDestroy } from '@nestjs/common';
import { Worker, Job } from 'bullmq';
import { PushService } from './push.service';
import { PrismaService } from '../../prisma/prisma.service';
import {
  QUEUE_NAMES,
  getRedisConnectionOptions,
} from '../../common/config';
import { withoutTenantScopingAsync } from '../../prisma/tenant-scoping/tenant-context';
import type { PushDispatchJob } from './push-dispatch.handler';

@Injectable()
export class PushDispatchWorker implements OnModuleInit, OnModuleDestroy {
  private readonly logger = new Logger(PushDispatchWorker.name);
  private worker!: Worker<PushDispatchJob>;

  constructor(
    private readonly pushService: PushService,
    private readonly prisma: PrismaService,
  ) {}

  async onModuleInit(): Promise<void> {
    const connection = getRedisConnectionOptions();
    this.worker = new Worker<PushDispatchJob>(
      QUEUE_NAMES.PUSH_DISPATCH,
      async (job: Job<PushDispatchJob>) => this.process(job),
      {
        connection,
        concurrency: 5,
        lockDuration: 30_000,
      },
    );

    this.worker.on('failed', (job, err) => {
      this.logger.error(`Push job ${job?.id} failed: ${err.message}`);
    });
  }

  async onModuleDestroy(): Promise<void> {
    await this.worker?.close();
  }

  private async process(job: Job<PushDispatchJob>): Promise<void> {
    const { tenantId, userId, emailId, subject, from, threadId, category } = job.data;

    // Get all subscriptions for this user (PUSH-04: multi-device)
    const subscriptions = await withoutTenantScopingAsync(() =>
      this.pushService.getSubscriptionsForUser(tenantId, userId),
    );

    if (subscriptions.length === 0) {
      this.logger.debug(`No push subscriptions for user ${userId}`);
      return;
    }

    // Build payload based on job name
    let payload: { title: string; body: string; icon: string; badge: string; tag?: string; data: { url: string } };

    if (job.name === 'push-email-grouped' && threadId) {
      // PUSH-07: Grouped notification -- count emails in thread window
      const redis = await this.worker.client;
      const dedupKey = `push:thread:${tenantId}:${userId}:${threadId}`;
      const countStr = await redis.get(dedupKey);
      const count = parseInt(countStr || '1', 10);

      // Truncate subject to 120 chars
      const truncatedSubject = subject.length > 120 ? subject.substring(0, 117) + '...' : subject;

      payload = {
        title: `${count} new emails in "${truncatedSubject}"`,
        body: `from ${from}`,
        icon: '/icons/icon-192x192.png',
        badge: '/icons/icon-72x72.png',
        tag: threadId,
        data: { url: `/mail/${emailId}` },
      };

      // Clean up Redis key after sending
      await redis.del(dedupKey);
    } else {
      // Single email notification
      const senderName = from.includes('<') ? from.split('<')[0].trim() : from;
      const truncatedSubject = subject.length > 120 ? subject.substring(0, 117) + '...' : subject;

      payload = {
        title: `New email from ${senderName}`,
        body: truncatedSubject,
        icon: '/icons/icon-192x192.png',
        badge: '/icons/icon-72x72.png',
        tag: threadId || emailId,
        data: { url: `/mail/${emailId}` },
      };
    }

    const payloadStr = JSON.stringify(payload);

    // Fan out to all subscriptions (PUSH-04: multi-device delivery)
    const results = await Promise.allSettled(
      subscriptions.map(async (sub) => {
        try {
          await this.pushService.sendNotification(
            { endpoint: sub.endpoint, p256dh: sub.p256dh, auth: sub.auth },
            payloadStr,
          );
        } catch (error: unknown) {
          const statusCode = (error as { statusCode?: number })?.statusCode;
          if (statusCode === 410 || statusCode === 404) {
            // PUSH-03: Stale subscription cleanup
            this.logger.log(`Deleting stale subscription ${sub.endpoint} (${statusCode})`);
            await withoutTenantScopingAsync(() =>
              this.pushService.deleteStaleSubscription(sub.id, tenantId),
            );
          } else {
            throw error; // Let BullMQ retry
          }
        }
      }),
    );

    const failed = results.filter((r) => r.status === 'rejected');
    if (failed.length > 0) {
      this.logger.warn(`${failed.length}/${subscriptions.length} push sends failed for user ${userId}`);
    }
  }
}
```

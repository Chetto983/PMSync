import { makeHistogramProvider, makeCounterProvider, makeGaugeProvider } from '@willsoto/nestjs-prometheus';

// Histogram buckets for latency (in seconds)
const LATENCY_BUCKETS = [0.05, 0.1, 0.25, 0.5, 1, 2.5, 5, 10, 30, 60];
const RAG_BUCKETS = [0.01, 0.025, 0.05, 0.1, 0.25, 0.5, 1, 2.5];

export const AI_METRICS = {
  // Chat response latency
  chatDuration: makeHistogramProvider({
    name: 'ai_chat_response_duration_seconds',
    help: 'AI chat response duration from request to complete response',
    labelNames: ['agent', 'model', 'status', 'intent'],
    buckets: LATENCY_BUCKETS,
  }),

  // First token latency (for streaming)
  firstTokenLatency: makeHistogramProvider({
    name: 'ai_first_token_latency_seconds',
    help: 'Time to first token in streaming response',
    labelNames: ['agent', 'model'],
    buckets: [0.1, 0.25, 0.5, 1, 2, 5],
  }),

  // RAG query latency
  ragDuration: makeHistogramProvider({
    name: 'ai_rag_query_duration_seconds',
    help: 'RAG retrieval query duration',
    labelNames: ['source', 'status'],
    buckets: RAG_BUCKETS,
  }),

  // Tool execution latency
  toolDuration: makeHistogramProvider({
    name: 'ai_tool_execution_duration_seconds',
    help: 'Tool execution duration by tool name',
    labelNames: ['tool', 'status'],
    buckets: LATENCY_BUCKETS,
  }),

  // Request counter
  requestsTotal: makeCounterProvider({
    name: 'ai_requests_total',
    help: 'Total AI requests by type',
    labelNames: ['type', 'status'],
  }),

  // Active conversations gauge
  activeConversations: makeGaugeProvider({
    name: 'ai_active_conversations',
    help: 'Number of active AI conversations',
    labelNames: ['tenant'],
  }),

  // Error counter
  errorsTotal: makeCounterProvider({
    name: 'ai_errors_total',
    help: 'Total AI errors by type',
    labelNames: ['operation', 'error_type'],
  }),
};

export const AI_METRICS_PROVIDERS = Object.values(AI_METRICS);

**Add Prometheus histogram injection:**
```typescript
import { InjectMetric } from '@willsoto/nestjs-prometheus';
import { Histogram, Counter, Gauge } from 'prom-client';

@Injectable()
export class AgentMetricsService {
  constructor(
    // Existing dependencies...

    @InjectMetric('ai_chat_response_duration_seconds')
    private readonly chatDuration: Histogram<string>,

    @InjectMetric('ai_first_token_latency_seconds')
    private readonly firstTokenLatency: Histogram<string>,

    @InjectMetric('ai_rag_query_duration_seconds')
    private readonly ragDuration: Histogram<string>,

    @InjectMetric('ai_tool_execution_duration_seconds')
    private readonly toolDuration: Histogram<string>,

    @InjectMetric('ai_requests_total')
    private readonly requestsTotal: Counter<string>,

    @InjectMetric('ai_errors_total')
    private readonly errorsTotal: Counter<string>,
  ) {}

  // Chat duration recording
  recordChatDuration(
    agent: string,
    model: string,
    status: 'success' | 'error',
    intent: string,
    durationSeconds: number,
  ): void {
    this.chatDuration.labels(agent, model, status, intent).observe(durationSeconds);
    this.requestsTotal.labels('chat', status).inc();
  }

  // First token latency (for streaming)
  recordFirstTokenLatency(agent: string, model: string, durationSeconds: number): void {
    this.firstTokenLatency.labels(agent, model).observe(durationSeconds);
  }

  // RAG query duration
  recordRagDuration(source: string, status: 'success' | 'error', durationSeconds: number): void {
    this.ragDuration.labels(source, status).observe(durationSeconds);
  }

  // Tool execution duration
  recordToolDuration(tool: string, status: 'success' | 'error', durationSeconds: number): void {
    this.toolDuration.labels(tool, status).observe(durationSeconds);
  }

  // Error recording
  recordError(operation: string, errorType: string): void {
    this.errorsTotal.labels(operation, errorType).inc();
    this.requestsTotal.labels(operation, 'error').inc();
  }

  // Timer helpers for easier usage
  startChatTimer(agent: string, model: string, intent: string) {
    const startTime = process.hrtime.bigint();
    return {
      end: (status: 'success' | 'error') => {
        const durationNs = Number(process.hrtime.bigint() - startTime);
        const durationSeconds = durationNs / 1e9;
        this.recordChatDuration(agent, model, status, intent, durationSeconds);
      },
    };
  }

  startRagTimer(source: string) {
    const startTime = process.hrtime.bigint();
    return {
      end: (status: 'success' | 'error') => {
        const durationNs = Number(process.hrtime.bigint() - startTime);
        const durationSeconds = durationNs / 1e9;
        this.recordRagDuration(source, status, durationSeconds);
      },
    };
  }
}
```

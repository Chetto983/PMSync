async subscribeToChanges(
  callback: ChangeCallback,
  folder: string,
  resolveFolderFn: (folder: string) => Promise<string>,
): Promise<ProviderSubscription> {
  if (!this.context?.imapConfig) {
    throw new Error('IDLE service not initialized');
  }

  await this.stopIdleSubscription();

  const folderPath = await resolveFolderFn(folder);
  this.idleFolder = folderPath;
  this.idleCallback = callback;
  this.idleShouldReconnect = true;
  this.idleReconnectAttempts = 0;

  await this.startIdleConnection();

  return {
    unsubscribe: async () => {
      await this.stopIdleSubscription();
    },
    isActive: () => {
      return this.idleActive && (this.idleConnection?.usable ?? false);
    },
  };
}

async sendMessage(message: ComposeMessage): Promise<string> {
  if (!this.context?.smtpConfig) {
    throw new Error('SMTP not configured');
  }

  const transporter = nodemailer.createTransport({
    host: this.context.smtpConfig.host,
    port: this.context.smtpConfig.port,
    secure: this.context.smtpConfig.secure,
    auth: this.context.smtpConfig.auth,
  });

  const result = await transporter.sendMail({
    from: message.fromEmail || this.context.userEmail,
    to: message.to.map((a) => this.formatEmailAddress(a)).join(', '),
    cc: message.cc?.map((a) => this.formatEmailAddress(a)).join(', '),
    bcc: message.bcc?.map((a) => this.formatEmailAddress(a)).join(', '),
    subject: message.subject,
    text: message.bodyText,
    html: message.bodyHtml,
    attachments: message.attachments?.map((att) => ({
      filename: att.filename,
      content: att.content ? Buffer.from(att.content, 'base64') : att.contentStream,
      contentType: att.mimeType,
    })),
    inReplyTo: message.inReplyTo,
    references: message.references,
  });

  // Save to Sent folder
  await this.saveSentMessage(message);

  return result.messageId || `sent-${Date.now()}`;
}

async sendDraft(
  draftId: string,
  getMessageFn: (id: string) => Promise<any>,
  deleteMessageFn: (id: string, permanent: boolean) => Promise<void>,
): Promise<void> {
  const draft = await getMessageFn(draftId);
  if (!draft) {
    throw new Error(`Draft not found: ${draftId}`);
  }

  const message: ComposeMessage = {
    to: draft.to,
    cc: draft.cc,
    bcc: draft.bcc,
    subject: draft.subject,
    bodyHtml: draft.bodyHtml || '',
    bodyText: draft.bodyText,
    inReplyTo: draft.inReplyTo,
    references: draft.references,
  };

  await this.sendMessage(message);
  await deleteMessageFn(draftId, true);
}

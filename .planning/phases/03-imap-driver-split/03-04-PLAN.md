import { Test, TestingModule } from '@nestjs/testing';
import { ImapMessageService } from '../services/imap-message.service';
import { ImapConnectionPoolService } from '../services/imap-connection-pool.service';
import { ImapFolderService } from '../services/imap-folder.service';

describe('ImapMessageService', () => {
  let service: ImapMessageService;
  let mockPool: jest.Mocked<ImapConnectionPoolService>;
  let mockFolderService: jest.Mocked<ImapFolderService>;

  const mockMessage = {
    uid: 123,
    flags: new Set(['\\Seen']),
    envelope: {
      date: new Date('2025-01-01'),
      subject: 'Test Subject',
      from: [{ address: 'sender@example.com', name: 'Sender' }],
      to: [{ address: 'recipient@example.com', name: 'Recipient' }],
      cc: [],
      bcc: [],
      messageId: '<test@example.com>',
    },
    bodyStructure: { childNodes: [] },
    size: 1000,
    modseq: 100n,
  };

  beforeEach(async () => {
    const mockClient = {
      getMailboxLock: jest.fn().mockResolvedValue({ release: jest.fn() }),
      mailbox: { exists: 10 },
      search: jest.fn().mockResolvedValue([123, 124, 125]),
      fetch: jest.fn().mockReturnValue({
        [Symbol.asyncIterator]: async function* () {
          yield mockMessage;
        },
      }),
      fetchOne: jest.fn().mockResolvedValue(mockMessage),
      messageMove: jest.fn().mockResolvedValue(true),
      messageCopy: jest.fn().mockResolvedValue({ uidMap: new Map([[123, 456]]) }),
      messageFlagsAdd: jest.fn().mockResolvedValue(true),
      messageFlagsRemove: jest.fn().mockResolvedValue(true),
      messageDelete: jest.fn().mockResolvedValue(true),
      status: jest.fn().mockResolvedValue({ highestModseq: 200n }),
      download: jest.fn().mockResolvedValue({
        content: { on: jest.fn(), [Symbol.asyncIterator]: async function* () { yield Buffer.from('test'); } },
        meta: { contentType: 'text/plain', expectedSize: 4 },
      }),
    };

    mockPool = {
      withConnection: jest.fn().mockImplementation(async (fn) => fn(mockClient)),
      acquireConnection: jest.fn().mockResolvedValue(mockClient),
      releaseConnection: jest.fn(),
    } as any;

    mockFolderService = {
      resolveFolder: jest.fn().mockImplementation(async (folder) => folder),
      pathToCanonicalFolder: jest.fn().mockImplementation((path) => path),
    } as any;

    const module: TestingModule = await Test.createTestingModule({
      providers: [
        ImapMessageService,
        { provide: ImapConnectionPoolService, useValue: mockPool },
        { provide: ImapFolderService, useValue: mockFolderService },
      ],
    }).compile();

    service = module.get<ImapMessageService>(ImapMessageService);
  });

  describe('listMessages', () => {
    it('should list messages from folder', async () => {
      const result = await service.listMessages({ folder: 'INBOX', pageSize: 50 });
      expect(result.messages).toHaveLength(1);
      expect(mockFolderService.resolveFolder).toHaveBeenCalledWith('INBOX');
    });

    it('should handle pagination', async () => {
      const result = await service.listMessages({ folder: 'INBOX', pageSize: 10 });
      expect(result.hasMore).toBeDefined();
    });
  });

  describe('getMessage', () => {
    it('should get message by ID', async () => {
      const message = await service.getMessage('INBOX:123');
      expect(message).toBeDefined();
      expect(message?.id).toBe('INBOX:123');
    });

    it('should return null for non-existent message', async () => {
      mockPool.withConnection = jest.fn().mockImplementation(async (fn) => {
        const client = {
          getMailboxLock: jest.fn().mockResolvedValue({ release: jest.fn() }),
          fetchOne: jest.fn().mockResolvedValue(null),
        };
        return fn(client);
      });
      const message = await service.getMessage('INBOX:999');
      expect(message).toBeNull();
    });
  });

  describe('parseMessageId', () => {
    it('should parse folder:uid format', () => {
      const [folder, uid] = service['parseMessageId']('INBOX:123');
      expect(folder).toBe('INBOX');
      expect(uid).toBe(123);
    });

    it('should handle nested folder paths', () => {
      const [folder, uid] = service['parseMessageId']('[Gmail]/Sent:456');
      expect(folder).toBe('[Gmail]/Sent');
      expect(uid).toBe(456);
    });

    it('should throw for invalid UID', () => {
      expect(() => service['parseMessageId']('INBOX:invalid')).toThrow();
    });
  });

  describe('moveMessage', () => {
    it('should move message to target folder', async () => {
      await service.moveMessage('INBOX:123', 'TRASH');
      expect(mockFolderService.resolveFolder).toHaveBeenCalledWith('TRASH');
    });
  });

  describe('batch operations', () => {
    it('should group by folder for efficiency', async () => {
      await service.batchSetReadStatus([
        { id: 'INBOX:1', isRead: true },
        { id: 'INBOX:2', isRead: true },
        { id: 'Sent:3', isRead: false },
      ]);
      // Should process INBOX and Sent separately
      expect(mockPool.withConnection).toHaveBeenCalled();
    });
  });

  describe('sync token handling', () => {
    it('should parse multi-folder sync token', () => {
      const result = service['parseSyncToken']('INBOX:100,SENT:200');
      expect(result.get('INBOX')).toBe(100n);
      expect(result.get('SENT')).toBe(200n);
    });

    it('should parse legacy single number token', () => {
      const result = service['parseSyncToken']('12345');
      expect(result.get('INBOX')).toBe(12345n);
    });

    it('should build multi-folder sync token', () => {
      const modSeqs = new Map([['INBOX', 100n], ['SENT', 200n]]);
      const token = service['buildSyncToken'](modSeqs);
      expect(token).toContain('INBOX:100');
      expect(token).toContain('SENT:200');
    });
  });

  describe('getChangesSince', () => {
    it('should return changes since sync token', async () => {
      const result = await service.getChangesSince('INBOX:50');
      expect(result.changes).toBeDefined();
      expect(result.newSyncToken).toBeDefined();
    });
  });
});

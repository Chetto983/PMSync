export class HashSearchService {
  constructor(private readonly tenantKeyService: TenantKeyService) {}

  /**
   * Build a Prisma WHERE filter for exact sender match using HMAC blind index.
   * Returns dual-path: hash lookup + plaintext fallback for unbackfilled rows.
   *
   * @param tenantId - Tenant for HMAC key derivation
   * @param senderEmail - Single email or array of emails to match
   * @returns Prisma.EmailWhereInput with OR clause
   */
  buildExactSenderFilter(
    tenantId: string,
    senderEmail: string | string[],
  ): Prisma.EmailWhereInput;
}

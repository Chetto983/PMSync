    ```typescript
    import { create } from 'zustand';

    interface TodayFeedState {
      dismissedIds: Set<string>;
      dismiss: (id: string) => void;
      restore: (id: string) => void;
      isDismissed: (id: string) => boolean;
      reset: () => void;
    }

    export const useTodayFeedStore = create<TodayFeedState>((set, get) => ({
      dismissedIds: new Set<string>(),
      dismiss: (id) => set((s) => {
        const next = new Set(s.dismissedIds);
        next.add(id);
        return { dismissedIds: next };
      }),
      restore: (id) => set((s) => {
        const next = new Set(s.dismissedIds);
        next.delete(id);
        return { dismissedIds: next };
      }),
      isDismissed: (id) => get().dismissedIds.has(id),
      reset: () => set({ dismissedIds: new Set<string>() }),
    }));
    ```

    **(7) frontend/hooks/queries/use-today.ts** — full file:
    ```typescript
    import { queryOptions, useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
    import { apiClient } from '@/lib/api/api-client';
    import { dashboardApi, type TodayFeedItem, type TodayFeedResponse, type GetTodayFeedQuery } from '@/lib/api/dashboard';
    import { todayKeys } from './today-keys';

    export function todayFeedOptions(opts?: GetTodayFeedQuery) {
      return queryOptions({
        queryKey: todayKeys.feed(opts),
        queryFn: async (): Promise<TodayFeedResponse> => {
          const res = await dashboardApi.getTodayFeed(opts);
          // Backend response interceptor wraps payloads in { data: ... } — unwrap once.
          return (res.data as any).data ?? (res.data as TodayFeedResponse);
        },
        staleTime: 5 * 60 * 1000,
        gcTime: 30 * 60 * 1000,
        retry: 1,
      });
    }

    export function useTodayFeed(opts?: GetTodayFeedQuery) {
      return useQuery(todayFeedOptions(opts));
    }

    export function useCompleteFeedItem() {
      const qc = useQueryClient();
      return useMutation({
        mutationFn: async (item: TodayFeedItem) => {
          if (!item.canComplete || !item.taskSourceType || !item.taskSourceId) {
            throw new Error('Item is not completable');
          }
          await apiClient.post(`/tasks/${item.taskSourceType}/${item.taskSourceId}/complete`);
        },
        onSuccess: () => qc.invalidateQueries({ queryKey: todayKeys.all }),
      });
    }

    export function useSnoozeFeedItem() {
      const qc = useQueryClient();
      return useMutation({
        mutationFn: async ({ item, until }: { item: TodayFeedItem; until: Date }) => {
          // Per Open Q1 (RESEARCH): snooze persists ONLY for follow-up-derived items in MVP.
          if (!item.canSnooze) throw new Error('Item is not snoozeable');
          if (item.taskSourceType !== 'follow_up' || !item.taskSourceId) {
            throw new Error('Snooze is only persisted for follow-up items in this release');
          }
          await apiClient.patch(`/ai/follow-ups/${item.taskSourceId}/snooze`, { dueDate: until.toISOString() });
        },
        onSuccess: () => qc.invalidateQueries({ queryKey: todayKeys.all }),
      });
    }
    ```

    **(1) frontend/components/dashboard/today/TodayFeedList.tsx** — Renders the three tier sections in order with framer-motion AnimatePresence wrapping each row. Filters out `useTodayFeedStore().dismissedIds` set client-side. Props: `{ items: TodayFeedItem[]; counts: TodayFeedCounts }`.

    Structure:
    ```typescript
    'use client';
    import { Box } from '@mui/material';
    import { AnimatePresence, motion } from 'framer-motion';
    import { useTodayFeedStore } from '@/stores/today-feed-store';
    import type { TodayFeedItem, TodayFeedCounts, TodayTier } from '@/lib/api/dashboard';
    import { TodayTierSection } from './TodayTierSection';
    import { TodayFeedItem as Row } from './TodayFeedItem';

    interface Props { items: TodayFeedItem[]; counts: TodayFeedCounts; }
    const TIER_ORDER: TodayTier[] = ['urgent', 'today', 'fyi'];

    export function TodayFeedList({ items, counts }: Props) {
      const dismissedIds = useTodayFeedStore((s) => s.dismissedIds);
      const visible = items.filter((i) => !dismissedIds.has(i.id));
      const byTier: Record<TodayTier, TodayFeedItem[]> = { urgent: [], today: [], fyi: [] };
      for (const it of visible) byTier[it.tier].push(it);

      return (
        <Box role="list" sx={{ display: 'flex', flexDirection: 'column', gap: 3 }}>
          {TIER_ORDER.map((tier) => {
            const tierItems = byTier[tier];
            if (tierItems.length === 0) return null;
            return (
              <TodayTierSection key={tier} tier={tier} count={tierItems.length}>
                <Box sx={{ display: 'flex', flexDirection: 'column', gap: 0.5 }}>
                  <AnimatePresence initial={false}>
                    {tierItems.map((item) => (
                      <motion.div
                        key={item.id}
                        layout
                        initial={{ opacity: 0, x: -20 }}
                        animate={{ opacity: 1, x: 0 }}
                        exit={{ opacity: 0, x: -20, height: 0 }}
                        transition={{ duration: 0.2, ease: [0.4, 0, 0.2, 1] }}
                      >
                        <Row item={item} />
                      </motion.div>
                    ))}
                  </AnimatePresence>
                </Box>
              </TodayTierSection>
            );
          })}
        </Box>
      );
    }
    ```

    **(2) frontend/components/dashboard/today/TodayDashboard.tsx** — Top-level container. Three-state render: loading -> Skeleton, error -> QueryErrorAlert, empty -> TodayEmptyState, otherwise -> Greeting + TodayFeedList. Calls `useTodayFeed({ tz, includeGreeting: true })`. ~120 lines.

    ```typescript
    'use client';
    import { Stack } from '@mui/material';
    import { PageContainer } from '@/components/common/PageContainer';
    import { QueryErrorAlert } from '@/components/common/QueryErrorAlert';
    import { useTranslations } from '@/lib/hooks/use-translations';
    import { useUserTimezone } from '@/hooks/use-user-timezone';
    import { useTodayFeed } from '@/hooks/queries/use-today';
    import { useTodayFeedStore } from '@/stores/today-feed-store';
    import { TodayLoadingSkeleton } from './TodayLoadingSkeleton';
    import { TodayEmptyState } from './TodayEmptyState';
    import { TodayGreeting } from './TodayGreeting';
    import { TodayFeedList } from './TodayFeedList';

    export function TodayDashboard() {
      const t = useTranslations();
      const tz = useUserTimezone();
      const { data, isLoading, error, refetch } = useTodayFeed({ tz, includeGreeting: true });
      const dismissedIds = useTodayFeedStore((s) => s.dismissedIds);

      const visibleItems = (data?.items ?? []).filter((i) => !dismissedIds.has(i.id));
      // Recompute counts AFTER applying client-side dismiss for greeting accuracy.
      const liveCounts = data
        ? {
            urgent: visibleItems.filter((i) => i.tier === 'urgent').length,
            today: visibleItems.filter((i) => i.tier === 'today').length,
            fyi: visibleItems.filter((i) => i.tier === 'fyi').length,
            total: visibleItems.length,
          }
        : { urgent: 0, today: 0, fyi: 0, total: 0 };

      const isEmpty = !isLoading && !error && visibleItems.length === 0;

      return (
        <PageContainer
          title={t.dashboard.today.title}
          subtitle={t.dashboard.today.subtitle}
          maxWidth={720}
        >
          <Stack spacing={3}>
            {isLoading && <TodayLoadingSkeleton />}
            {error && (
              <QueryErrorAlert error={error} onRetry={() => refetch()} message={t.dashboard.today.error} />
            )}
            {!isLoading && !error && data && (
              <>
                {data.greeting !== null || liveCounts.total > 0 ? (
                  <TodayGreeting greeting={data.greeting} counts={liveCounts} />
                ) : null}
                {isEmpty ? (
                  <TodayEmptyState nextItem={data.items[0] ?? undefined} />
                ) : (
                  <TodayFeedList items={data.items} counts={liveCounts} />
                )}
              </>
            )}
          </Stack>
        </PageContainer>
      );
    }
    ```

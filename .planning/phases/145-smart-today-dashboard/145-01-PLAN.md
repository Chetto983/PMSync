    import { IsBoolean, IsIn, IsOptional, IsString, Validate, ValidatorConstraint, ValidatorConstraintInterface, ValidationArguments } from 'class-validator';
    import { Transform } from 'class-transformer';

    @ValidatorConstraint({ name: 'isIanaTimezone', async: false })
    export class IsIanaTimezone implements ValidatorConstraintInterface {
      validate(value: unknown): boolean {
        if (typeof value !== 'string') return false;
        try {
          // Allowlist via Intl — throws RangeError on bad tz
          const list = (Intl as any).supportedValuesOf?.('timeZone') as string[] | undefined;
          if (list) return list.includes(value);
          // Fallback: probe with DateTimeFormat
          new Intl.DateTimeFormat('en-US', { timeZone: value });
          return true;
        } catch { return false; }
      }
      defaultMessage(_args: ValidationArguments) { return 'tz must be a valid IANA timezone identifier'; }
    }

    **(3) backend/src/modules/dashboard/services/today-scorer.service.ts** — `@Injectable()` class implementing the algorithm in RESEARCH.md lines 590-642 verbatim:
    ```typescript
    import { Injectable } from '@nestjs/common';
    import { TodayItemType, TodayTier } from '../dto/today-feed.dto';

    export interface ScoreInput {
      type: TodayItemType;
      dueAt: Date | null;
      sourcePriority?: 'low' | 'medium' | 'high' | 'urgent';
      isOverdue: boolean;
      isHardDeadline?: boolean;
      isStuck?: boolean;
    }

    @Injectable()
    export class TodayScorerService {
      scoreItem(input: ScoreInput, now: Date = new Date()): { tier: TodayTier; score: number } {
        if (input.isOverdue) return { tier: 'urgent', score: this.scoreUrgent(input, now) };
        if (input.sourcePriority === 'urgent') return { tier: 'urgent', score: this.scoreUrgent(input, now) };
        if (input.dueAt && this.isToday(input.dueAt, now)) return { tier: 'today', score: this.scoreToday(input, now) };
        if (input.isHardDeadline && input.dueAt && this.isToday(input.dueAt, now)) return { tier: 'today', score: this.scoreToday(input, now) };
        if (input.isStuck) return { tier: 'fyi', score: this.scoreFyi(input, now) };
        return { tier: 'fyi', score: this.scoreFyi(input, now) };
      }
      scoreUrgent(i: { dueAt: Date | null }, now: Date): number {
        if (!i.dueAt) return 500;
        const overdueHours = Math.max(0, (now.getTime() - i.dueAt.getTime()) / 3_600_000);
        return Math.min(1000, 800 + overdueHours);
      }
      scoreToday(i: { dueAt: Date | null }, now: Date): number {
        if (!i.dueAt) return 400;
        const minsUntil = (i.dueAt.getTime() - now.getTime()) / 60_000;
        return Math.max(100, 700 - minsUntil);
      }
      scoreFyi(i: { dueAt: Date | null }, now: Date): number {
        if (!i.dueAt) return 50;
        const hoursUntil = (i.dueAt.getTime() - now.getTime()) / 3_600_000;
        return Math.max(10, 100 - hoursUntil);
      }
      isToday(d: Date, now: Date): boolean {
        return d.getFullYear() === now.getFullYear()
          && d.getMonth() === now.getMonth()
          && d.getDate() === now.getDate();
      }
    }
    ```

      async getTodayFeed(tenantId: string, userId: string, opts: GetTodayOpts = {}): Promise<TodayFeedResponseDto> {

        const tz = opts.tz || 'UTC';
        const locale = opts.locale || 'en';
        const { startOfDay, endOfDay, tomorrowEod, sevenDaysFromNow, fourteenDaysAgo } = this.computeWindows(now, tz);

        // Promise.all + .catch on each member: a single rejection does NOT kill the feed.

      private computeWindows(now: Date, tz: string) {
        // Copy verbatim from briefing.service.ts:376-388 (tz resolution).
        const todayStr = now.toLocaleDateString('en-CA', { timeZone: tz });
        const parts = todayStr.split('-').map(Number);
        const naive = new Date(Date.UTC(parts[0], parts[1] - 1, parts[2]));
        const probe = new Date(naive.toLocaleString('en-US', { timeZone: tz }));
        const offsetMs = probe.getTime() - naive.getTime();
        const startOfDay = new Date(naive.getTime() - offsetMs);

        const sevenDaysFromNow = new Date(now.getTime() + 7 * 24 * 60 * 60 * 1000);
        const fourteenDaysAgo = new Date(now.getTime() - STUCK_DEAL_DAYS * 24 * 60 * 60 * 1000);
        return { startOfDay, endOfDay, tomorrowEod, sevenDaysFromNow, fourteenDaysAgo };
      }

      private normalizeTasks(tasks: any[], now: Date): TodayFeedItemDto[] {

              taskSourceId: r.id,
              dealId: null, eventId: null,
              canComplete: false, canDismiss: false, canSnooze: true,
            };
          });
      }

      private normalizeDeals(deals: any[], now: Date, isStuck: boolean): TodayFeedItemDto[] {
        return deals.map((d) => {
          const dueAt = d.expectedCloseDate ? new Date(d.expectedCloseDate) : null;

            dealId: d.id, eventId: null,
            canComplete: false, canDismiss: false, canSnooze: false,
          };
        });
      }

            dealId: null, eventId: e.id,
            canComplete: false, canDismiss: false, canSnooze: false,
          };
        });
      }
    }

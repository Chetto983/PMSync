```typescript
export type TaskSourceType = 'follow_up' | 'reminder' | 'deal_action' | 'ticket' | 'manual';
export type UnifiedTaskStatus = 'pending' | 'in_progress' | 'completed' | 'dismissed' | 'expired';
export type UnifiedTaskPriority = 'low' | 'medium' | 'high' | 'urgent';

export interface UnifiedTask {
  id: string;
  sourceType: TaskSourceType;
  sourceId: string;
  title: string;
  description: string | null;
  status: UnifiedTaskStatus;
  priority: UnifiedTaskPriority;
  dueDate: string | null;
  contactName: string | null;
  contactId: string | null;
  dealId: string | null;
  emailId: string | null;
  ticketId: string | null;
  createdAt: string;
  updatedAt: string | null;
}

export interface UnifiedTaskListResponse {
  data: UnifiedTask[];
  meta: { total: number; page: number; limit: number; counts: Record<TaskSourceType, number> };
}

export interface ListTasksQuery {
  sourceType?: TaskSourceType[];
  status?: UnifiedTaskStatus[];
  priority?: UnifiedTaskPriority[];
  dueBefore?: string;
  dueAfter?: string;
  contactId?: string;
  page?: number;
  limit?: number;
  sortBy?: 'dueDate' | 'createdAt' | 'priority';
  sortOrder?: 'asc' | 'desc';
}

export interface CreateManualTaskDto {
  title: string; description?: string; priority?: UnifiedTaskPriority;
  dueDate?: string; contactId?: string; dealId?: string; emailId?: string;
}

export const tasksApi = { list, create, update, delete, complete };
```

**Create `frontend/components/dashboard/tasks/TasksKanbanBoard.tsx`** (~250 lines) following `KanbanBoard.tsx` pattern:

Columns represent statuses: `['pending', 'in_progress', 'completed']` (skip dismissed/expired — those are terminal states users don't drag to).

Props: `{ tasks: UnifiedTask[]; onTaskClick?: (task: UnifiedTask) => void }`

The component receives tasks from the parent (TasksList) — it does NOT fetch its own data. This avoids double-fetching.

**StatusColumn** sub-component (internal, like StageColumn):
- Uses `useDroppable({ id: status })` 
- Header: colored dot (pending=warning.main, in_progress=info.main, completed=success.main) + translated status label + count
- Body: list of TaskCards filtered by `task.status === status`
- Styling: same column layout as StageColumn (minWidth: 280, maxWidth: 320, flex: 0 0 280px, border, borderRadius: 2)

**DnD logic** (following KanbanBoard pattern exactly):
- `useSensors` with PointerSensor (distance: 8) + KeyboardSensor
- `onDragStart`: set `activeDeal` equivalent (`activeTask`)
- `onDragOver`: set `overColumnId`
- `onDragEnd`: 
  - Extract `task` from `active.data.current.task`
  - Extract `newStatus` from `over.id`
  - If `task.status === newStatus`, return (no-op)
  - **CRITICAL constraint**: If `task.sourceType !== 'manual'` AND `newStatus !== 'completed'`:
    - Reject the drop. Show `toast.error(t.tasks.kanban.dragRestricted)`. Return.
  - If `task.sourceType !== 'manual'` AND `newStatus === 'completed'`:
    - Call `completeTask.mutateAsync({ sourceType: task.sourceType, id: task.sourceId })`
  - If `task.sourceType === 'manual'`:
    - If `newStatus === 'completed'`: call `completeTask.mutateAsync({ sourceType: 'manual', id: task.sourceId })`
    - Else: call `updateTask.mutateAsync({ sourceType: 'manual', id: task.sourceId, dto: { status: newStatus } })`
  - On success: `toast.success(t.tasks.kanban.movedSuccess.replace('{status}', t.tasks.statuses[newStatus]))` then `queryClient.invalidateQueries({ queryKey: tasksKeys.all })`
  - On error: `toast.error(t.tasks.kanban.moveError)`
  - NOTE: Do NOT attempt optimistic cache updates — this component receives tasks as a prop and does not own the query key. Use `invalidateQueries` on `tasksKeys.all` to trigger a refetch instead. This is simpler and always correct.

- `DragOverlay`: Show a simplified task card (title + priority) with `opacity: 0.9, boxShadow: 3, border: primary.main`

Import `useUpdateTask, useCompleteTask` from hooks, `useQueryClient` from tanstack, `tasksKeys` from keys file.

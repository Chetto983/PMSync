
2. **Add to constructor**:
   ```typescript

   ```

3. **Replace inline retry in syncFoldersToDatabase** (around lines 91-138):

   BEFORE:
   ```typescript
   let retryCount = 0;
   while (true) {
     try {
       const folder = await this.prisma.folder.upsert({ ... });
       // ...
     } catch (error) {
       const isPrismaError = error && typeof error === 'object' && 'code' in error;
       const errorCode = isPrismaError ? (error as { code: string }).code : null;
       if ((errorCode === 'P2002' || errorCode === 'P2034') && retryCount < MAX_UPSERT_RETRIES) {
         retryCount++;
         await new Promise(resolve => setTimeout(resolve, 50 * retryCount));
         continue;
       }
       throw error;
     }
   }
   ```

   AFTER:
   ```typescript
   const folder = await this.prismaRetry.withRetry(
     () => this.prisma.folder.upsert({
       where: { providerId_path: { providerId, path: folderInfo.path } },
       create: { ... },
       update: { ... },
     }),
     { maxAttempts: MAX_UPSERT_RETRIES, loggerName: 'FolderSync' }
   );
   results.push(folder);
   ```

4. **Replace inline retry in parent update loop** (around lines 158-175):

   BEFORE:
   ```typescript
   let retryCount = 0;
   while (retryCount < MAX_UPSERT_RETRIES) {
     try {
       await this.prisma.folder.update({ ... });
       break;
     } catch (error) {
       // inline retry logic
     }
   }
   ```

   AFTER:
   ```typescript
   await this.prismaRetry.withRetry(
     () => this.prisma.folder.update({
       where: { id_tenantId: { id: folderId, tenantId } },
       data: { parentId },
     }),
     { maxAttempts: MAX_UPSERT_RETRIES, loggerName: 'FolderSync' }
   );
   ```

5. **Remove the MAX_UPSERT_RETRIES constant** if it's only used for these retries and use PrismaRetryService defaults instead. If used elsewhere, keep it.

```typescript
import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { PrismaService } from '../../../prisma/prisma.service';
import {
  ScheduledJobsQueueService,
  ScheduledJobType,
} from '../../../common/services/scheduled-jobs.queue';

@Injectable()
export class OrphanEmbeddingCleanupService implements OnModuleInit {
  private readonly logger = new Logger(OrphanEmbeddingCleanupService.name);
  private readonly BATCH_SIZE = 500;
  private readonly jobsEnabled: boolean;

  constructor(
    private readonly prisma: PrismaService,
    private readonly configService: ConfigService,
    private readonly scheduledJobsQueue: ScheduledJobsQueueService,
  ) {
    this.jobsEnabled =
      (this.configService.get<string>('JOBS_ENABLED') || 'true').toLowerCase() !== 'false';
  }

  async onModuleInit() {
    if (!this.jobsEnabled) {
      this.logger.warn('OrphanEmbeddingCleanupService disabled via JOBS_ENABLED=false');
      return;
    }

    this.scheduledJobsQueue.registerHandler(
      ScheduledJobType.ORPHAN_EMBEDDING_CLEANUP,
      async () => this.cleanupOrphanEmbeddings(),
    );

    this.logger.log('OrphanEmbeddingCleanupService registered with BullMQ scheduler');
  }

  /**
   * Cleanup embeddings with null emailId AND null contactId
   * These are orphans from deleted content that were not properly cleaned up
   */
  async cleanupOrphanEmbeddings(): Promise<{ deleted: number; batches: number }> {
    if (!this.jobsEnabled) {
      return { deleted: 0, batches: 0 };
    }

    let totalDeleted = 0;
    let batches = 0;

    while (true) {
      // Use indexed fields (emailId, contactId) - NOT metadata JSONB queries
      const orphans = await this.prisma.embedding.findMany({
        where: {
          emailId: null,
          contactId: null,
        },
        select: { id: true, tenantId: true },
        take: this.BATCH_SIZE,
      });

      if (orphans.length === 0) {
        break;
      }

      // Delete in batch using compound key
      const deleted = await this.prisma.embedding.deleteMany({
        where: {
          OR: orphans.map(o => ({ id: o.id, tenantId: o.tenantId })),
        },
      });

      totalDeleted += deleted.count;
      batches++;

      this.logger.log(`Deleted ${deleted.count} orphan embeddings (batch ${batches})`);

      // Safety limit: max 20 batches per run (10k embeddings)
      if (batches >= 20) {
        this.logger.warn('Reached batch limit (20), stopping cleanup. Will continue next run.');
        break;
      }
    }

    this.logger.log(`Orphan embedding cleanup complete: ${totalDeleted} deleted in ${batches} batches`);
    return { deleted: totalDeleted, batches };
  }
}
```

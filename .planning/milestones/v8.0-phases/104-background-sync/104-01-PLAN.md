**2. Create `frontend/lib/offline-action-queue.ts`**

The queue stores actions in a SEPARATE idb-keyval store from the query persist store. Use `DB_NAME = 'pmsync-offline'` (same DB) but `STORE_NAME = 'pmsync-action-queue'` (different store).

```typescript
import { get, set, del, clear, keys, createStore } from 'idb-keyval';

const DB_NAME = 'pmsync-offline';
const STORE_NAME = 'pmsync-action-queue';
const actionStore = createStore(DB_NAME, STORE_NAME);

const TTL_MS = 30 * 60 * 1000; // 30 minutes
const MAX_QUEUE_SIZE = 100;

export interface QueuedAction {
  id: string;
  type: 'update-email';
  emailId: string;
  dto: Record<string, unknown>;
  queuedAt: number;
}

export async function enqueueAction(
  action: Omit<QueuedAction, 'id' | 'queuedAt'>
): Promise<string> {
  // Dedup: if same emailId exists, merge dto fields
  const existing = await findByEmailId(action.emailId);
  if (existing) {
    const merged: QueuedAction = {
      ...existing,
      dto: { ...existing.dto, ...action.dto },
      queuedAt: Date.now(), // refresh timestamp on merge
    };
    await set(existing.id, merged, actionStore);
    return existing.id;
  }

  // Cap enforcement: if at MAX_QUEUE_SIZE, remove oldest
  const allKeys = await keys(actionStore);
  if (allKeys.length >= MAX_QUEUE_SIZE) {
    const all = await getAllActions();
    // Remove oldest (first in sorted array)
    if (all.length > 0) {
      await del(all[0].id, actionStore);
    }
  }

  const id = crypto.randomUUID();
  const entry: QueuedAction = { ...action, id, queuedAt: Date.now() };
  await set(id, entry, actionStore);
  return id;
}

export async function getAllActions(): Promise<QueuedAction[]> {
  const allKeys = await keys(actionStore);
  const actions: QueuedAction[] = [];
  for (const key of allKeys) {
    const action = await get<QueuedAction>(key, actionStore);
    if (action) actions.push(action);
  }
  return actions.sort((a, b) => a.queuedAt - b.queuedAt);
}

export async function removeAction(id: string): Promise<void> {
  await del(id, actionStore);
}

export async function clearActionQueue(): Promise<void> {
  await clear(actionStore);
}

export function isExpired(action: QueuedAction): boolean {
  return Date.now() - action.queuedAt > TTL_MS;
}

async function findByEmailId(emailId: string): Promise<QueuedAction | null> {
  const allKeys = await keys(actionStore);
  for (const key of allKeys) {
    const action = await get<QueuedAction>(key, actionStore);
    if (action && action.emailId === emailId) return action;
  }
  return null;
}

// Export for testing
export { actionStore, TTL_MS, MAX_QUEUE_SIZE };
```

**3. Create `frontend/stores/offline-queue-store.ts`**

Minimal Zustand store for reactive queue count. Follow existing store patterns (see label-store.test.ts for test patterns):

```typescript
import { create } from 'zustand';

interface OfflineQueueState {
  pendingCount: number;
  incrementPending: () => void;
  decrementPending: () => void;
  setPendingCount: (count: number) => void;
  reset: () => void;
}

export const useOfflineQueueStore = create<OfflineQueueState>((set) => ({
  pendingCount: 0,
  incrementPending: () => set((s) => ({ pendingCount: s.pendingCount + 1 })),
  decrementPending: () => set((s) => ({ pendingCount: Math.max(0, s.pendingCount - 1) })),
  setPendingCount: (count: number) => set({ pendingCount: count }),
  reset: () => set({ pendingCount: 0 }),
}));
```

mutationFn: async ({ emailId, dto }: { emailId: string; dto: UpdateEmailDto }) => {
  if (!navigator.onLine) {
    // Queue for replay when online (SYNC-01)
    await enqueueAction({ type: 'update-email', emailId, dto });
    useOfflineQueueStore.getState().incrementPending();
    // Return fake response -- optimistic data already applied in onMutate
    return { id: emailId, ...dto } as Email;

  try {
    return await emailsApi.updateEmail(emailId, dto);
  } catch (err) {
    // Network error while supposedly online (captive portal, flaky connection)
    // Queue the action instead of failing (Pitfall 1 from RESEARCH.md)
    if (err instanceof Error && (err.message.includes('Network') || err.message.includes('fetch'))) {
      await enqueueAction({ type: 'update-email', emailId, dto });
      useOfflineQueueStore.getState().incrementPending();
      return { id: emailId, ...dto } as Email;
    }
    throw err; // Re-throw non-network errors (4xx, 5xx)
  }
},

```typescript
import { defaultCache } from '@serwist/next/worker';
import type { PrecacheEntry, SerwistGlobalConfig } from 'serwist';
import { Serwist } from 'serwist';

declare global {
  interface WorkerGlobalScope extends SerwistGlobalConfig {
    __SW_MANIFEST: (PrecacheEntry | string)[] | undefined;
  }
}

declare const self: ServiceWorkerGlobalScope;

// Filter out any defaultCache entries that match /api/ routes.
// Serwist's defaultCache includes a NetworkFirst rule for /api/ which
// would cache API responses and break httpOnly cookie auth + SSE streams.
const filteredCache = defaultCache.filter((entry) => {
  const pattern = entry.urlPattern;
  if (pattern instanceof RegExp) return !pattern.source.includes('api');
  if (typeof pattern === 'string') return !pattern.includes('/api');
  return true;
});

const serwist = new Serwist({
  precacheEntries: self.__SW_MANIFEST,
  skipWaiting: true,
  clientsClaim: true,
  navigationPreload: true,
  navigateFallbackDenylist: [/^\/api\//],
  runtimeCaching: [
    // CRITICAL: /api/* uses NetworkOnly to prevent ANY SW interception.
    // This MUST come before ...filteredCache to take priority.
    // Protects: httpOnly cookie auth, SSE streaming, TanStack Query caching.
    // Anchored regex (^) ensures only paths starting with /api/ are matched,
    // not arbitrary URLs that happen to contain /api/ elsewhere.
    {
      urlPattern: /^\/api\/.*/,
      handler: 'NetworkOnly',
    },
    ...filteredCache,
  ],
});

serwist.addEventListeners();
```

Key decisions in this file:
- `skipWaiting: true` + `clientsClaim: true`: New SW activates immediately (Phase 105 will add user-facing "update available" notification)
- `navigationPreload: true`: Speeds up navigation requests by making the network request in parallel with SW boot
- `navigateFallbackDenylist: [/^\/api\//]`: Prevents SW from serving a cached HTML page for /api/ navigation requests
- NetworkOnly for `/^\/api\/.*/` placed FIRST in runtimeCaching array: Takes priority over any remaining defaultCache entries. The `^` anchor is critical — it ensures consistency with `navigateFallbackDenylist` and prevents edge-case bypasses where `/api/` appears in a non-root path segment.
- `filteredCache`: Defensively removes any /api/ related entries from defaultCache before spreading

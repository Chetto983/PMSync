3. Generate PWA icons from `frontend/public/Logo PMSync.svg` using sharp (install as dev dep if needed: `npm i -D sharp`):
   - Create `frontend/public/icons/` directory
   - **IMPORTANT: Before generating icons, run an early diagnostic to verify sharp can load the SVG.** Call `sharp(srcSvg).metadata()` and log the result. If it throws (e.g., librsvg absent), stop immediately with a clear error message instead of producing 4 corrupt files silently.
   - Generate 4 PNGs using a Node.js script:
     - `icon-192x192.png`: 192x192, transparent background, logo centered
     - `icon-512x512.png`: 512x512, transparent background, logo centered
     - `icon-maskable-192x192.png`: 192x192, background `#0c1017`, logo within central 80% (40px padding each side = content area 112x112, centered)
     - `icon-maskable-512x512.png`: 512x512, background `#0c1017`, logo within central 80% (51px padding each side = content area 410x410, centered)
   - Use sharp to read the SVG, resize, and composite onto backgrounds.
   - Delete the generation script after use.

   Example generation script (run with node, then delete):
   ```javascript
   const sharp = require('sharp');
   const path = require('path');
   const fs = require('fs');

   const srcSvg = path.join(__dirname, 'public/Logo PMSync.svg');
   const outDir = path.join(__dirname, 'public/icons');
   fs.mkdirSync(outDir, { recursive: true });

   async function generate() {
     // Diagnostic: verify sharp can process the SVG before generating all icons
     const meta = await sharp(srcSvg).metadata();
     console.log('SVG loaded OK:', meta.format, meta.width + 'x' + meta.height);
     if (!meta.format) throw new Error('sharp failed to detect SVG format — librsvg may be missing');

     // Standard icons - transparent background
     await sharp(srcSvg).resize(192, 192, { fit: 'contain', background: { r: 0, g: 0, b: 0, alpha: 0 } }).png().toFile(path.join(outDir, 'icon-192x192.png'));
     await sharp(srcSvg).resize(512, 512, { fit: 'contain', background: { r: 0, g: 0, b: 0, alpha: 0 } }).png().toFile(path.join(outDir, 'icon-512x512.png'));

     // Maskable icons - dark background, content in safe zone (80%)
     const logo192 = await sharp(srcSvg).resize(112, 112, { fit: 'contain', background: { r: 0, g: 0, b: 0, alpha: 0 } }).png().toBuffer();
     await sharp({ create: { width: 192, height: 192, channels: 4, background: { r: 12, g: 16, b: 23, alpha: 255 } } })
       .composite([{ input: logo192, gravity: 'centre' }]).png().toFile(path.join(outDir, 'icon-maskable-192x192.png'));

     const logo410 = await sharp(srcSvg).resize(410, 410, { fit: 'contain', background: { r: 0, g: 0, b: 0, alpha: 0 } }).png().toBuffer();
     await sharp({ create: { width: 512, height: 512, channels: 4, background: { r: 12, g: 16, b: 23, alpha: 255 } } })
       .composite([{ input: logo410, gravity: 'centre' }]).png().toFile(path.join(outDir, 'icon-maskable-512x512.png'));

     console.log('Icons generated successfully');
   }
   generate();
   ```

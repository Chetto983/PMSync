export interface PushDispatchJob {
  tenantId: string; userId: string; emailId: string;
  subject: string; from: string; threadId: string | null; category: 'email' | 'calendar' | 'ai';
}

export class PushDispatchHandler implements OnModuleInit, OnModuleDestroy {
  private queue!: Queue<PushDispatchJob>;
  private redis!: Redis;
  constructor(private readonly prisma: PrismaService, private readonly pushService: PushService) {}

  async onModuleInit(): Promise<void> {
    const connection = getRedisConnectionOptions();
    this.queue = new Queue<PushDispatchJob>(QUEUE_NAMES.PUSH_DISPATCH, { connection, defaultJobOptions: getDefaultJobOptions('HIGH') });
    this.redis = getSharedRedisConnection();
  }

  @OnEvent(EmailIndexedEvent.eventName)
  async handleEmailIndexed(event: EmailIndexedEvent): Promise<void> {
    // 1. Lookup userId from providerId via prisma.providerConfig.findUnique
    // 2. Check pushEmailEnabled via pushService.getUserPushPreferences
    // 3. Lookup threadId via prisma.email.findFirst
    // 4. If threadId: redis.incr(dedupKey), if count===1: redis.expire + queue.add('push-email-grouped', ..., { delay: 60000 })
    //    If count > 1: return (skip)
    // 5. If no threadId: queue.add('push-email', { ..., threadId: null })
  }
}
```

export class PushDispatchWorker implements OnModuleInit, OnModuleDestroy {
  private worker!: Worker<PushDispatchJob>;
  private redis!: Redis;
  constructor(private readonly pushService: PushService) {}

  async onModuleInit(): Promise<void> {
    this.redis = getSharedRedisConnection();
    this.worker = new Worker<PushDispatchJob>(QUEUE_NAMES.PUSH_DISPATCH,
      async (job) => this.process(job), { connection, concurrency: 5 });
  }

  private async process(job: Job<PushDispatchJob>): Promise<void> {
    // 1. getSubscriptionsForUser(tenantId, userId) via pushService
    // 2. Build payload: if job.name === 'push-email-grouped' && threadId -> grouped title with count from redis.get
    //    else -> single email title "New email from {senderName}"
    // 3. payload.data.url = `/mail/${emailId}`
    // 4. Promise.allSettled fan-out to all subscriptions via pushService.sendNotification
    // 5. On 410/404 error: pushService.deleteStaleSubscription(sub.id, tenantId)
  }
}
```

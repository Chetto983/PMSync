    ```typescript
    /**
     * Batch retrieve memories anchored to multiple entities.
     * Single DB round-trip using OR conditions. Avoids N+1.
     */
    async getMemoriesByEntities(
      tenantId: string,
      userId: string,
      contactIds: string[],
      emailIds: string[],
      limit: number = 10,
    ): Promise<MemoryResult[]> {
      if (contactIds.length === 0 && emailIds.length === 0) return [];

      const orConditions: Prisma.AIMemoryWhereInput[] = [];
      if (contactIds.length > 0) {
        orConditions.push({ contactId: { in: contactIds } });
      }
      if (emailIds.length > 0) {
        orConditions.push({ emailId: { in: emailIds } });
      }

      const memories = await this.prisma.aIMemory.findMany({
        where: { tenantId, userId, OR: orConditions },
        orderBy: { updatedAt: 'desc' },
        take: limit,
        select: { id: true, content: true, category: true, createdAt: true },
      });

      return memories.map((m) => ({
        id: m.id,
        content: m.content,
        category: m.category,
        createdAt: m.createdAt,
      }));
    }
    ```

    const contactIds: string[] = [];
    const emailIds: string[] = [];
    for (const node of graphResults) {
      if (node.externalId) {
        if (node.nodeType === 'contact') contactIds.push(node.externalId);
        else if (node.nodeType === 'email') emailIds.push(node.externalId);
      }
    }

    // Fetch entity-anchored memories and merge with vector memories (entity-first for priority)
    const entityMemories = await this.memoryService.getMemoriesByEntities(
      tenantId, userId, contactIds, emailIds, 10,
    );
    const seenIds = new Set<string>();
    const mergedMemories: import('./memory.service').MemoryResult[] = [];
    for (const m of entityMemories) {
      if (m.id && !seenIds.has(m.id)) { seenIds.add(m.id); mergedMemories.push(m); }
    }
    for (const m of memories) {
      if (m.id && !seenIds.has(m.id)) { seenIds.add(m.id); mergedMemories.push(m); }
    }

      graphResults,
      mergedMemories.length > 0 ? mergedMemories : memories,

      ```typescript
      interface HitlApprovalWrapperProps {

        args: Record<string, unknown>;
        result?: unknown;
      }
      ```

   c. Remove the early return `if (status.type !== 'requires-action') return null;`

   d. Determine the display state using this logic:
      ```typescript
      type HitlDisplayState = 'pending' | 'approved' | 'rejected';

      function getDisplayState(status: { type: string }, result: unknown): HitlDisplayState {
        if (status.type === 'requires-action') return 'pending';
        // After addResult, status.type becomes 'complete'
        const r = result as Record<string, unknown> | undefined;
        if (r && r.approved === true) return 'approved';
        if (r && r.approved === false) return 'rejected';
        // If complete but no approval data, treat as approved (tool executed normally)
        return 'approved';
      }
      ```

   e. Render based on `displayState`:
      - **pending**: Keep current rendering (warning border, approve/reject buttons). Update hardcoded "View parameters" to `t.chat.hitl.viewParameters`. Update Chip label from hardcoded "HIGH" to `t.chat.hitl.highChip`.

   f. Use the UI-SPEC color values:

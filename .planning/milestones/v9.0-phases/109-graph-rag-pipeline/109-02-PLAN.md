```typescript
export interface GraphTraversalResult {
  nodeId: string;
  nodeType: string;
  label: string;
  metadata: Record<string, unknown>;
  hop: number;
  score: number;
  edgeType?: string;
}

export interface TraverseOptions {
  maxHops?: number;   // default 2, max 3
  limit?: number;     // default 20
  edgeTypes?: string[]; // filter by edge type, default all
}

@Injectable()
export class EntityGraphService {
  private readonly logger = new Logger(EntityGraphService.name);
  constructor(private readonly prisma: PrismaService) {}

  async traverse(tenantId: string, seedNodeIds: string[], options?: TraverseOptions): Promise<GraphTraversalResult[]>
  async findSeedNodes(tenantId: string, query: string, limit?: number): Promise<string[]>
  async getNodesByExternalId(tenantId: string, externalIds: string[], nodeType?: string): Promise<GraphTraversalResult[]>
}
```

**traverse()** implementation:
- If seedNodeIds empty, return []
- Clamp maxHops to Math.min(options.maxHops ?? 2, 3)
- Use `Prisma.sql` tagged template for the recursive CTE from RESEARCH.md Pattern 2:
  - Base case: seed nodes with hop=0, score=1.0
  - Recursive case: follow edges bidirectionally, score = `parent_score * edge_weight * power(0.7, hop+1)`
  - Cycle detection: `NOT n2.id = ANY(w.path)`
  - Depth limit: `w.hop < maxHops`
- `DISTINCT ON (id)` with `ORDER BY id, score DESC` to keep highest score per node
- Final `ORDER BY score DESC LIMIT ${limit}`
- If edgeTypes filter provided, add `AND e."edgeType" = ANY(${edgeTypes}::text[])` in recursive case

**findSeedNodes()** implementation:
- Use pg_trgm similarity on `graph_nodes.label` column: `WHERE "tenantId" = $1 AND label % $2 ORDER BY similarity(label, $2) DESC LIMIT $3`
- Default limit: 5
- Returns just the node IDs

**getNodesByExternalId()** implementation:
- Simple Prisma query: `prisma.graphNode.findMany({ where: { tenantId, externalId: { in: externalIds }, ...(nodeType ? { nodeType } : {}) } })`
- Map to GraphTraversalResult with hop=0, score=1.0

```typescript
import { encode } from 'gpt-tokenizer';

export interface AssembledContext {
  graphContext: string;      // ~2400 tokens max
  memoryContext: string;     // ~800 tokens max
  fewShotContext: string;    // ~800 tokens max (populated by Plan 03)
  rawTokenCount: number;     // Total tokens of all raw results before assembly
  assembledTokenCount: number; // Total tokens after structured assembly
}

export interface AssembleOptions {
  tokenBudget?: number;      // default 4096
  graphRatio?: number;       // default 0.6
  memoryRatio?: number;      // default 0.2
  fewShotRatio?: number;     // default 0.2
}

@Injectable()
export class ContextAssemblyService {
  private readonly logger = new Logger(ContextAssemblyService.name);

  countTokens(text: string): number -- uses `encode(text).length` from gpt-tokenizer

  assemble(
    graphResults: GraphTraversalResult[],
    memories: MemoryResult[],
    fewShotExamples: string[],
    options?: AssembleOptions,
  ): AssembledContext

  assembleNodeContext(node: GraphTraversalResult): string
}
```

**assemble()** implementation:
- Calculate budgets: `graphBudget = floor(tokenBudget * graphRatio)`, same for memory and fewShot
- Calculate rawTokenCount: sum `countTokens(JSON.stringify(node))` for all graph results + memories (this represents what raw injection would cost)
- Sort graphResults by score DESC
- For each graph node, call `assembleNodeContext()`, count tokens, add to graphLines until graphBudget exceeded
- For each memory, format as `[Memory] content (category)`, count tokens, add to memoryLines until memoryBudget exceeded
- For each fewShotExample string, count tokens, add until fewShotBudget exceeded
- Return AssembledContext with all sections and token counts

**assembleNodeContext()** implementation (per RESEARCH.md Pattern 3):
- Switch on `node.nodeType`:
  - `'email'`: `[Email, hop ${hop}, via ${edgeType}] Subject: ${metadata.subject} | From: ${metadata.from} | Date: ${metadata.receivedAt} | Snippet: ${metadata.snippet?.slice(0, 100)}`
  - `'contact'`: `[Contact, hop ${hop}] ${label} | ${metadata.company ?? ''} | ${metadata.jobTitle ?? ''} | Score: ${metadata.relationshipScore ?? 'N/A'}`
  - `'thread'`: `[Thread, hop ${hop}] ${label} | ${metadata.messageCount} messages`
  - `'calendar_event'`: `[Event, hop ${hop}] ${label} | ${metadata.startTime} | ${metadata.location ?? ''}`
  - `'topic'`: `[Topic, hop ${hop}] ${label}`
  - `'memory'`: `[Memory] ${label} (${metadata.category ?? 'general'})`

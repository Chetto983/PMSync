**Part B: Add apiPatch and apiDelete to validation-helpers.mjs**

6. Add `apiPatch(context, path, body, cookieHeader, csrfToken)` function to `scripts/lib/validation-helpers.mjs`:
   ```javascript
   export async function apiPatch(context, path, body, cookieHeader, csrfToken) {
     const api = context.request;
     const url = BACKEND_URL + path;
     const headers = { Cookie: cookieHeader };
     if (csrfToken) headers['x-csrf-token'] = csrfToken;
     const opts = { headers };
     if (body !== null && body !== undefined) opts.data = body;
     const response = await api.patch(url, opts);
     const status = response.status();
     let data;
     try { data = await response.json(); } catch { data = await response.text(); }
     if (status < 200 || status >= 300) {
       throw new Error(`PATCH ${path} returned ${status}: ${JSON.stringify(data).substring(0, 200)}`);
     }
     return data;
   }
   ```

7. Add `apiDelete(context, path, cookieHeader, csrfToken)` function:
   ```javascript
   export async function apiDelete(context, path, cookieHeader, csrfToken) {
     const api = context.request;
     const url = BACKEND_URL + path;
     const headers = { Cookie: cookieHeader };
     if (csrfToken) headers['x-csrf-token'] = csrfToken;
     const response = await api.delete(url, { headers });
     const status = response.status();
     if (status === 204) return null;
     let data;
     try { data = await response.json(); } catch { data = await response.text(); }
     if (status < 200 || status >= 300) {
       throw new Error(`DELETE ${path} returned ${status}: ${JSON.stringify(data).substring(0, 200)}`);
     }
     return data;
   }
   ```

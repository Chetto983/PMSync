```typescript
import { test, expect } from '@playwright/test';

test.describe('Dashboard', () => {
  test('renders default widgets', async ({ page }) => {
    await page.goto('/dashboard');
    await page.waitForLoadState('networkidle');

    // Dashboard container should be visible
    const gridLayout = page.locator('.react-grid-layout');
    await expect(gridLayout).toBeVisible({ timeout: 15_000 });

    // At least 3 widget items should be rendered (fresh user gets all 10,
    // but we only assert a minimum to handle partial loading)
    const widgetItems = page.locator('.react-grid-item');
    await expect(widgetItems.first()).toBeVisible({ timeout: 15_000 });
    const count = await widgetItems.count();
    expect(count).toBeGreaterThanOrEqual(3);
  });

  test('customization drawer opens, toggles widget, persists after refresh', async ({ page }) => {
    await page.goto('/dashboard');
    await page.waitForLoadState('networkidle');

    // Wait for grid to fully render
    const gridLayout = page.locator('.react-grid-layout');
    await expect(gridLayout).toBeVisible({ timeout: 15_000 });

    // Count widgets before toggling
    const widgetItems = page.locator('.react-grid-item');
    await expect(widgetItems.first()).toBeVisible({ timeout: 15_000 });
    const countBefore = await widgetItems.count();

    // Find and click the customize button
    const customizeBtn = page.getByRole('button', { name: /customize/i });
    await expect(customizeBtn).toBeVisible({ timeout: 15_000 });
    await customizeBtn.click();

    // Drawer should appear
    const drawer = page.locator('[role="presentation"]');
    await expect(drawer).toBeVisible({ timeout: 5_000 });

    // Toggle a widget OFF: find the first switch/checkbox in the drawer and click it
    // MUI Switch uses role="checkbox" inside the drawer
    const switches = drawer.locator('input[type="checkbox"], [role="switch"]');
    const firstSwitch = switches.first();
    await expect(firstSwitch).toBeVisible({ timeout: 5_000 });
    await firstSwitch.click();

    // Close the drawer
    await page.keyboard.press('Escape');

    // After toggling one widget OFF, count should decrease by 1
    // Wait for the grid to re-render after drawer close
    await expect(gridLayout).toBeVisible({ timeout: 10_000 });
    const countAfterToggle = await widgetItems.count();
    expect(countAfterToggle).toBe(countBefore - 1);

    // Reload the page to test persistence
    await page.reload();
    await page.waitForLoadState('networkidle');
    await expect(gridLayout).toBeVisible({ timeout: 15_000 });

    // Widget count should persist (same reduced count)
    const countAfterReload = await widgetItems.count();
    expect(countAfterReload).toBe(countAfterToggle);
  });
});
```

async onModuleInit() {

    const extended = this.$extends(
      createFieldEncryptionExtension(
        (tenantId) => this.tenantKeyService.getCurrentKey(tenantId),

        (tenantId) => this.tenantKeyService.getHmacKey(tenantId),

  }
  await this.$connect();

}

**2. AccountDeletionService -- add purgeUserDeletion method:**

Add a new method `purgeUserDeletion(userId: string, tenantId: string): Promise<boolean>` to `AccountDeletionService`:

```typescript
/**
 * Purge a single user whose deletion grace period has expired.
 * Called by event handler (scoped to event payload), NOT the daily cron.
 */
async purgeUserDeletion(userId: string, tenantId: string): Promise<boolean> {
  const cutoff = new Date(Date.now() - GRACE_PERIOD_MS);

  const user = await this.prisma.user.findFirst({
    where: { id: userId, tenantId, deletedAt: { not: null, lt: cutoff } },
    select: { id: true },
  });

  if (!user) {
    this.logger.debug(`User ${userId} not found or deletion not yet expired, skipping`);
    return false;
  }

  return this.hardDeleteUser(userId, tenantId);
}
```

Place it after `purgeExpiredDeletions()` and before `hardDeleteUser()`. The existing `purgeExpiredDeletions()` method stays unchanged (still called by daily cron).
